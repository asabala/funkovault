enum AtlasObjectTrackState { tentative, confirmed, cooldown }

class AtlasTrackedBarcodeEvidence {
  const AtlasTrackedBarcodeEvidence({
    required this.key,
    required this.latestCode,
    required this.hitCount,
  });

  final String key;
  final ScanCode latestCode;
  final int hitCount;

  AtlasTrackedBarcodeEvidence addHit(ScanCode code) {
    return AtlasTrackedBarcodeEvidence(
      key: key,
      latestCode: code,
      hitCount: hitCount + 1,
    );
  }

class AtlasObjectTrack {
  AtlasObjectTrack({
    required this.trackId,
    required this.state,
    required this.firstSeenAt,
    required this.lastSeenAt,
    required this.observationCount,
    required this.latestObservation,
    required this.centroid,
    required this.normalizedBounds,
    required Map<String, AtlasTrackedBarcodeEvidence> barcodeEvidenceByKey,
    this.cooldownStartedAt,
  }) : barcodeEvidenceByKey =
           Map<String, AtlasTrackedBarcodeEvidence>.unmodifiable(
             barcodeEvidenceByKey,
           );

  factory AtlasObjectTrack.createInitial({
    required String trackId,
    required AtlasScanObservation observation,
  }) {
    return AtlasObjectTrack(
      trackId: trackId,
      state: AtlasObjectTrackState.tentative,
      firstSeenAt: observation.capturedAt,
      lastSeenAt: observation.capturedAt,
      observationCount: 1,
      latestObservation: observation,
      centroid: observation.centroid,
      normalizedBounds: observation.normalizedBounds,
      barcodeEvidenceByKey: _barcodeEvidenceForObservation(observation),
    );
  }

  final String trackId;
  final AtlasObjectTrackState state;
  final DateTime firstSeenAt;
  final DateTime lastSeenAt;
  final int observationCount;
  final AtlasScanObservation latestObservation;
  final AtlasNormalizedPoint? centroid;
  final AtlasNormalizedBounds? normalizedBounds;
  final Map<String, AtlasTrackedBarcodeEvidence> barcodeEvidenceByKey;
  final DateTime? cooldownStartedAt;

  bool get isTentative => state == AtlasObjectTrackState.tentative;
  bool get isConfirmed => state == AtlasObjectTrackState.confirmed;
  bool get isCooldown => state == AtlasObjectTrackState.cooldown;

  int get usableBarcodeHitCount => barcodeEvidenceByKey.values.fold<int>(
    0,
    (int total, AtlasTrackedBarcodeEvidence evidence) =>
        total + evidence.hitCount,
  );

  AtlasObjectTrack incorporateObservation(
    AtlasScanObservation observation, {
    double centroidSmoothingAlpha = 0.30,
  }) {
    if (!centroidSmoothingAlpha.isFinite ||
        centroidSmoothingAlpha < 0.0 ||
        centroidSmoothingAlpha > 1.0) {
      throw ArgumentError.value(
        centroidSmoothingAlpha,
        'centroidSmoothingAlpha',
        'centroidSmoothingAlpha must be between 0.0 and 1.0.',
      );
    }

    final Map<String, AtlasTrackedBarcodeEvidence> updatedEvidence =
        Map<String, AtlasTrackedBarcodeEvidence>.from(barcodeEvidenceByKey);

    final ScanCode? observedCode = observation.code;
    if (observation.hasUsableProductBarcode && observedCode != null) {
      final String key = barcodeEvidenceKey(observedCode);
      final AtlasTrackedBarcodeEvidence? existingEvidence =
          updatedEvidence[key];
      updatedEvidence[key] = existingEvidence == null
          ? AtlasTrackedBarcodeEvidence(
              key: key,
              latestCode: observedCode,
              hitCount: 1,
            )
          : existingEvidence.addHit(observedCode);
    }

    final AtlasNormalizedPoint? observationCentroid = observation.centroid;
    final AtlasNormalizedPoint? updatedCentroid = _smoothedCentroid(
      current: centroid,
      observation: observationCentroid,
      alpha: centroidSmoothingAlpha,
    );

    return AtlasObjectTrack(
      trackId: trackId,
      state: state,
      firstSeenAt: firstSeenAt,
      lastSeenAt: observation.capturedAt,
      observationCount: observationCount + 1,
      latestObservation: observation,
      centroid: updatedCentroid,
      normalizedBounds: observation.normalizedBounds ?? normalizedBounds,
      barcodeEvidenceByKey: updatedEvidence,
      cooldownStartedAt: cooldownStartedAt,
    );
  }

  AtlasObjectTrack confirm() {
    if (!isTentative) return this;

    return AtlasObjectTrack(
      trackId: trackId,
      state: AtlasObjectTrackState.confirmed,
      firstSeenAt: firstSeenAt,
      lastSeenAt: lastSeenAt,
      observationCount: observationCount,
      latestObservation: latestObservation,
      centroid: centroid,
      normalizedBounds: normalizedBounds,
      barcodeEvidenceByKey: barcodeEvidenceByKey,
      cooldownStartedAt: cooldownStartedAt,
    );
  }

  AtlasObjectTrack startCooldown(DateTime startedAt) {
    return AtlasObjectTrack(
      trackId: trackId,
      state: AtlasObjectTrackState.cooldown,
      firstSeenAt: firstSeenAt,
      lastSeenAt: lastSeenAt,
      observationCount: observationCount,
      latestObservation: latestObservation,
      centroid: centroid,
      normalizedBounds: normalizedBounds,
      barcodeEvidenceByKey: barcodeEvidenceByKey,
      cooldownStartedAt: startedAt,
    );
  }

  static String barcodeEvidenceKey(ScanCode code) {
    return '${code.format.name}|${code.normalizedValue.trim()}';
  }

  static Map<String, AtlasTrackedBarcodeEvidence>
  _barcodeEvidenceForObservation(AtlasScanObservation observation) {
    final ScanCode? code = observation.code;
    if (!observation.hasUsableProductBarcode || code == null) {
      return const <String, AtlasTrackedBarcodeEvidence>{};
    }

    final String key = barcodeEvidenceKey(code);
    return <String, AtlasTrackedBarcodeEvidence>{
      key: AtlasTrackedBarcodeEvidence(key: key, latestCode: code, hitCount: 1),
    };
  }

  static AtlasNormalizedPoint? _smoothedCentroid({
    required AtlasNormalizedPoint? current,
    required AtlasNormalizedPoint? observation,
    required double alpha,
  }) {
    if (observation == null) return current;
    if (current == null) return observation;

    return AtlasNormalizedPoint(
      x: (current.x * (1.0 - alpha)) + (observation.x * alpha),
      y: (current.y * (1.0 - alpha)) + (observation.y * alpha),
    );
  }
}

import 'dart:math' as math;

import '../../scan/models/scan_code.dart';

class AtlasNormalizedPoint {
  AtlasNormalizedPoint({required this.x, required this.y}) {
    _validateCoordinate(x, 'x');
    _validateCoordinate(y, 'y');
  }

  final double x;
  final double y;

  double distanceTo(AtlasNormalizedPoint other) {
    final double dx = x - other.x;
    final double dy = y - other.y;
    return math.sqrt((dx * dx) + (dy * dy));
  }

  static void _validateCoordinate(double value, String fieldName) {
    if (!value.isFinite || value < 0.0 || value > 1.0) {
      throw ArgumentError.value(
        value,
        fieldName,
        '$fieldName must be between 0.0 and 1.0.',
      );
    }
  }

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is AtlasNormalizedPoint &&
          runtimeType == other.runtimeType &&
          x == other.x &&
          y == other.y;

  @override
  int get hashCode => Object.hash(x, y);
}

class AtlasNormalizedBounds {
  AtlasNormalizedBounds({
    required this.left,
    required this.top,
    required this.right,
    required this.bottom,
  }) {
    AtlasNormalizedPoint._validateCoordinate(left, 'left');
    AtlasNormalizedPoint._validateCoordinate(top, 'top');
    AtlasNormalizedPoint._validateCoordinate(right, 'right');
    AtlasNormalizedPoint._validateCoordinate(bottom, 'bottom');

    if (right < left) {
      throw ArgumentError.value(
        right,
        'right',
        'right cannot be less than left.',
      );
    }
    if (bottom < top) {
      throw ArgumentError.value(
        bottom,
        'bottom',
        'bottom cannot be less than top.',
      );
    }
  }

  final double left;
  final double top;
  final double right;
  final double bottom;

  double get width => right - left;
  double get height => bottom - top;

  AtlasNormalizedPoint get centroid =>
      AtlasNormalizedPoint(x: left + (width / 2), y: top + (height / 2));
}

class AtlasScanObservation {
  AtlasScanObservation({
    required this.observationId,
    required this.capturedAt,
    this.code,
    List<AtlasNormalizedPoint> cornerPoints = const <AtlasNormalizedPoint>[],
    this.confidence = 1.0,
  }) : cornerPoints = List<AtlasNormalizedPoint>.unmodifiable(cornerPoints) {
    if (observationId.trim().isEmpty) {
      throw ArgumentError.value(
        observationId,
        'observationId',
        'observationId cannot be empty.',
      );
    }

    if (!confidence.isFinite || confidence < 0.0 || confidence > 1.0) {
      throw ArgumentError.value(
        confidence,
        'confidence',
        'confidence must be between 0.0 and 1.0.',
      );
    }
  }

  final String observationId;
  final DateTime capturedAt;
  final ScanCode? code;
  final List<AtlasNormalizedPoint> cornerPoints;
  final double confidence;

  bool get hasGeometry => cornerPoints.isNotEmpty;
  bool get hasCode => code != null;

  bool get hasUsableProductBarcode {
    final ScanCode? observedCode = code;
    return observedCode != null &&
        observedCode.isValid &&
        observedCode.isProductBarcode &&
        observedCode.normalizedValue.trim().isNotEmpty;
  }

  AtlasNormalizedPoint? get centroid {
    if (cornerPoints.isEmpty) return null;

    double sumX = 0.0;
    double sumY = 0.0;

    for (final AtlasNormalizedPoint point in cornerPoints) {
      sumX += point.x;
      sumY += point.y;
    }

    return AtlasNormalizedPoint(
      x: sumX / cornerPoints.length,
      y: sumY / cornerPoints.length,
    );
  }

  AtlasNormalizedBounds? get normalizedBounds {
    if (cornerPoints.isEmpty) return null;

    double minX = cornerPoints.first.x;
    double maxX = cornerPoints.first.x;
    double minY = cornerPoints.first.y;
    double maxY = cornerPoints.first.y;

    for (final AtlasNormalizedPoint point in cornerPoints.skip(1)) {
      minX = math.min(minX, point.x);
      maxX = math.max(maxX, point.x);
      minY = math.min(minY, point.y);
      maxY = math.max(maxY, point.y);
    }

    return AtlasNormalizedBounds(
      left: minX,
      top: minY,
      right: maxX,
      bottom: maxY,
    );
  }
}

class AtlasEvidenceAccumulatorConfig {
  const AtlasEvidenceAccumulatorConfig({
    this.minBarcodeHits = 3,
    this.dominantBarcodeThreshold = 0.75,
  });

  final int minBarcodeHits;
  final double dominantBarcodeThreshold;
}

class AtlasAccumulatedEvidence {
  const AtlasAccumulatedEvidence({
    required this.trackId,
    required this.stableCode,
    required this.agreementConfidence,
    required this.totalBarcodeHits,
    required this.dominantBarcodeHits,
    required this.isReadyForResolution,
  });

  final String trackId;
  final ScanCode stableCode;
  final double agreementConfidence;
  final int totalBarcodeHits;
  final int dominantBarcodeHits;
  final bool isReadyForResolution;
}

class AtlasEvidenceAccumulator {
  AtlasEvidenceAccumulator({
    this.config = const AtlasEvidenceAccumulatorConfig(),
  }) {
    _validateConfig(config);
  }

  final AtlasEvidenceAccumulatorConfig config;

  AtlasAccumulatedEvidence? evaluateTrack(AtlasObjectTrack track) {
    if (track.isCooldown || track.barcodeEvidenceByKey.isEmpty) {
      return null;
    }

    final int totalBarcodeHits = track.usableBarcodeHitCount;
    if (totalBarcodeHits == 0) return null;

    AtlasTrackedBarcodeEvidence? dominantEvidence;

    for (final AtlasTrackedBarcodeEvidence evidence
        in track.barcodeEvidenceByKey.values) {
      final AtlasTrackedBarcodeEvidence? currentDominant = dominantEvidence;

      if (currentDominant == null ||
          evidence.hitCount > currentDominant.hitCount ||
          (evidence.hitCount == currentDominant.hitCount &&
              evidence.latestCode.capturedAt.isAfter(
                currentDominant.latestCode.capturedAt,
              ))) {
        dominantEvidence = evidence;
      }
    }

    if (dominantEvidence == null) return null;

    final double agreementConfidence =
        dominantEvidence.hitCount / totalBarcodeHits;

    final bool isReadyForResolution =
        track.isConfirmed &&
        totalBarcodeHits >= config.minBarcodeHits &&
        agreementConfidence >= config.dominantBarcodeThreshold;

    return AtlasAccumulatedEvidence(
      trackId: track.trackId,
      stableCode: dominantEvidence.latestCode,
      agreementConfidence: agreementConfidence,
      totalBarcodeHits: totalBarcodeHits,
      dominantBarcodeHits: dominantEvidence.hitCount,
      isReadyForResolution: isReadyForResolution,
    );
  }

  static void _validateConfig(AtlasEvidenceAccumulatorConfig config) {
    if (config.minBarcodeHits < 1) {
      throw ArgumentError.value(
        config.minBarcodeHits,
        'minBarcodeHits',
        'minBarcodeHits must be at least 1.',
      );
    }

    if (!config.dominantBarcodeThreshold.isFinite ||
        config.dominantBarcodeThreshold <= 0.0 ||
        config.dominantBarcodeThreshold > 1.0) {
      throw ArgumentError.value(
        config.dominantBarcodeThreshold,
        'dominantBarcodeThreshold',
        'dominantBarcodeThreshold must be greater than 0.0 and at most 1.0.',
      );
    }
  }
}

import 'dart:ui';

import 'package:mobile_scanner/mobile_scanner.dart';

import '../models/scan_code.dart';
import '../tracking/atlas_scan_observation.dart';

/// Adapts mobile_scanner capture data into Atlas acquisition observations.
///
/// This is intentionally the plugin boundary. The tracking domain remains
/// independent from Flutter camera/plugin geometry.
abstract final class AtlasMobileScannerObservationAdapter {
  static List<AtlasScanObservation> fromCapture(
    BarcodeCapture capture, {
    required DateTime capturedAt,
  }) {
    final List<AtlasScanObservation> observations = <AtlasScanObservation>[];

    final Size captureSize = capture.size;
    final bool hasUsableCaptureSize =
        captureSize.width.isFinite &&
        captureSize.height.isFinite &&
        captureSize.width > 0.0 &&
        captureSize.height > 0.0;

    int observationSequence = 0;

    for (final Barcode barcode in capture.barcodes) {
      final String rawValue = barcode.rawValue?.trim() ?? '';

      if (rawValue.isEmpty) {
        continue;
      }

      final ScanCodeFormat format = _mapBarcodeFormat(barcode.format);

      final ScanCode scanCode = ScanCode(
        rawValue: rawValue,
        normalizedValue: _normalizeCodeValue(rawValue),
        format: format,
        source: ScanCodeSource.camera,
        capturedAt: capturedAt,
        isValid: format != ScanCodeFormat.unknown,
        confidence: 1.0,
      );

      final List<AtlasNormalizedPoint> normalizedCorners = hasUsableCaptureSize
          ? _normalizeCorners(barcode.corners, captureSize)
          : const <AtlasNormalizedPoint>[];

      observations.add(
        AtlasScanObservation(
          observationId: _createObservationId(
            capturedAt: capturedAt,
            sequence: observationSequence,
            code: scanCode,
          ),
          capturedAt: capturedAt,
          code: scanCode,
          cornerPoints: normalizedCorners,
          confidence: scanCode.confidence,
        ),
      );

      observationSequence++;
    }

    return List<AtlasScanObservation>.unmodifiable(observations);
  }

  static List<AtlasNormalizedPoint> _normalizeCorners(
    List<Offset> corners,
    Size captureSize,
  ) {
    if (corners.isEmpty) {
      return const <AtlasNormalizedPoint>[];
    }

    final List<AtlasNormalizedPoint> normalized = <AtlasNormalizedPoint>[];

    for (final Offset corner in corners) {
      if (!corner.dx.isFinite || !corner.dy.isFinite) {
        continue;
      }

      normalized.add(
        AtlasNormalizedPoint(
          x: _normalizeCoordinate(corner.dx, captureSize.width),
          y: _normalizeCoordinate(corner.dy, captureSize.height),
        ),
      );
    }

    return List<AtlasNormalizedPoint>.unmodifiable(normalized);
  }

  static double _normalizeCoordinate(double value, double extent) {
    if (!value.isFinite || !extent.isFinite || extent <= 0.0) {
      return 0.0;
    }

    return (value / extent).clamp(0.0, 1.0).toDouble();
  }

  static ScanCodeFormat _mapBarcodeFormat(BarcodeFormat format) {
    return switch (format) {
      BarcodeFormat.upcA => ScanCodeFormat.upcA,
      BarcodeFormat.upcE => ScanCodeFormat.upcE,
      BarcodeFormat.ean8 => ScanCodeFormat.ean8,
      BarcodeFormat.ean13 => ScanCodeFormat.ean13,
      BarcodeFormat.qrCode => ScanCodeFormat.qrCode,
      _ => ScanCodeFormat.unknown,
    };
  }

  static String _normalizeCodeValue(String rawValue) {
    return rawValue.trim();
  }

  static String _createObservationId({
    required DateTime capturedAt,
    required int sequence,
    required ScanCode code,
  }) {
    final int codeHash = code.normalizedValue.hashCode.abs();

    return 'obs_'
        '${capturedAt.microsecondsSinceEpoch}_'
        '${sequence}_'
        '$codeHash';
  }
}

AtlasScanObservation _observation({
  required String id,
  required DateTime capturedAt,
  required String barcode,
  double centerX = 0.50,
  double centerY = 0.50,
}) {
  return AtlasScanObservation(
    observationId: id,
    capturedAt: capturedAt,
    code: ScanCode(
      rawValue: barcode,
      normalizedValue: barcode,
      format: ScanCodeFormat.upcA,
      source: ScanCodeSource.camera,
      capturedAt: capturedAt,
      isValid: true,
      confidence: 1.0,
    ),
    cornerPoints: _boxAround(centerX: centerX, centerY: centerY),
  );
}

AtlasScanObservation _observationWithoutBarcode({
  required String id,
  required DateTime capturedAt,
}) {
  return AtlasScanObservation(
    observationId: id,
    capturedAt: capturedAt,
    cornerPoints: _boxAround(centerX: 0.50, centerY: 0.50),
  );
}

List<AtlasNormalizedPoint> _boxAround({
  required double centerX,
  required double centerY,
}) {
  const double halfWidth = 0.02;
  const double halfHeight = 0.03;

  return <AtlasNormalizedPoint>[
    AtlasNormalizedPoint(x: centerX - halfWidth, y: centerY - halfHeight),
    AtlasNormalizedPoint(x: centerX + halfWidth, y: centerY - halfHeight),
    AtlasNormalizedPoint(x: centerX + halfWidth, y: centerY + halfHeight),
    AtlasNormalizedPoint(x: centerX - halfWidth, y: centerY + halfHeight),
  ];
}
