/// Immutable projected state derived from scanner intelligence contributions.
///
/// Scan engines contribute evidence. The projector converts that evidence into
/// the canonical state displayed by Atlas.
class ScanSessionProjection {
  const ScanSessionProjection({
    required this.status,
    required this.resolutionStatus,
    required this.atlasProductId,
    required this.resolvedProduct,
    required this.confidence,
  });

  final ScanSessionStatus status;

  final IdentityResolutionStatus resolutionStatus;

  final AtlasProductId? atlasProductId;

  final ResolvedProductSnapshot? resolvedProduct;

  final double confidence;
}

/// Converts immutable intelligence contributions into trusted ScanSession
/// state.
///
/// Engines should produce contributions rather than independently deciding
/// what the scanner UI displays.
abstract final class ScanSessionProjector {
  static ScanSessionProjection project({
    required ContributionCollection contributions,
    required ScanSessionStatus currentStatus,
    required IdentityResolutionStatus currentResolutionStatus,
    required AtlasProductId? currentAtlasProductId,
    required ResolvedProductSnapshot? currentResolvedProduct,
    required double currentConfidence,
    required bool isOffline,
  }) {
    final ProductResolutionContribution? resolutionContribution =
    contributions.latestOfType<ProductResolutionContribution>();

    if (resolutionContribution == null) {
      return ScanSessionProjection(
        status: currentStatus,
        resolutionStatus: currentResolutionStatus,
        atlasProductId: currentAtlasProductId,
        resolvedProduct: currentResolvedProduct,
        confidence: _normalizeConfidence(
          currentConfidence,
        ),
      );
    }

    final resolution = resolutionContribution.resolution;
    final double confidence = _normalizeConfidence(
      resolution.matchConfidence,
    );

    if (resolution.requiresReview) {
      return ScanSessionProjection(
        status: ScanSessionStatus.awaitingConfirmation,
        resolutionStatus:
        IdentityResolutionStatus.ambiguous,
        atlasProductId: resolution.productId,
        resolvedProduct: resolution.resolvedProduct,
        confidence: confidence,
      );
    }

    return ScanSessionProjection(
      status: ScanSessionStatus.completed,
      resolutionStatus: isOffline
          ? IdentityResolutionStatus.resolvedLocal
          : IdentityResolutionStatus.resolvedRemote,
      atlasProductId: resolution.productId,
      resolvedProduct: resolution.resolvedProduct,
      confidence: confidence,
    );
  }

  static double _normalizeConfidence(double value) {
    if (!value.isFinite) {
      return 0.0;
    }

    return value.clamp(0.0, 1.0).toDouble();
  }
}

class AtlasVisualInspectionService {
  const AtlasVisualInspectionService({
    this.authenticityEngine = const AtlasAuthenticityEngine(),
    this.highSignalPipeline = const AtlasHighSignalInspectionPipeline(),
  });

  final AtlasAuthenticityEngine authenticityEngine;
  final AtlasHighSignalInspectionPipeline highSignalPipeline;

  Future<ScanSession> inspect({
    required ScanSession session,
    required AtlasVisualInspection inspection,
  }) async {
    final AtlasVisualInspection enrichedInspection = highSignalPipeline.enrich(
      sessionId: session.sessionId,
      inspection: inspection,
      expectedResolvedVariant: session.resolvedProduct?.variantName,
    );

    var updated = session;

    for (final image in enrichedInspection.images) {
      updated = updated.addImage(image);
    }

    updated = updated.addContribution(
      VisualInspectionContribution(
        contributionId: 'visual-inspection-${session.sessionId}',
        sourceEngineId: 'atlas.visual-inspection',
        createdAt: DateTime.now().toUtc(),
        inspection: enrichedInspection,
        confidence: enrichedInspection.resolvedEvidenceConfidence,
        attributes: <String, Object?>{
          'inspectionAreas': enrichedInspection.inspectedAreas
              .map((area) => area.name)
              .toList(growable: false),
          'structuredObservationCount': enrichedInspection.observations.length,
          'hasReferenceCriteria': enrichedInspection.referenceCriteria != null,
          'highSignalEnginesApplied': true,
        },
      ),
      conflictPolicy: ContributionConflictPolicy.replaceExisting,
    );

    final result = await authenticityEngine.process(
      updated,
      EngineContext(
        requestId: session.sessionId,
        createdAt: DateTime.now().toUtc(),
        deviceType: EngineDeviceType.mobile,
      ),
    );

    return result.session;
  }
}

final class AtlasAuthenticityEngine extends AbstractScanEngine {
  const AtlasAuthenticityEngine();

  static const id = 'atlas.scan.authenticity';
  static const capabilityId = 'atlas.capability.authenticity';

  @override
  String get engineId => id;

  @override
  String get displayName => 'Authenticity Intelligence';

  @override
  Future<EngineResult> execute(
    ScanSession session,
    EngineContext context,
  ) async {
    final visual = session
        .latestContributionOfType<VisualInspectionContribution>();
    if (visual == null) {
      return EngineResult.skipped(
        session: session,
        reason: 'No visual inspection evidence is available.',
      );
    }

    final assessment = AtlasAuthenticityRules.evaluate(
      session: session,
      inspection: visual.inspection,
    );

    final contribution = AuthenticityContribution(
      contributionId: EngineSupport.buildContributionId(
        namespace: 'atlas.authenticity',
        segments: <Object?>[session.sessionId],
      ),
      sourceEngineId: engineId,
      createdAt: DateTime.now().toUtc(),
      assessment: assessment,
      confidence: assessment.evidenceConfidence,
      attributes: <String, Object?>{
        'status': assessment.status.name,
        'concernLevel': assessment.concernLevel.name,
        'evidenceConfidence': assessment.evidenceConfidence,
        'referenceCoverage': assessment.referenceCoverage,
        'inspectionCoverage': assessment.inspectionCoverage,
        'concernScore': assessment.aggregatedConcernScore,
      },
    );

    final updated = session.addContribution(
      contribution,
      conflictPolicy: ContributionConflictPolicy.replaceExisting,
    );

    return EngineResult.success(session: updated);
  }
}

final class AtlasStickerVariantInspectionEngine {
  const AtlasStickerVariantInspectionEngine();

  static const String id = 'atlas.scan.authenticity.sticker-variant';

  List<AtlasVisualEvidenceObservation> inspect({
    required String sessionId,
    required AtlasVisualInspection inspection,
    String? expectedResolvedVariant,
    required DateTime observedAt,
  }) {
    final String stickerText = inspection.stickerText.trim();
    if (stickerText.isEmpty) {
      return const <AtlasVisualEvidenceObservation>[];
    }

    final analysis = AtlasIdentityAnalysisEngine.create().analyze(stickerText);
    final Set<String> detected = <String>{
      ...analysis.stickerResult.stickers.map((value) => value.displayName),
      ...analysis.variantResult.variants.map((value) => value.displayName),
    }..removeWhere((String value) => value.trim().isEmpty);

    final AtlasInspectionReferenceCriteria? reference =
        inspection.referenceCriteria;
    final bool hasReferenceVariants =
        reference != null && reference.legitimateVariantTypes.isNotEmpty;
    final bool referenceMismatch =
        hasReferenceVariants &&
        detected.isNotEmpty &&
        !_hasAnySemanticMatch(detected, reference.legitimateVariantTypes);
    final bool criticalReferenceMismatch =
        referenceMismatch && _hasHighConfidenceVettedReference(reference);
    final bool visualMismatch =
        inspection.stickerPrintClear == false ||
        inspection.stickerCutEven == false;

    final String expected = expectedResolvedVariant?.trim() ?? '';
    final bool resolvedVariantMismatch =
        expected.isNotEmpty &&
        detected.isNotEmpty &&
        !_hasAnySemanticMatch(detected, <String>[expected]);

    return <AtlasVisualEvidenceObservation>[
      AtlasVisualEvidenceObservation(
        observationId: 'scanner5c-$sessionId-sticker-variant',
        sourceEngineId: id,
        area: AtlasInspectionArea.windowSticker,
        category: AtlasEvidenceCategory.variantSticker,
        label: detected.isEmpty
            ? 'Sticker text recorded'
            : 'Sticker / variant classification',
        observedValue: stickerText,
        confidence: inspection.resolvedEvidenceConfidence,
        scanImageId: _imageIdForArea(
          inspection,
          AtlasInspectionArea.windowSticker,
        ),
        attributes: <String, Object?>{
          'classified': detected.isNotEmpty,
          'detectedLabels': detected.toList(growable: false),
          if (expected.isNotEmpty) 'expectedResolvedVariant': expected,
          if (resolvedVariantMismatch) 'resolvedVariantMismatch': true,
          if (inspection.stickerPrintClear != null)
            'stickerPrintClear': inspection.stickerPrintClear,
          if (inspection.stickerCutEven != null)
            'stickerCutEven': inspection.stickerCutEven,
          if (visualMismatch) 'visualMismatch': true,
          if (hasReferenceVariants)
            'referenceVariantFamilies': reference.legitimateVariantTypes,
          if (referenceMismatch) 'referenceMismatch': true,
          if (criticalReferenceMismatch) 'criticalReferenceMismatch': true,
        },
        observedAt: observedAt,
      ),
    ];
  }

  static bool _hasAnySemanticMatch(
    Iterable<String> observed,
    Iterable<String> expected,
  ) {
    for (final String left in observed) {
      final String normalizedLeft = _normalize(left);
      if (normalizedLeft.isEmpty) continue;
      for (final String right in expected) {
        final String normalizedRight = _normalize(right);
        if (normalizedRight.isEmpty) continue;
        if (normalizedLeft == normalizedRight ||
            normalizedLeft.contains(normalizedRight) ||
            normalizedRight.contains(normalizedLeft)) {
          return true;
        }
      }
    }
    return false;
  }

  static bool _hasHighConfidenceVettedReference(
    AtlasInspectionReferenceCriteria reference,
  ) {
    if (reference.coverageLevel != AtlasReferenceCoverageLevel.comprehensive ||
        reference.completenessScore < 0.85) {
      return false;
    }
    return reference.provenance.any(
          (AtlasReferenceProvenance source) => source.specialistVetted,
        ) ||
        reference.knownRuns.any(
          (AtlasProductionRunReference run) => run.provenance.specialistVetted,
        );
  }

  static String _normalize(String value) =>
      value.trim().toLowerCase().replaceAll(RegExp(r'[^a-z0-9]'), '');

  static String? _imageIdForArea(
    AtlasVisualInspection inspection,
    AtlasInspectionArea area,
  ) {
    for (final image in inspection.images) {
      final bool matches = switch (area) {
        AtlasInspectionArea.windowSticker => image.type.name == 'sticker',
        _ => false,
      };
      if (matches) return image.imageId;
    }
    return null;
  }
}

      completedAt: completedAt,
    );
  }

  Future<void> _inspectItem({
    required AtlasScanLedgerEntry entry,
    required AtlasContinuousScanResult result,
  }) async {
    final session = await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (_) => AtlasVisualInspectionScreen(session: result.session),
      ),
    );
    if (!mounted || session == null) return;
    widget.coordinator.applyInspection(
      trackId: entry.trackId,
      session: session,
    );
    setState(() {});
  }

  Future<bool> _runSaveForTrack({
    required AtlasScanLedgerEntry entry,
    required Widget Function(BuildContext context) screenBuilder,
    required String successLabel,
  }) async {
    if (_savingTrackIds.contains(entry.trackId)) {
      return false;
    }

    setState(() {
      _savingTrackIds.add(entry.trackId);
    });

    try {
      final bool? added = await Navigator.push<bool>(
        context,
        MaterialPageRoute<bool>(builder: screenBuilder),
      );

      if (!mounted || added != true) {
        return false;
      }

      final bool committed = widget.coordinator.markCommitted(
        trackId: entry.trackId,
      );

      if (!committed) {
        return false;
      }

      setState(() {});

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(
            content: Text(
              '$successLabel committed to ${widget.collectionName}.',
            ),
          ),
        );

      return true;
    } finally {
      if (mounted) {
        setState(() {
          _savingTrackIds.remove(entry.trackId);
        });
      } else {
        _savingTrackIds.remove(entry.trackId);
      }
    }
  }

  Future<void> _addCatalogItem({
    required AtlasScanLedgerEntry entry,
    required AtlasCatalogProduct product,
  }) async {
    await _runSaveForTrack(
      entry: entry,
      successLabel: product.displayName,
      screenBuilder: (_) {
        final inspection = widget.coordinator
            .resultForTrack(entry.trackId)
            ?.session
            .visualInspectionContribution
            ?.inspection;
        return AddPopScreen(
          atlasProduct: product,
          collectionName: widget.collectionName,
          visualInspection: inspection,
        );
      },
    );
  }

          collectionName: widget.collectionName,
          visualInspection: inspection,
        );
      },
    );
  }

  Future<void> _linkExistingVaultItem({
    required AtlasScanLedgerEntry entry,
    required List<Pop> collectionPops,
  }) async {
    if (_savingTrackIds.contains(entry.trackId)) {
      return;
    }

    final Pop? selected = await showModalBottomSheet<Pop>(
      context: context,
      isScrollControlled: true,
      builder: (_) {
        return _VaultItemPickerSheet(
          collectionName: widget.collectionName,
          collectionPops: collectionPops,
          scannedBarcode: entry.stableCode.normalizedValue,
        );
      },
    );

    if (!mounted || selected == null) {
      return;
    }

    final String scannedBarcode = entry.stableCode.normalizedValue.trim();
    final String existingBarcode = selected.sourceUpc.trim();

    if (existingBarcode.isNotEmpty && existingBarcode != scannedBarcode) {
      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(
            content: Text(
              '${selected.name} already has barcode $existingBarcode. Atlas did not overwrite it.',
            ),
          ),
        );
      return;
    }

    setState(() {
      _savingTrackIds.add(entry.trackId);
    });

    try {
      final AppState appState = context.read<AppState>();
      final Pop linkedPop = existingBarcode == scannedBarcode
          ? selected
          : selected.copyWith(sourceUpc: scannedBarcode);

      final Pop? persisted = await appState.updatePop(
        linkedPop,
        collectionName: widget.collectionName,
        existingRecordKey: selected.collectionRecordKey,
      );

      if (!mounted) {
        return;
      }

      if (persisted == null) {
        ScaffoldMessenger.of(context)
          ..hideCurrentSnackBar()
          ..showSnackBar(
            const SnackBar(
              content: Text(
                'Atlas could not safely link that Vault item. Nothing was changed.',
              ),
            ),
          );
        return;
      }

      final bool committed = widget.coordinator.markCommitted(
        trackId: entry.trackId,
      );

      if (!committed) {
        return;
      }

      setState(() {});

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(
            content: Text(
              'Linked barcode $scannedBarcode to ${persisted.name} and committed this haul item.',
            ),
          ),
        );
    } finally {
      if (mounted) {
        setState(() {
          _savingTrackIds.remove(entry.trackId);
        });
      } else {
        _savingTrackIds.remove(entry.trackId);
      }
    }

        .where((Pop pop) {
          if (query.isEmpty) {
            return true;

            pop.name,
            pop.number,
            pop.series,
            pop.variant,
            pop.retailExclusive,
            pop.eventSticker,
            pop.limitedEdition,
            pop.signature,
            pop.sourceUpc,

            (String value) => value.toLowerCase().contains(query),

  Future<void> _captureStep(
    AtlasGuidedInspectionStep step, {
    TextEditingController? textTarget,
  }) async {
    final source = await showModalBottomSheet<AtlasInspectionImageSource>(
      context: context,
      builder: (context) => SafeArea(
        child: Wrap(
          children: [
            ListTile(
              leading: const Icon(Icons.photo_camera_outlined),
              title: const Text('Camera'),

            ),
          ],
        ),
      ),
    );

    if (source == null) return;

    setState(() => _busy = true);
    try {
      final ScanImage? image = await _capture.capture(
        source: source,
        type: step.imageType,
      );
      if (image == null) return;

      String recognizedText = '';
      if (step.supportsTextRecognition &&
          textTarget != null &&
          image.hasLocalPath) {
        recognizedText = await _recognizer.recognize(image.localPath!);

      setState(() {
        _images.removeWhere((ScanImage item) => item.type == step.imageType);
        _images.add(image);
        if (recognizedText.isNotEmpty && textTarget != null) {
          textTarget.text = recognizedText;
        }
      });
    } finally {
      if (mounted) setState(() => _busy = false);
    }
