  ScanSession addDetectedCode(ScanCode code, {bool makeBestCandidate = false}) {
    final bool alreadyDetected = detectedCodes.any((ScanCode existing) {
      return existing.format == code.format &&
          existing.normalizedValue == code.normalizedValue;
    });

    final List<ScanCode> updatedCodes = alreadyDetected
        ? detectedCodes
        : <ScanCode>[...detectedCodes, code];

    return copyWith(
      detectedCodes: updatedCodes,
      bestCandidate: makeBestCandidate ? code : bestCandidate,
    );
  }

    ContributionConflictPolicy conflictPolicy =
        ContributionConflictPolicy.reject,
  }) {
    final ContributionCollection updatedContributions = contributions.add(

      conflictPolicy: conflictPolicy,
    );

    return _withProjectedContributions(updatedContributions);
  }

    Iterable<IntelligenceContribution> values, {
    ContributionConflictPolicy conflictPolicy =
        ContributionConflictPolicy.reject,
  }) {
    final ContributionCollection updatedContributions = contributions.addAll(
      values,
      conflictPolicy: conflictPolicy,
    );

    return _withProjectedContributions(updatedContributions);
  }

    ContributionCollection updatedContributions,
  ) {
    final ScanSessionProjection projection = ScanSessionProjector.project(
      contributions: updatedContributions,
      currentStatus: status,
      currentResolutionStatus: resolutionStatus,
      currentAtlasProductId: atlasProductId,
      currentResolvedProduct: resolvedProduct,
      currentConfidence: confidence,
      isOffline: isOffline,
    );

    return copyWith(
      contributions: updatedContributions,
      status: projection.status,
      resolutionStatus: projection.resolutionStatus,
      atlasProductId: projection.atlasProductId,
      resolvedProduct: projection.resolvedProduct,
      confidence: projection.confidence,
    );
  }

  ScanSession removeContributionById(String contributionId) {
    final ContributionCollection updated = contributions.removeById(
      contributionId,
    );

    if (identical(updated, contributions)) {
      return this;
    }

    return copyWith(contributions: updated);
  }

  ScanSession removeContributionsFromEngine(String engineId) {
    final ContributionCollection updated = contributions.removeFromEngine(
      engineId,
    );

    if (identical(updated, contributions)) {
      return this;
    }

    return copyWith(contributions: updated);
  }

  ScanSession addImage(ScanImage image) {
    final List<ScanImage> updatedImages =
        images
            .where((ScanImage existing) => existing.imageId != image.imageId)
            .toList()
          ..add(image);

    return copyWith(images: updatedImages);
  }

  ScanSession addNote(String note) {
    final String normalized = note.trim();

    if (normalized.isEmpty || notes.contains(normalized)) {
      return this;
    }

    return copyWith(notes: <String>[...notes, normalized]);
  }

  ScanSession addWarning(String warning) {
    final String normalized = warning.trim();

    if (normalized.isEmpty || warnings.contains(normalized)) {
      return this;
    }

    return copyWith(warnings: <String>[...warnings, normalized]);
  }

  @override
  String toString() {
    return 'ScanSession('
        'sessionId: $sessionId, '
        'status: ${status.name}, '
        'intent: ${intent.name}, '
        'codes: ${detectedCodes.length}, '
        'images: ${images.length}, '
        'resolutionStatus: ${resolutionStatus.name}, '
        'atlasProductId: $atlasProductId, '
        'confidence: $confidence, '
        'isOffline: $isOffline, '
        'requiresSync: $requiresSync'
        ')';
  }

import 'contribution_category.dart';

// Shared contract for immutable intelligence produced by an Atlas engine.
//
// Concrete contribution models own their domain-specific fields while this
// contract provides the common information needed for discovery, diagnostics,
// conflict handling, and downstream intelligence processing.
abstract interface class IntelligenceContribution {
  String get contributionId;

  ContributionCategory get category;

  String get sourceEngineId;

  DateTime get createdAt;

  double? get confidence;

  Map<String, Object?> get attributes;
}

// Shared validation helpers for concrete contribution implementations.
abstract final class IntelligenceContributionValidation {
  static String requireIdentifier(
      String value, {
        required String fieldName,
      }) {
    final String normalized = value.trim();

    if (normalized.isEmpty) {
      throw ArgumentError.value(
        value,
        fieldName,
        '$fieldName cannot be empty.',
      );
    }

    return normalized;
  }

  static double? normalizeConfidence(
      double? value,
      ) {
    if (value == null) {
      return null;
    }

    if (value.isNaN ||
        value.isInfinite ||
        value < 0 ||
        value > 1) {
      throw ArgumentError.value(
        value,
        'confidence',
        'Confidence must be between 0 and 1.',
      );
    }

    return value;
  }

  static Map<String, Object?> immutableAttributes(
      Map<String, Object?> values,
      ) {
    return Map<String, Object?>.unmodifiable(
      values,
    );
  }
}

import 'contribution_category.dart';
import 'intelligence_contribution.dart';

// Reusable immutable base for intelligence produced by Atlas engines.
//
// Concrete contribution classes inherit the common identity, source,
// timestamp, confidence, and attribute behavior while defining only their
// domain-specific fields and category.
abstract base class AbstractIntelligenceContribution
    implements IntelligenceContribution {
  AbstractIntelligenceContribution({
    required String contributionId,
    required String sourceEngineId,
    required this.createdAt,
    double? confidence,
    Map<String, Object?> attributes =
    const <String, Object?>{},
  })  : contributionId =
  IntelligenceContributionValidation.requireIdentifier(
    contributionId,
    fieldName: 'contributionId',
  ),
        sourceEngineId =
        IntelligenceContributionValidation.requireIdentifier(
          sourceEngineId,
          fieldName: 'sourceEngineId',
        ),
        confidence =
        IntelligenceContributionValidation.normalizeConfidence(
          confidence,
        ),
        attributes =
        IntelligenceContributionValidation.immutableAttributes(
          attributes,
        );

  @override
  final String contributionId;

  @override
  final String sourceEngineId;

  @override
  final DateTime createdAt;

  @override
  final double? confidence;

  @override
  final Map<String, Object?> attributes;

  @override
  ContributionCategory get category;

  bool get hasConfidence {
    return confidence != null;
  }

  bool get hasAttributes {
    return attributes.isNotEmpty;
  }

  bool get hasNoAttributes {
    return attributes.isEmpty;
  }

  bool wasProducedBy(
      String engineId,
      ) {
    final String normalized = engineId.trim();

    return normalized.isNotEmpty &&
        sourceEngineId == normalized;
  }

  T? attribute<T>(
      String key,
      ) {
    final Object? value = attributes[key];

    if (value is T) {
      return value;
    }

    return null;
  }

  bool hasAttribute(
      String key,
      ) {
    final String normalized = key.trim();

    return normalized.isNotEmpty &&
        attributes.containsKey(normalized);
  }

  @override
  String toString() {
    return '$runtimeType('
        'contributionId: $contributionId, '
        'category: ${category.name}, '
        'sourceEngineId: $sourceEngineId, '
        'confidence: $confidence'
        ')';
  }
}

import 'abstract_intelligence_contribution.dart';
import 'contribution_category.dart';

// Base class for recognition-based intelligence contributions.
//
// Recognition contributions represent evidence extracted from a collector
// item before identity or product resolution occurs. Examples include
// barcode recognition, OCR, vision analysis, logo detection, and sticker
// detection.
abstract base class AbstractRecognitionContribution
    extends AbstractIntelligenceContribution {
  AbstractRecognitionContribution({
    required super.contributionId,
    required super.sourceEngineId,
    required super.createdAt,
    super.confidence,
    super.attributes,
    this.passedValidation = true,
    String? validationMessage,
  }) : validationMessage =
  _normalizeValidationMessage(validationMessage);

  @override
  ContributionCategory get category {
    return ContributionCategory.evidence;
  }

  final bool passedValidation;

  final String? validationMessage;

  bool get failedValidation {
    return !passedValidation;
  }

  bool get hasValidationMessage {
    return validationMessage != null;
  }

  bool get isTrusted {
    final double? value = confidence;

    return passedValidation &&
        value != null &&
        value >= 0.75;
  }

  bool get isLowConfidence {
    final double? value = confidence;

    return value != null && value < 0.50;
  }

  bool get hasUnknownConfidence {
    return confidence == null;
  }

  bool get shouldReview {
    return failedValidation ||
        hasUnknownConfidence ||
        isLowConfidence;
  }

  static String? _normalizeValidationMessage(
      String? value,
      ) {
    final String? normalized = value?.trim();

    if (normalized == null || normalized.isEmpty) {
      return null;
    }

    return normalized;
  }

  @override
  String toString() {
    return '$runtimeType('
        'confidence: $confidence, '
        'passedValidation: $passedValidation'
        ')';
  }
}

import 'contribution_category.dart';
import 'contribution_conflict_policy.dart';

// Immutable collection of intelligence contributed during an Atlas scan.
//
// Contributions can be discovered by ID, concrete type, category, or source
// engine without requiring consumers to understand ScanSession internals.
class ContributionCollection {
  factory ContributionCollection({
    Iterable<IntelligenceContribution> contributions =
    const <IntelligenceContribution>[],
  }) {
    final Map<String, IntelligenceContribution> byId =
    <String, IntelligenceContribution>{};

    for (final IntelligenceContribution contribution
    in contributions) {
      final String contributionId =
      _normalizedContributionId(contribution);

      if (byId.containsKey(contributionId)) {
        throw StateError(
          'Duplicate contribution ID: $contributionId',
        );
      }

      byId[contributionId] = contribution;
    }

    return ContributionCollection._(
      Map<String, IntelligenceContribution>.unmodifiable(
        byId,
      ),
    );
  }

  const ContributionCollection._(this._byId);

  static final ContributionCollection empty =
  ContributionCollection();

  final Map<String, IntelligenceContribution> _byId;

  int get length {
    return _byId.length;
  }

  bool get isEmpty {
    return _byId.isEmpty;
  }

  bool get isNotEmpty {
    return _byId.isNotEmpty;
  }

  List<IntelligenceContribution> get values {
    return List<IntelligenceContribution>.unmodifiable(
      _byId.values,
    );
  }

  bool containsId(
      String contributionId,
      ) {
    return _byId.containsKey(
      contributionId.trim(),
    );
  }

  IntelligenceContribution? contributionForId(
      String contributionId,
      ) {
    return _byId[contributionId.trim()];
  }

  T? contributionForIdAs<
  T extends IntelligenceContribution>(
      String contributionId,
      ) {
    final IntelligenceContribution? contribution =
    contributionForId(contributionId);

    if (contribution is T) {
      return contribution;
    }

    return null;
  }

  List<T> ofType<T extends IntelligenceContribution>() {
    return List<T>.unmodifiable(
      _byId.values.whereType<T>(),
    );
  }

  T? latestOfType<
  T extends IntelligenceContribution>() {
    T? latest;

    for (final T contribution
    in _byId.values.whereType<T>()) {
      if (latest == null ||
          contribution.createdAt.isAfter(
            latest.createdAt,
          )) {
        latest = contribution;
      }
    }

    return latest;
  }

  List<IntelligenceContribution> forCategory(
      ContributionCategory category,
      ) {
    return List<IntelligenceContribution>.unmodifiable(
      _byId.values.where(
            (IntelligenceContribution contribution) {
          return contribution.category == category;
        },
      ),
    );
  }

  List<IntelligenceContribution> fromEngine(
      String engineId,
      ) {
    final String normalizedEngineId =
    engineId.trim();

    if (normalizedEngineId.isEmpty) {
      return const <IntelligenceContribution>[];
    }

    return List<IntelligenceContribution>.unmodifiable(
      _byId.values.where(
            (IntelligenceContribution contribution) {
          return contribution.sourceEngineId ==
              normalizedEngineId;
        },
      ),
    );
  }

  List<IntelligenceContribution>
  withMinimumConfidence(
      double minimumConfidence,
      ) {
    final double normalizedMinimum =
    IntelligenceContributionValidation
        .normalizeConfidence(
      minimumConfidence,
    )!;

    return List<IntelligenceContribution>.unmodifiable(
      _byId.values.where(
            (IntelligenceContribution contribution) {
          final double? confidence =
              contribution.confidence;

          return confidence != null &&
              confidence >= normalizedMinimum;
        },
      ),
    );
  }

  ContributionCollection add(
      IntelligenceContribution contribution, {
        ContributionConflictPolicy conflictPolicy =
            ContributionConflictPolicy.reject,
      }) {
    final String contributionId =
    _normalizedContributionId(contribution);

    final bool alreadyExists =
    _byId.containsKey(contributionId);

    if (alreadyExists) {
      switch (conflictPolicy) {
        case ContributionConflictPolicy.reject:
          throw StateError(
            'Contribution "$contributionId" '
                'already exists.',
          );

        case ContributionConflictPolicy.keepExisting:
          return this;

        case ContributionConflictPolicy.replaceExisting:
          break;
      }
    }

    return ContributionCollection(
      contributions:
      <IntelligenceContribution>[
        ..._byId.values.where(
              (IntelligenceContribution existing) {
            return existing.contributionId.trim() !=
                contributionId;
          },
        ),
        contribution,
      ],
    );
  }

  ContributionCollection addAll(
      Iterable<IntelligenceContribution> contributions, {
        ContributionConflictPolicy conflictPolicy =
            ContributionConflictPolicy.reject,
      }) {
    ContributionCollection current = this;

    for (final IntelligenceContribution contribution
    in contributions) {
      current = current.add(
        contribution,
        conflictPolicy: conflictPolicy,
      );
    }

    return current;
  }

  ContributionCollection removeById(
      String contributionId,
      ) {
    final String normalized =
    contributionId.trim();

    if (normalized.isEmpty ||
        !_byId.containsKey(normalized)) {
      return this;
    }

    return ContributionCollection(
      contributions: _byId.values.where(
            (IntelligenceContribution contribution) {
          return contribution.contributionId.trim() !=
              normalized;
        },
      ),
    );
  }

  ContributionCollection removeFromEngine(
      String engineId,
      ) {
    final String normalizedEngineId =
    engineId.trim();

    if (normalizedEngineId.isEmpty) {
      return this;
    }

    final List<IntelligenceContribution> remaining =
    _byId.values.where(
          (IntelligenceContribution contribution) {
        return contribution.sourceEngineId !=
            normalizedEngineId;
      },
    ).toList(growable: false);

    if (remaining.length == length) {
      return this;
    }

    return ContributionCollection(
      contributions: remaining,
    );
  }

  ContributionCollection removeCategory(
      ContributionCategory category,
      ) {
    final List<IntelligenceContribution> remaining =
    _byId.values.where(
          (IntelligenceContribution contribution) {
        return contribution.category != category;
      },
    ).toList(growable: false);

    if (remaining.length == length) {
      return this;
    }

    return ContributionCollection(
      contributions: remaining,
    );
  }

  static String _normalizedContributionId(
      IntelligenceContribution contribution,
      ) {
    return IntelligenceContributionValidation
        .requireIdentifier(
      contribution.contributionId,
      fieldName: 'contributionId',
    );
  }

  @override
  String toString() {
    return 'ContributionCollection('
        'length: $length'
        ')';
  }
}

/// Immutable projected state derived from scanner intelligence contributions.
///
/// Scan engines contribute evidence. The projector converts that evidence into
/// the canonical state displayed by Atlas.
class ScanSessionProjection {
  const ScanSessionProjection({
    required this.status,
    required this.resolutionStatus,
    required this.atlasProductId,
    required this.resolvedProduct,
    required this.confidence,
  });

  final ScanSessionStatus status;

  final IdentityResolutionStatus resolutionStatus;

  final AtlasProductId? atlasProductId;

  final ResolvedProductSnapshot? resolvedProduct;

  final double confidence;
}

/// Converts immutable intelligence contributions into trusted ScanSession
/// state.
///
/// Engines should produce contributions rather than independently deciding
/// what the scanner UI displays.
abstract final class ScanSessionProjector {
  static ScanSessionProjection project({
    required ContributionCollection contributions,
    required ScanSessionStatus currentStatus,
    required IdentityResolutionStatus currentResolutionStatus,
    required AtlasProductId? currentAtlasProductId,
    required ResolvedProductSnapshot? currentResolvedProduct,
    required double currentConfidence,
    required bool isOffline,
  }) {
    final ProductResolutionContribution? resolutionContribution =
    contributions.latestOfType<ProductResolutionContribution>();

    if (resolutionContribution == null) {
      return ScanSessionProjection(
        status: currentStatus,
        resolutionStatus: currentResolutionStatus,
        atlasProductId: currentAtlasProductId,
        resolvedProduct: currentResolvedProduct,
        confidence: _normalizeConfidence(
          currentConfidence,
        ),
      );
    }

    final resolution = resolutionContribution.resolution;
    final double confidence = _normalizeConfidence(
      resolution.matchConfidence,
    );

    if (resolution.requiresReview) {
      return ScanSessionProjection(
        status: ScanSessionStatus.awaitingConfirmation,
        resolutionStatus:
        IdentityResolutionStatus.ambiguous,
        atlasProductId: resolution.productId,
        resolvedProduct: resolution.resolvedProduct,
        confidence: confidence,
      );
    }

    return ScanSessionProjection(
      status: ScanSessionStatus.completed,
      resolutionStatus: isOffline
          ? IdentityResolutionStatus.resolvedLocal
          : IdentityResolutionStatus.resolvedRemote,
      atlasProductId: resolution.productId,
      resolvedProduct: resolution.resolvedProduct,
      confidence: confidence,
    );
  }

  static double _normalizeConfidence(double value) {
    if (!value.isFinite) {
      return 0.0;
    }

    return value.clamp(0.0, 1.0).toDouble();
  }
}

// Canonical identifier for a product known by Atlas.
//
// Example:
//
// ATLAS-FNK-000001
//
// This ID is owned by Atlas and remains stable regardless of changes to
// external product databases.
final class AtlasProductId {
  AtlasProductId({
    required String categoryCode,
    required int numericId,
  })  : categoryCode = _normalizeCategoryCode(categoryCode),
        numericId = _validateNumericId(numericId);

  // Convenience constructor for Funko products.
  AtlasProductId.funko({
    required int numericId,
  }) : this(
    categoryCode: 'FNK',
    numericId: numericId,
  );

  // Future convenience constructors.
  AtlasProductId.pokemon({
    required int numericId,
  }) : this(
    categoryCode: 'PKM',
    numericId: numericId,
  );

  AtlasProductId.hotWheels({
    required int numericId,
  }) : this(
    categoryCode: 'HW',
    numericId: numericId,
  );

  AtlasProductId.sports({
    required int numericId,
  }) : this(
    categoryCode: 'SPT',
    numericId: numericId,
  );

  // Product family/category.
  final String categoryCode;

  // Stable Atlas-owned identifier.
  final int numericId;

  String get formattedNumericId {
    return numericId.toString().padLeft(
      6,
      '0',
    );
  }

  // Human-readable identifier.
  String get value {
    return 'ATLAS-$categoryCode-$formattedNumericId';
  }

  // Atlas URI.
  Uri get uri {
    return Uri(
      scheme: 'atlas',
      host: 'product',
      pathSegments: <String>[
        value,
      ],
    );
  }

  static String _normalizeCategoryCode(
      String value,
      ) {
    final String normalized =
    value.trim().toUpperCase();

    if (normalized.isEmpty) {
      throw ArgumentError.value(
        value,
        'categoryCode',
        'Category code cannot be empty.',
      );
    }

    return normalized;
  }

  static int _validateNumericId(
      int value,
      ) {
    if (value <= 0) {
      throw ArgumentError.value(
        value,
        'numericId',
        'Numeric ID must be greater than zero.',
      );
    }

    return value;
  }

  @override
  bool operator ==(
      Object other,
      ) {
    return identical(this, other) ||
        other is AtlasProductId &&
            other.categoryCode == categoryCode &&
            other.numericId == numericId;
  }

  @override
  int get hashCode {
    return Object.hash(
      categoryCode,
      numericId,
    );
  }

  @override
  String toString() {
    return value;
  }
}

/// Immutable scanner-owned snapshot of a resolved Atlas product.
///
/// The scanner stores this snapshot instead of depending directly on the
/// catalog repository or database model. This allows the result screen and

  ResolvedProductSnapshot({
    required this.productId,
    required String displayName,
    required String categoryCode,
    String? brand,
    String? productLine,
    String? series,
    String? characterName,
    String? productNumber,
    String? variantName,
    Iterable<String> barcodes = const <String>[],
    Map<String, Object?> attributes = const <String, Object?>{},
  })  : displayName = _requireText(
    displayName,
    'displayName',
  ),
        categoryCode = _normalizeCode(categoryCode),
        brand = _normalizeOptionalText(brand),
        productLine = _normalizeOptionalText(productLine),
        series = _normalizeOptionalText(series),
        characterName = _normalizeOptionalText(characterName),
        productNumber = _normalizeOptionalText(productNumber),
        variantName = _normalizeOptionalText(variantName),
        barcodes = List<String>.unmodifiable(
          _normalizeBarcodes(barcodes),
        ),
        attributes = Map<String, Object?>.unmodifiable(
          Map<String, Object?>.from(attributes),
        );

  factory ResolvedProductSnapshot.fromCatalogProduct(
      AtlasCatalogProduct product,
      ) {
    return ResolvedProductSnapshot(
      productId: product.productId,
      displayName: product.displayName,
      categoryCode: product.categoryCode,
      brand: product.brand,
      productLine: product.productLine,
      series: product.series,
      characterName: product.characterName,
      productNumber: product.productNumber,
      variantName: product.variantName,
      barcodes: product.barcodes,
      attributes: product.attributes,
    );
  }

  final String categoryCode;

  final String? brand;

  final String? productLine;

  final String? series;

  final String? characterName;

  final String? productNumber;

  final String? variantName;

  final List<String> barcodes;

  final Map<String, Object?> attributes;

  bool get hasBarcode => barcodes.isNotEmpty;

  bool get hasVariant => variantName != null;

  bool get hasProductNumber => productNumber != null;

  String? get primaryBarcode {
    if (barcodes.isEmpty) {
      return null;
    }

    return barcodes.first;
  }

  T? attribute<T>(String key) {
    final Object? value = attributes[key];

    if (value is T) {
      return value;
    }

    return null;
  }

  bool hasAttribute(String key) {
    return attributes.containsKey(key);
  }

  bool matchesBarcode(String value) {
    final String normalized = _normalizeBarcode(value);

    return normalized.isNotEmpty &&
        barcodes.contains(normalized);

    return AtlasCatalogProduct(

      categoryCode: categoryCode,

      productLine: productLine,
      series: series,
      characterName: characterName,
      productNumber: productNumber,
      variantName: variantName,
      barcodes: barcodes,
      attributes: attributes,

    final String normalized = value.trim();

    if (normalized.isEmpty) {
      throw ArgumentError.value(
        value,
        parameterName,
        '$parameterName cannot be empty.',
      );
    }

    return normalized;

    if (normalized.isEmpty) {
      throw ArgumentError.value(

        'categoryCode',
        'Category code cannot be empty.',
      );
    }

    return normalized;

    final String normalized = value?.trim() ?? '';

  static List<String> _normalizeBarcodes(

      RegExp(r'[^0-9A-Za-z]'),
      '',
    );

  }

  @override
  int get hashCode => productId.hashCode;

  @override
  String toString() {

  }
