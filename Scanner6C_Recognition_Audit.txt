package com.example.funkovault2

import android.Manifest
import android.app.Activity
import android.content.Intent
import android.content.pm.PackageManager
import android.media.projection.MediaProjectionManager
import android.net.Uri
import android.os.Build
import android.os.Bundle
import android.provider.Settings
import androidx.core.content.ContextCompat
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodCall
import io.flutter.plugin.common.MethodChannel

class MainActivity : FlutterActivity() {
    companion object {
        private const val CHANNEL = "atlas/live_sniffer"
        private const val REQUEST_SCREEN_CAPTURE = 6201
        private const val REQUEST_NOTIFICATIONS = 6202
    }

    private var pendingScreenCaptureResult: MethodChannel.Result? = null
    private var pendingNotificationResult: MethodChannel.Result? = null

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL)
            .setMethodCallHandler(::handleLiveSnifferCall)
    }

    private fun handleLiveSnifferCall(call: MethodCall, result: MethodChannel.Result) {
        when (call.method) {
            "getStatus" -> result.success(statusMap())
            "requestOverlayPermission" -> {
                if (canDrawOverlays()) {
                    result.success(true)
                    return
                }
                val intent = Intent(
                    Settings.ACTION_MANAGE_OVERLAY_PERMISSION,
                    Uri.parse("package:$packageName"),
                )
                startActivity(intent)
                result.success(false)
            }
            "requestNotificationPermission" -> requestNotificationPermission(result)
            "start" -> requestScreenCapture(result)
            "stop" -> {
                val intent = Intent(this, AtlasLiveSnifferService::class.java).apply {
                    action = AtlasLiveSnifferService.ACTION_STOP
                }
                startService(intent)
                result.success(true)
            }
            else -> result.notImplemented()
        }
    }

    private fun canDrawOverlays(): Boolean {
        return Build.VERSION.SDK_INT < Build.VERSION_CODES.M || Settings.canDrawOverlays(this)
    }

    private fun requestNotificationPermission(result: MethodChannel.Result) {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.TIRAMISU) {
            result.success(true)
            return
        }
        if (ContextCompat.checkSelfPermission(
                this,
                Manifest.permission.POST_NOTIFICATIONS,
            ) == PackageManager.PERMISSION_GRANTED
        ) {
            result.success(true)
            return
        }
        if (pendingNotificationResult != null) {
            result.error("busy", "A notification permission request is already active.", null)
            return
        }
        pendingNotificationResult = result
        requestPermissions(
            arrayOf(Manifest.permission.POST_NOTIFICATIONS),
            REQUEST_NOTIFICATIONS,
        )
    }

    private fun requestScreenCapture(result: MethodChannel.Result) {
        if (!canDrawOverlays()) {
            result.error(
                "overlay_permission_required",
                "Display-over-other-apps permission is required before Live Sniffer can start.",
                null,
            )
            return
        }
        if (pendingScreenCaptureResult != null) {
            result.error("busy", "A screen-capture request is already active.", null)
            return
        }

        AtlasLiveSnifferStateStore.resetError()
        val manager = getSystemService(MEDIA_PROJECTION_SERVICE) as MediaProjectionManager
        pendingScreenCaptureResult = result
        startActivityForResult(manager.createScreenCaptureIntent(), REQUEST_SCREEN_CAPTURE)
    }

    @Deprecated("Deprecated in Android; retained for FlutterActivity compatibility.")
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode != REQUEST_SCREEN_CAPTURE) return

        val callback = pendingScreenCaptureResult
        pendingScreenCaptureResult = null

        if (resultCode != Activity.RESULT_OK || data == null) {
            callback?.success(false)
            return
        }

        val serviceIntent = Intent(this, AtlasLiveSnifferService::class.java).apply {
            action = AtlasLiveSnifferService.ACTION_START
            putExtra(AtlasLiveSnifferService.EXTRA_RESULT_CODE, resultCode)
            putExtra(AtlasLiveSnifferService.EXTRA_RESULT_DATA, data)
        }
        ContextCompat.startForegroundService(this, serviceIntent)
        callback?.success(true)
    }

    override fun onRequestPermissionsResult(
        requestCode: Int,
        permissions: Array<out String>,
        grantResults: IntArray,
    ) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        if (requestCode != REQUEST_NOTIFICATIONS) return

        val callback = pendingNotificationResult
        pendingNotificationResult = null
        val granted = grantResults.firstOrNull() == PackageManager.PERMISSION_GRANTED
        callback?.success(granted)
    }

    private fun statusMap(): Map<String, Any?> {
        val notificationGranted =
            Build.VERSION.SDK_INT < Build.VERSION_CODES.TIRAMISU ||
                ContextCompat.checkSelfPermission(
                    this,
                    Manifest.permission.POST_NOTIFICATIONS,
                ) == PackageManager.PERMISSION_GRANTED

        return mapOf(
            "supported" to true,
            "overlayPermissionGranted" to canDrawOverlays(),
            "notificationPermissionGranted" to notificationGranted,
            "isCapturing" to AtlasLiveSnifferStateStore.isCapturing,
            "isOverlayVisible" to AtlasLiveSnifferStateStore.isOverlayVisible,
            "lastError" to AtlasLiveSnifferStateStore.lastError,
        )
    }
}

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.Service
import android.content.Context
import android.content.Intent
import android.graphics.Color
import android.graphics.PixelFormat
import android.graphics.drawable.GradientDrawable
import android.hardware.display.DisplayManager
import android.hardware.display.VirtualDisplay
import android.media.ImageReader
import android.media.projection.MediaProjection
import android.media.projection.MediaProjectionManager
import android.os.Build

import android.os.IBinder
import android.provider.Settings
import android.view.Gravity
import android.view.MotionEvent
import android.view.WindowManager

class AtlasLiveSnifferService : Service() {
    companion object {
        const val ACTION_START = "com.example.funkovault2.action.START_LIVE_SNIFFER"
        const val ACTION_STOP = "com.example.funkovault2.action.STOP_LIVE_SNIFFER"
        const val EXTRA_RESULT_CODE = "result_code"
        const val EXTRA_RESULT_DATA = "result_data"

        private const val NOTIFICATION_CHANNEL_ID = "atlas_live_sniffer"
        private const val NOTIFICATION_ID = 6201
    }

    private var mediaProjection: MediaProjection? = null
    private var virtualDisplay: VirtualDisplay? = null
    private var imageReader: ImageReader? = null
    private var captureThread: HandlerThread? = null
    private var captureHandler: Handler? = null

    private var windowManager: WindowManager? = null
    private var overlayView: TextView? = null

    private val projectionCallback = object : MediaProjection.Callback() {
        override fun onStop() {
            stopSelfSafely()
        }
    }

    override fun onCreate() {
        super.onCreate()
        createNotificationChannel()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> {
                stopSelfSafely()
                return START_NOT_STICKY
            }
            ACTION_START -> startLiveSniffer(intent)
        }
        return START_NOT_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        releaseCapture()
        removeOverlay()
        AtlasLiveSnifferStateStore.isCapturing = false
        super.onDestroy()
    }

    private fun startLiveSniffer(intent: Intent) {
        AtlasLiveSnifferStateStore.resetError()

        if (!canDrawOverlays()) {
            AtlasLiveSnifferStateStore.fail("Display-over-other-apps permission is required.")
            stopSelf()
            return
        }

        val resultCode = intent.getIntExtra(EXTRA_RESULT_CODE, Int.MIN_VALUE)
        val resultData = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            intent.getParcelableExtra(EXTRA_RESULT_DATA, Intent::class.java)
        } else {
            @Suppress("DEPRECATION")
            intent.getParcelableExtra(EXTRA_RESULT_DATA)
        }

        if (resultCode == Int.MIN_VALUE || resultData == null) {
            AtlasLiveSnifferStateStore.fail("Screen-capture permission data was missing.")
            stopSelf()
            return
        }

        startForeground(NOTIFICATION_ID, buildNotification())

        try {
            showOverlay()
            startProjection(resultCode, resultData)
            AtlasLiveSnifferStateStore.isCapturing = true
        } catch (error: Throwable) {
            AtlasLiveSnifferStateStore.fail(error.message ?: "Atlas Live Sniffer could not start.")
            stopSelfSafely()
        }
    }

    private fun startProjection(resultCode: Int, resultData: Intent) {
        releaseCapture()

        val projectionManager =
            getSystemService(Context.MEDIA_PROJECTION_SERVICE) as MediaProjectionManager
        val projection = projectionManager.getMediaProjection(resultCode, resultData)
            ?: throw IllegalStateException("Android did not return a MediaProjection session.")

        mediaProjection = projection
        projection.registerCallback(projectionCallback, Handler(mainLooper))

        val metrics = resources.displayMetrics
        val sourceWidth = metrics.widthPixels.coerceAtLeast(1)
        val sourceHeight = metrics.heightPixels.coerceAtLeast(1)
        val density = metrics.densityDpi.coerceAtLeast(1)

        // 6C foundation intentionally samples into a reduced surface and drops the
        // frames. Recognition is connected in the next 6C checkpoint.
        val longEdge = 960
        val scale = if (sourceWidth >= sourceHeight) {
            longEdge.toDouble() / sourceWidth.toDouble()
        } else {
            longEdge.toDouble() / sourceHeight.toDouble()
        }.coerceAtMost(1.0)

        val width = (sourceWidth * scale).toInt().coerceAtLeast(1)
        val height = (sourceHeight * scale).toInt().coerceAtLeast(1)

        val thread = HandlerThread("AtlasLiveSnifferCapture").also { it.start() }
        captureThread = thread
        captureHandler = Handler(thread.looper)

        val reader = ImageReader.newInstance(width, height, PixelFormat.RGBA_8888, 2)
        imageReader = reader
        reader.setOnImageAvailableListener({ source ->
            source.acquireLatestImage()?.close()
        }, captureHandler)

        virtualDisplay = projection.createVirtualDisplay(
            "AtlasLiveSniffer",
            width,
            height,
            density,
            DisplayManager.VIRTUAL_DISPLAY_FLAG_AUTO_MIRROR,
            reader.surface,
            null,
            captureHandler,
        )
    }

    private fun canDrawOverlays(): Boolean {
        return Build.VERSION.SDK_INT < Build.VERSION_CODES.M || Settings.canDrawOverlays(this)
    }

    private fun showOverlay() {
        removeOverlay()

        val manager = getSystemService(Context.WINDOW_SERVICE) as WindowManager
        windowManager = manager

        val textView = TextView(this).apply {
            text = "ATLAS LIVE\nListening..."
            setTextColor(Color.WHITE)
            textSize = 13f
            gravity = Gravity.CENTER
            setPadding(28, 18, 28, 18)
            background = GradientDrawable().apply {
                cornerRadius = 28f
                setColor(Color.argb(225, 18, 22, 28))
                setStroke(2, Color.argb(210, 110, 168, 255))
            }
            elevation = 12f
        }

        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
            } else {
                @Suppress("DEPRECATION")
                WindowManager.LayoutParams.TYPE_PHONE
            },
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL,
            PixelFormat.TRANSLUCENT,
        ).apply {
            gravity = Gravity.TOP or Gravity.END
            x = 24
            y = 180
        }

        var startX = 0
        var startY = 0
        var touchX = 0f
        var touchY = 0f

        textView.setOnTouchListener { _, event ->
            when (event.action) {
                MotionEvent.ACTION_DOWN -> {
                    startX = params.x
                    startY = params.y
                    touchX = event.rawX
                    touchY = event.rawY
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    params.x = startX - (event.rawX - touchX).toInt()
                    params.y = startY + (event.rawY - touchY).toInt()
                    manager.updateViewLayout(textView, params)
                    true
                }
                else -> false
            }
        }

        manager.addView(textView, params)
        overlayView = textView
        AtlasLiveSnifferStateStore.isOverlayVisible = true
    }

    private fun removeOverlay() {
        val view = overlayView ?: return
        try {
            windowManager?.removeView(view)
        } catch (_: Throwable) {
            // The view may already have been detached by Android.
        }
        overlayView = null
        AtlasLiveSnifferStateStore.isOverlayVisible = false
    }

    private fun releaseCapture() {
        virtualDisplay?.release()
        virtualDisplay = null

        imageReader?.setOnImageAvailableListener(null, null)
        imageReader?.close()
        imageReader = null

        mediaProjection?.unregisterCallback(projectionCallback)
        mediaProjection?.stop()
        mediaProjection = null

        captureHandler = null
        captureThread?.quitSafely()
        captureThread = null
    }

    private fun stopSelfSafely() {
        AtlasLiveSnifferStateStore.isCapturing = false
        releaseCapture()
        removeOverlay()
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            stopForeground(STOP_FOREGROUND_REMOVE)
        } else {
            @Suppress("DEPRECATION")
            stopForeground(true)
        }
        stopSelf()
    }

    private fun buildNotification(): Notification {
        val builder = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            Notification.Builder(this, NOTIFICATION_CHANNEL_ID)
        } else {
            @Suppress("DEPRECATION")
            Notification.Builder(this)
        }

        return builder
            .setSmallIcon(android.R.drawable.ic_menu_view)
            .setContentTitle("Atlas Live Sniffer")
            .setContentText("Screen intelligence is active.")
            .setOngoing(true)
            .build()
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.O) return
        val manager = getSystemService(NotificationManager::class.java)
        val channel = NotificationChannel(
            NOTIFICATION_CHANNEL_ID,
            "Atlas Live Sniffer",
            NotificationManager.IMPORTANCE_LOW,
        )
        channel.description = "Required while Atlas Live Sniffer is capturing the screen."
        manager.createNotificationChannel(channel)
    }
}

import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart';

import 'atlas_live_sniffer_status.dart';

abstract interface class AtlasLiveSnifferPlatform {
  Future<AtlasLiveSnifferStatus> getStatus();

  Future<bool> requestOverlayPermission();

  Future<bool> requestNotificationPermission();

  Future<bool> start();

  Future<bool> stop();
}

final class AtlasLiveSnifferBridge implements AtlasLiveSnifferPlatform {
  const AtlasLiveSnifferBridge({MethodChannel? channel})
    : _channel = channel ?? const MethodChannel('atlas/live_sniffer');

  final MethodChannel _channel;

  bool get _isAndroid => defaultTargetPlatform == TargetPlatform.android;

  @override
  Future<AtlasLiveSnifferStatus> getStatus() async {
    if (!_isAndroid) {
      return const AtlasLiveSnifferStatus.unsupported();
    }

    final Map<Object?, Object?>? result = await _channel
        .invokeMapMethod<Object?, Object?>('getStatus');

    if (result == null) {
      return const AtlasLiveSnifferStatus.unsupported();
    }

    return AtlasLiveSnifferStatus.fromMap(result);
  }

  @override
  Future<bool> requestOverlayPermission() async {
    if (!_isAndroid) return false;
    return await _channel.invokeMethod<bool>('requestOverlayPermission') ??
        false;
  }

  @override
  Future<bool> requestNotificationPermission() async {
    if (!_isAndroid) return false;
    return await _channel.invokeMethod<bool>('requestNotificationPermission') ??
        false;
  }

  @override
  Future<bool> start() async {
    if (!_isAndroid) return false;
    return await _channel.invokeMethod<bool>('start') ?? false;
  }

  @override
  Future<bool> stop() async {
    if (!_isAndroid) return false;
    return await _channel.invokeMethod<bool>('stop') ?? false;
  }
}

import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart';

import 'atlas_live_sniffer_bridge.dart';
import 'atlas_live_sniffer_status.dart';

final class AtlasLiveSnifferController extends ChangeNotifier {
  AtlasLiveSnifferController({AtlasLiveSnifferPlatform? platform})
    : _platform = platform ?? const AtlasLiveSnifferBridge();

  final AtlasLiveSnifferPlatform _platform;

  AtlasLiveSnifferStatus _status = const AtlasLiveSnifferStatus.unsupported();
  bool _busy = false;
  String? _message;

  AtlasLiveSnifferStatus get status => _status;
  bool get busy => _busy;
  String? get message => _message;

  Future<void> refresh() async {
    await _run(() async {
      _status = await _platform.getStatus();
    });
  }

  Future<void> requestOverlayPermission() async {
    await _run(() async {
      final bool alreadyGranted = await _platform.requestOverlayPermission();
      _status = await _platform.getStatus();
      _message = alreadyGranted || _status.overlayPermissionGranted
          ? 'Display-over-other-apps permission is ready.'
          : 'Android opened the overlay permission screen. Grant permission, '
                'then return to Atlas.';
    });
  }

  Future<void> requestNotificationPermission() async {
    await _run(() async {
      final bool granted = await _platform.requestNotificationPermission();
      _status = await _platform.getStatus();
      _message = granted || _status.notificationPermissionGranted
          ? 'Notification permission is ready.'
          : 'Notification permission was not granted.';
    });
  }

  Future<void> start() async {
    await _run(() async {
      _status = await _platform.getStatus();

      if (!_status.supported) {
        _message = 'Live Sniffer is currently available on Android only.';
        return;
      }

      if (!_status.overlayPermissionGranted) {
        _message =
            'Grant Display over other apps before starting Live Sniffer.';
        return;
      }

      if (!_status.notificationPermissionGranted) {
        _message =
            'Grant notification permission before starting Live Sniffer.';
        return;
      }

      final bool started = await _platform.start();
      _status = await _platform.getStatus();
      _message = started
          ? 'Atlas Live Sniffer started. You can switch to another app.'
          : 'Screen capture was cancelled.';
    });
  }

  Future<void> stop() async {
    await _run(() async {
      await _platform.stop();
      _status = await _platform.getStatus();
      _message = 'Atlas Live Sniffer stopped.';
    });
  }

  void clearMessage() {
    if (_message == null) return;
    _message = null;
    notifyListeners();
  }

  Future<void> _run(Future<void> Function() action) async {
    if (_busy) return;

    _busy = true;
    _message = null;
    notifyListeners();

    try {
      await action();
    } on PlatformException catch (error) {
      _message = error.message ?? 'Android could not complete that request.';
      _status = await _safeStatus();
    } catch (error) {
      _message = error.toString();
      _status = await _safeStatus();
    } finally {
      _busy = false;
      notifyListeners();
    }
  }

  Future<AtlasLiveSnifferStatus> _safeStatus() async {
    try {
      return await _platform.getStatus();
    } catch (_) {
      return _status;
    }
  }
}

final class AtlasLiveSnifferStatus {
  const AtlasLiveSnifferStatus({
    required this.supported,
    required this.overlayPermissionGranted,
    required this.notificationPermissionGranted,
    required this.isCapturing,
    required this.isOverlayVisible,
    this.lastError,
  });

  const AtlasLiveSnifferStatus.unsupported()
    : supported = false,
      overlayPermissionGranted = false,
      notificationPermissionGranted = false,
      isCapturing = false,
      isOverlayVisible = false,
      lastError = null;

  factory AtlasLiveSnifferStatus.fromMap(Map<Object?, Object?> map) {
    return AtlasLiveSnifferStatus(
      supported: map['supported'] == true,
      overlayPermissionGranted: map['overlayPermissionGranted'] == true,
      notificationPermissionGranted:
          map['notificationPermissionGranted'] == true,
      isCapturing: map['isCapturing'] == true,
      isOverlayVisible: map['isOverlayVisible'] == true,
      lastError: _stringOrNull(map['lastError']),
    );
  }

  final bool supported;
  final bool overlayPermissionGranted;
  final bool notificationPermissionGranted;
  final bool isCapturing;
  final bool isOverlayVisible;
  final String? lastError;

  bool get isReadyToStart =>
      supported && overlayPermissionGranted && notificationPermissionGranted;

  static String? _stringOrNull(Object? value) {
    final String normalized = value?.toString().trim() ?? '';
    return normalized.isEmpty ? null : normalized;
  }
}

final class AtlasContinuousPerceptionConfig {
  const AtlasContinuousPerceptionConfig({
    this.minimumFrameInterval = const Duration(milliseconds: 250),
  });

  /// Default 250 ms cadence targets approximately 4 FPS, inside the 3-5 FPS
  /// continuous-perception budget. Callers may tune this without changing the
  /// tracking or evidence contracts.
  final Duration minimumFrameInterval;

/// Shared 6A foundation for camera Shelf Scan and future live-feed ingestion.
///
/// Responsibilities:
/// - sample cadence enforcement
/// - frame-quality fast-fail
/// - reuse AtlasObjectTracker for spatial persistence
/// - reuse AtlasEvidenceAccumulator for stable barcode agreement
/// - emit transient candidate/result models for HUD consumers
/// - optionally cross the existing Scanner 4 submission boundary
///
/// This coordinator never mutates the Vault and never rewrites product identity.
final class AtlasContinuousPerceptionCoordinator {
  AtlasContinuousPerceptionCoordinator({
    AtlasObjectTracker? objectTracker,
    AtlasEvidenceAccumulator? evidenceAccumulator,
    AtlasFrameTriageGate? frameTriageGate,
    this.submitStableCandidate,
    this.config = const AtlasContinuousPerceptionConfig(),
  }) : objectTracker = objectTracker ?? AtlasObjectTracker(),
       evidenceAccumulator = evidenceAccumulator ?? AtlasEvidenceAccumulator(),
       frameTriageGate = frameTriageGate ?? AtlasFrameTriageGate() {
    if (config.minimumFrameInterval.isNegative) {
      throw ArgumentError.value(
        config.minimumFrameInterval,
        'minimumFrameInterval',
        'minimumFrameInterval cannot be negative.',
      );
    }
  }

  final AtlasObjectTracker objectTracker;
  final AtlasEvidenceAccumulator evidenceAccumulator;
  final AtlasFrameTriageGate frameTriageGate;
  final AtlasStableCandidateSubmitter? submitStableCandidate;
  final AtlasContinuousPerceptionConfig config;

  final StreamController<AtlasPerceptionFrameResult> _resultsController =
      StreamController<AtlasPerceptionFrameResult>.broadcast(sync: true);

  final Set<String> _knownTrackIds = <String>{};
  final Set<String> _submittedTrackIds = <String>{};

  DateTime? _lastInputAt;
  DateTime? _lastProcessedAt;
  AtlasPerceptionFrameResult? _latestResult;
  bool _isDisposed = false;

  Stream<AtlasPerceptionFrameResult> get results => _resultsController.stream;

  AtlasPerceptionFrameResult? get latestResult => _latestResult;

  AtlasPerceptionFrameResult processFrame({
    required DateTime timestamp,
    required List<AtlasScanObservation> observations,
    AtlasFrameQualitySample? qualitySample,
  }) {
    if (_isDisposed) {
      throw StateError('AtlasContinuousPerceptionCoordinator is disposed.');
    }

    final DateTime normalizedTimestamp = timestamp.toUtc();
    final DateTime? previousInput = _lastInputAt;
    if (previousInput != null && normalizedTimestamp.isBefore(previousInput)) {
      throw ArgumentError.value(
        timestamp,
        'timestamp',
        'Frame timestamps must be monotonic.',
      );
    }
    _lastInputAt = normalizedTimestamp;

    final AtlasFrameQualityAssessment qualityAssessment = frameTriageGate
        .evaluate(qualitySample ?? AtlasFrameQualitySample());

    final DateTime? previousProcessed = _lastProcessedAt;
    if (previousProcessed != null &&
        normalizedTimestamp.difference(previousProcessed) <
            config.minimumFrameInterval) {
      return _emit(
        AtlasPerceptionFrameResult(
          timestamp: normalizedTimestamp,
          disposition: AtlasPerceptionFrameDisposition.skippedCadence,
          qualityAssessment: qualityAssessment,
          candidates: _projectCurrentCandidates(),
        ),
      );
    }

    _lastProcessedAt = normalizedTimestamp;

    final List<AtlasObjectTrack> tracked = objectTracker.processFrame(
      observations: qualityAssessment.isAccepted
          ? observations
          : const <AtlasScanObservation>[],
      timestamp: normalizedTimestamp,
    );

    final Set<String> currentTrackIds = tracked
        .map((AtlasObjectTrack track) => track.trackId)
        .toSet();
    final Set<String> expiredTrackIds = _knownTrackIds.difference(
      currentTrackIds,
    );
    _knownTrackIds
      ..clear()
      ..addAll(currentTrackIds);

    final Set<String> submittedThisFrame = <String>{};

    if (qualityAssessment.isAccepted && submitStableCandidate != null) {
      for (final AtlasObjectTrack track in tracked) {
        if (track.isCooldown || _submittedTrackIds.contains(track.trackId)) {
          continue;
        }

        final AtlasAccumulatedEvidence? evidence = evidenceAccumulator
            .evaluateTrack(track);
        if (evidence == null || !evidence.isReadyForResolution) {
          continue;
        }

        final bool accepted = submitStableCandidate!(
          trackId: track.trackId,
          stableCode: evidence.stableCode,
        );

        if (accepted) {
          _submittedTrackIds.add(track.trackId);
          submittedThisFrame.add(track.trackId);
          objectTracker.markCooldown(
            trackId: track.trackId,
            startedAt: normalizedTimestamp,
          );
        }
      }
    }

    final AtlasPerceptionFrameResult result = AtlasPerceptionFrameResult(
      timestamp: normalizedTimestamp,
      disposition: qualityAssessment.isAccepted
          ? AtlasPerceptionFrameDisposition.processed
          : AtlasPerceptionFrameDisposition.rejectedQuality,
      qualityAssessment: qualityAssessment,
      candidates: _projectCurrentCandidates(),
      expiredTrackIds: expiredTrackIds,
      submittedTrackIds: submittedThisFrame,
    );

    return _emit(result);
  }

  void reset() {
    if (_isDisposed) return;
    objectTracker.reset();
    _knownTrackIds.clear();
    _submittedTrackIds.clear();
    _lastInputAt = null;
    _lastProcessedAt = null;
    _latestResult = null;
  }

  Future<void> dispose() async {
    if (_isDisposed) return;
    _isDisposed = true;
    await _resultsController.close();
  }

  List<AtlasContinuousCandidate> _projectCurrentCandidates() {
    return objectTracker.tracks
        .map((AtlasObjectTrack track) {
          final AtlasAccumulatedEvidence? evidence = evidenceAccumulator
              .evaluateTrack(track);
          return AtlasContinuousCandidateProjector.project(
            track: track,
            accumulatedEvidence: evidence,
          );
        })
        .toList(growable: false);
  }

  AtlasPerceptionFrameResult _emit(AtlasPerceptionFrameResult result) {
    _latestResult = result;
    _resultsController.add(result);
    return result;
  }
}

/// Coordinates stable physical tracks with Atlas's existing resolution stack.
///
/// This class owns queueing and ledger state, but it does not mutate the Vault.
/// A resolved item remains separate from an explicitly committed item.
class AtlasContinuousScanCoordinator {
  AtlasContinuousScanCoordinator({
    AtlasResolutionQueue? resolutionQueue,
    AtlasScanLedger? ledger,
    AtlasBarcodeCatalogResolver? catalogResolver,
    this.onChanged,
  }) : resolutionQueue =
           resolutionQueue ?? AtlasResolutionQueue(maxConcurrentJobs: 4),
       ledger = ledger ?? AtlasScanLedger(),
       _catalogResolver = catalogResolver ?? AtlasBarcodeCatalogResolver();

  final AtlasResolutionQueue resolutionQueue;
  final AtlasScanLedger ledger;
  final AtlasBarcodeCatalogResolver _catalogResolver;
  final AtlasContinuousScanChanged? onChanged;

  final Map<String, AtlasContinuousScanResult> _resultsByTrack =
      <String, AtlasContinuousScanResult>{};

  bool _isDisposed = false;

  List<AtlasContinuousScanResult> get results {
    return List<AtlasContinuousScanResult>.unmodifiable(_resultsByTrack.values);
  }

  AtlasContinuousScanResult? resultForTrack(String trackId) {
    return _resultsByTrack[trackId.trim()];
  }

  AtlasContinuousScanSnapshot get snapshot {
    final AtlasResolutionQueueSnapshot queueSnapshot = resolutionQueue.snapshot;

    int observed = 0;
    int queued = 0;
    int resolving = 0;
    int resolved = 0;
    int review = 0;
    int failed = 0;
    int committed = 0;

    for (final AtlasScanLedgerEntry entry in ledger.entries) {
      switch (entry.status) {
        case AtlasScanLedgerStatus.observed:
          observed++;
        case AtlasScanLedgerStatus.queued:
          queued++;
        case AtlasScanLedgerStatus.resolving:
          resolving++;
        case AtlasScanLedgerStatus.resolved:
          resolved++;
        case AtlasScanLedgerStatus.needsReview:
          review++;
        case AtlasScanLedgerStatus.failed:
          failed++;
        case AtlasScanLedgerStatus.committed:
          committed++;
      }
    }

    return AtlasContinuousScanSnapshot(
      activeResolutionCount: queueSnapshot.activeCount,
      pendingResolutionCount: queueSnapshot.pendingCount,
      observedCount: observed,
      queuedCount: queued,
      resolvingCount: resolving,
      resolvedCount: resolved,
      reviewCount: review,
      failedCount: failed,
      committedCount: committed,
    );
  }

  /// Submits stable evidence from one physical track.
  ///
  /// Returns true only when a new background resolution job was accepted.
  bool submit({required String trackId, required ScanCode stableCode}) {
    if (_isDisposed) {
      return false;
    }

    final String normalizedTrackId = trackId.trim();

    if (normalizedTrackId.isEmpty) {
      throw ArgumentError.value(trackId, 'trackId', 'trackId cannot be empty.');
    }

    final DateTime now = DateTime.now().toUtc();
    final AtlasScanLedgerEntry? existing = ledger.entryForTrack(
      normalizedTrackId,
    );

    if (existing != null) {
      return false;
    }

    ledger.observe(trackId: normalizedTrackId, stableCode: stableCode, at: now);

    ledger.markQueued(trackId: normalizedTrackId, at: now);

    _notifyChanged();

    _enqueueResolution(
      trackId: normalizedTrackId,
      stableCode: stableCode,
      createdAt: now,
    );

    return true;
  }

  /// Explicitly retries a failed physical track.
  ///
  /// Retry never creates a second ledger entry and never auto-commits. The
  /// original stable barcode evidence is reused for a new resolution attempt.
  bool retryFailedTrack({required String trackId}) {
    if (_isDisposed) {
      return false;
    }

    final String normalizedTrackId = trackId.trim();

    if (normalizedTrackId.isEmpty ||
        resolutionQueue.containsTrack(normalizedTrackId)) {
      return false;
    }

    final AtlasScanLedgerEntry? entry = ledger.entryForTrack(normalizedTrackId);

    if (entry == null || !entry.hasFailed) {
      return false;
    }

    final DateTime now = DateTime.now().toUtc();

    ledger.markQueued(trackId: normalizedTrackId, at: now, allowRetry: true);

    _resultsByTrack.remove(normalizedTrackId);

    _notifyChanged();

    _enqueueResolution(
      trackId: normalizedTrackId,
      stableCode: entry.stableCode,
      createdAt: now,
    );

    return true;
  }

  bool markCommitted({required String trackId, DateTime? at}) {
    if (_isDisposed) {
      return false;
    }

    final AtlasScanLedgerEntry? entry = ledger.entryForTrack(trackId);

    if (entry == null) {
      return false;
    }

    if (entry.isCommitted) {
      return true;
    }

    if (!entry.isResolved && !entry.needsReview) {
      return false;
    }

    ledger.markCommitted(trackId: trackId, at: at ?? DateTime.now().toUtc());

    _notifyChanged();
    return true;
  }

  bool applyInspection({
    required String trackId,
    required ScanSession session,
  }) {
    if (_isDisposed) return false;
    final normalizedTrackId = trackId.trim();
    final existing = _resultsByTrack[normalizedTrackId];
    if (existing == null || existing.session.sessionId != session.sessionId) {
      return false;
    }
    final assessment = session.authenticityContribution?.assessment;
    if (assessment == null) return false;
    _resultsByTrack[normalizedTrackId] = existing.copyWith(session: session);
    ledger.applyInspectionReview(
      trackId: normalizedTrackId,
      requiresReview:
          assessment.requiresReview ||
          session.requiresUserAction ||
          !existing.catalogResolution.hasSingleMatch,
      at: DateTime.now().toUtc(),
    );
    _notifyChanged();
    return true;
  }

  bool get canResetCompletedSession {
    final List<AtlasScanLedgerEntry> entries = ledger.entries;

    return !_isDisposed &&
        resolutionQueue.snapshot.isIdle &&
        entries.isNotEmpty &&
        entries.every((AtlasScanLedgerEntry entry) => entry.isCommitted);
  }

  bool resetCompletedSession() {
    if (!canResetCompletedSession) {
      return false;
    }

    ledger.clear();
    _resultsByTrack.clear();
    _notifyChanged();
    return true;
  }

  void dispose() {
    if (_isDisposed) {
      return;
    }

    _isDisposed = true;
    resolutionQueue.dispose();
  }

  void _enqueueResolution({
    required String trackId,
    required ScanCode stableCode,
    required DateTime createdAt,
  }) {
    final String jobId = _createJobId(trackId: trackId, createdAt: createdAt);

    unawaited(
      resolutionQueue
          .enqueue(
            AtlasResolutionJob(
              jobId: jobId,
              trackId: trackId,
              createdAt: createdAt,
              execute: () =>
                  _resolveTrack(trackId: trackId, stableCode: stableCode),
            ),
          )
          .catchError((Object error, StackTrace stackTrace) {
            _handleQueueFailure(trackId: trackId, error: error);
          }),
    );
  }

  Future<void> _resolveTrack({
    required String trackId,
    required ScanCode stableCode,
  }) async {
    if (_isDisposed) {
      return;
    }

    final DateTime startedAt = DateTime.now().toUtc();

    ledger.markResolving(trackId: trackId, at: startedAt);
    _notifyChanged();

    try {
      final String executionId = _createExecutionId(
        startedAt,
        stableCode.normalizedValue,
      );

      final EngineContext engineContext = EngineContext(
        requestId: executionId,
        createdAt: startedAt,
        connectivity: EngineConnectivity.unknown,
        deviceType: EngineDeviceType.mobile,
        deviceCapabilities: const <String>{
          BarcodeIntelligenceEngine.capabilityId,
        },
        attributes: <String, Object?>{
          'scanSource': ScanCodeSource.camera.name,
          'scannerType': ScannerType.mobileScanner.name,
          'barcodeFormat': stableCode.format.name,
          'trackId': trackId,
          'continuousScan': true,
        },
      );

      final ScanSession initialSession = ScanSession(
        sessionId: executionId,
        startedAt: startedAt,
        status: ScanSessionStatus.analyzing,
        intent: ScanIntent.identifyProduct,
        performanceMode: ScanPerformanceMode.fastBarcodeOnly,
        scannerType: ScannerType.mobileScanner,
        detectedCodes: <ScanCode>[stableCode],
        bestCandidate: stableCode,
      );

      final pipeline = StandardScanPipeline.create(engineContext);
      final result = await pipeline.process(initialSession, engineContext);

      final ScanSession finalSession = result.finalSession;

      final ScanCode? resolvedCode =
          finalSession.bestCandidate ?? _firstDetectedCode(finalSession);

      final AtlasBarcodeResolution catalogResolution;

      if (resolvedCode == null ||
          !resolvedCode.isValid ||
          !resolvedCode.isProductBarcode) {
        catalogResolution = AtlasBarcodeResolution.invalid(
          barcode: resolvedCode?.normalizedValue ?? '',
        );
      } else if (finalSession.catalogProduct != null) {
        catalogResolution = AtlasBarcodeResolution.matches(
          barcode: resolvedCode.normalizedValue,
          matches: <AtlasCatalogProduct>[finalSession.catalogProduct!],
        );
      } else {
        catalogResolution = await _catalogResolver.resolve(
          resolvedCode.normalizedValue,
        );
      }

      final DateTime completedAt = DateTime.now().toUtc();

      final AtlasContinuousScanResult continuousResult =
          AtlasContinuousScanResult(
            trackId: trackId,
            session: finalSession,
            catalogResolution: catalogResolution,
            completedAt: completedAt,
          );

      _resultsByTrack[trackId] = continuousResult;

      if (continuousResult.requiresUserReview) {
        ledger.markNeedsReview(
          trackId: trackId,
          resolutionSessionId: finalSession.sessionId,
          at: completedAt,
        );
      } else {
        ledger.markResolved(
          trackId: trackId,
          resolutionSessionId: finalSession.sessionId,
          at: completedAt,
        );
      }
    } catch (error) {
      ledger.markFailed(
        trackId: trackId,
        error: error,
        at: DateTime.now().toUtc(),
      );
      rethrow;
    } finally {
      _notifyChanged();
    }
  }

  void _handleQueueFailure({required String trackId, required Object error}) {
    final AtlasScanLedgerEntry? entry = ledger.entryForTrack(trackId);

    if (entry == null || entry.hasFailed) {
      _notifyChanged();
      return;
    }

    if (entry.isQueued) {
      ledger.markResolving(trackId: trackId, at: DateTime.now().toUtc());
    }

    final AtlasScanLedgerEntry? current = ledger.entryForTrack(trackId);

    if (current != null && current.isResolving) {
      ledger.markFailed(
        trackId: trackId,
        error: error,
        at: DateTime.now().toUtc(),
      );
    }

    _notifyChanged();
  }

  ScanCode? _firstDetectedCode(ScanSession session) {
    if (session.detectedCodes.isEmpty) {
      return null;
    }

    return session.detectedCodes.first;
  }

  String _createExecutionId(DateTime createdAt, String code) {
    final int codeHash = code.hashCode.abs();

    return 'scan_'
        '${createdAt.microsecondsSinceEpoch}_'
        '$codeHash';
  }

  String _createJobId({required String trackId, required DateTime createdAt}) {
    return 'resolution_'
        '${createdAt.microsecondsSinceEpoch}_'
        '${trackId.hashCode.abs()}';
  }

  void _notifyChanged() {
    if (_isDisposed) {
      return;
    }

    onChanged?.call();
  }
}

/// Describes the outcome of resolving a scanned barcode against
/// the Atlas product catalog.
enum AtlasBarcodeResolutionStatus {
  invalidBarcode,
  noMatch,
  singleMatch,
  multipleMatches,
}

/// Immutable result returned after checking a barcode against
/// the Atlas product catalog.
final class AtlasBarcodeResolution {
  const AtlasBarcodeResolution._({
    required this.barcode,
    required this.status,
    required this.matches,
  });

  factory AtlasBarcodeResolution.invalid({
    required String barcode,
  }) {
    return AtlasBarcodeResolution._(
      barcode: barcode,
      status: AtlasBarcodeResolutionStatus.invalidBarcode,
      matches: const <AtlasCatalogProduct>[],
    );
  }

  factory AtlasBarcodeResolution.noMatch({
    required String barcode,
  }) {
    return AtlasBarcodeResolution._(
      barcode: barcode,
      status: AtlasBarcodeResolutionStatus.noMatch,
      matches: const <AtlasCatalogProduct>[],
    );
  }

  factory AtlasBarcodeResolution.matches({
    required String barcode,
    required List<AtlasCatalogProduct> matches,
  }) {
    final List<AtlasCatalogProduct> immutableMatches =
    List<AtlasCatalogProduct>.unmodifiable(
      matches,
    );

    return AtlasBarcodeResolution._(
      barcode: barcode,
      status: immutableMatches.length == 1
          ? AtlasBarcodeResolutionStatus.singleMatch
          : AtlasBarcodeResolutionStatus.multipleMatches,
      matches: immutableMatches,
    );
  }

  final String barcode;

  final AtlasBarcodeResolutionStatus status;

  final List<AtlasCatalogProduct> matches;

  bool get isValidBarcode {
    return status != AtlasBarcodeResolutionStatus.invalidBarcode;
  }

  bool get hasMatch {
    return matches.isNotEmpty;
  }

  bool get hasSingleMatch {
    return status == AtlasBarcodeResolutionStatus.singleMatch;
  }

  bool get hasMultipleMatches {
    return status == AtlasBarcodeResolutionStatus.multipleMatches;
  }

  AtlasCatalogProduct? get singleMatch {
    if (!hasSingleMatch) {
      return null;
    }

    return matches.first;
  }
}

/// Resolves scanner barcode values through the shared Atlas catalog service.
///
/// Scanner screens and scan workflows should use this resolver rather than
/// creating repository implementations or performing catalog scans directly.
final class AtlasBarcodeCatalogResolver {
  AtlasBarcodeCatalogResolver({
    AtlasCatalogService? catalogService,
  }) : _catalogService =
      catalogService ?? AtlasCatalogService.instance;

  final AtlasCatalogService _catalogService;

  Future<AtlasBarcodeResolution> resolve(
      String barcode,
      ) async {
    final String normalizedBarcode = _normalizeBarcode(
      barcode,
    );

    if (normalizedBarcode.isEmpty) {
      return AtlasBarcodeResolution.invalid(
        barcode: barcode,
      );
    }

    final List<AtlasCatalogProduct> matches =
    await _catalogService.findByBarcode(
      normalizedBarcode,
    );

    if (matches.isEmpty) {
      return AtlasBarcodeResolution.noMatch(
        barcode: normalizedBarcode,
      );
    }

    return AtlasBarcodeResolution.matches(
      barcode: normalizedBarcode,
      matches: matches,
    );
  }

  static String _normalizeBarcode(
      String value,
      ) {
    return value.replaceAll(
      RegExp(r'[^0-9A-Za-z]'),
      '',
    );
  }
}

import 'package:funkovault2/atlas/collection/models/atlas_collector_state.dart';
import 'package:funkovault2/models/pop.dart';

class AtlasCollectorStateEngine {
  const AtlasCollectorStateEngine();

  AtlasCollectorState evaluate({
    required Pop product,
    required List<Pop> collectionPops,
    String scannedBarcode = '',
    Set<String> ownedWishlistIdentityKeys =
    const <String>{},
  }) {
    final Set<String> scannedBarcodeCandidates =
    _barcodeCandidates(
      scannedBarcode,
    );

    final String productIdentityKey =
    product.collectibleIdentityKey.trim();

    final String productSimilarityKey =
    product.similarityReviewKey.trim();

    final List<Pop> ownedMatches = <Pop>[];
    final List<Pop> wishlistMatches = <Pop>[];
    final List<Pop> relatedMatches = <Pop>[];

    for (final Pop collectionPop in collectionPops) {
      final bool isExactMatch = _isExactMatch(
        product: product,
        collectionPop: collectionPop,
        productIdentityKey: productIdentityKey,
        scannedBarcodeCandidates:
        scannedBarcodeCandidates,
      );

      if (isExactMatch) {
        if (collectionPop.isWishlist) {
          wishlistMatches.add(
            collectionPop,
          );
        } else {
          ownedMatches.add(
            collectionPop,
          );
        }

        continue;
      }

      final String collectionSimilarityKey =
      collectionPop.similarityReviewKey.trim();

      final bool isRelatedMatch =
          productSimilarityKey.isNotEmpty &&
              collectionSimilarityKey.isNotEmpty &&
              collectionSimilarityKey ==
                  productSimilarityKey;

      if (isRelatedMatch) {
        relatedMatches.add(
          collectionPop,
        );
      }
    }

    final bool isProductIdentityWishlisted =
        productIdentityKey.isNotEmpty &&
            ownedWishlistIdentityKeys.contains(
              productIdentityKey,
            );

    final bool isOwnedMatchWishlisted =
    ownedMatches.any(
          (Pop ownedPop) {
        final String identityKey =
        ownedPop.collectibleIdentityKey.trim();

        return identityKey.isNotEmpty &&
            ownedWishlistIdentityKeys.contains(
              identityKey,
            );
      },
    );

    final bool isOwnedWishlisted =
        isProductIdentityWishlisted ||
            isOwnedMatchWishlisted;

    return AtlasCollectorState(
      ownedMatches: ownedMatches,
      wishlistMatches: wishlistMatches,
      relatedMatches: relatedMatches,
      isOwnedWishlisted: isOwnedWishlisted,
    );
  }

  List<Pop> findByBarcode({
    required String barcode,
    required List<Pop> collectionPops,
  }) {
    final Set<String> barcodeCandidates =
    _barcodeCandidates(
      barcode,
    );

    if (barcodeCandidates.isEmpty) {
      return const <Pop>[];
    }

    final List<Pop> matches = <Pop>[];

    for (final Pop pop in collectionPops) {
      final Set<String> popBarcodeCandidates =
      _barcodeCandidates(
        pop.sourceUpc,
      );

      final bool matchesBarcode =
          popBarcodeCandidates.isNotEmpty &&
              barcodeCandidates.any(
                popBarcodeCandidates.contains,
              );

      if (matchesBarcode) {
        matches.add(
          pop,
        );
      }
    }

    return List<Pop>.unmodifiable(
      matches,
    );
  }

  bool _isExactMatch({
    required Pop product,
    required Pop collectionPop,
    required String productIdentityKey,
    required Set<String> scannedBarcodeCandidates,
  }) {
    final String collectionIdentityKey =
    collectionPop.collectibleIdentityKey.trim();

    // Strongest release-level comparison.
    if (productIdentityKey.isNotEmpty &&
        collectionIdentityKey.isNotEmpty &&
        productIdentityKey == collectionIdentityKey) {
      return true;
    }

    final Set<String> collectionBarcodeCandidates =
    _barcodeCandidates(
      collectionPop.sourceUpc,
    );

    final bool isBarcodeMatch =
        scannedBarcodeCandidates.isNotEmpty &&
            collectionBarcodeCandidates.isNotEmpty &&
            scannedBarcodeCandidates.any(
              collectionBarcodeCandidates.contains,
            );

    if (isBarcodeMatch) {
      return true;
    }

    final String productRecordKey =
    product.collectionRecordKey.trim();

    final String collectionRecordKey =
    collectionPop.collectionRecordKey.trim();

    return productRecordKey.isNotEmpty &&
        collectionRecordKey.isNotEmpty &&
        productRecordKey == collectionRecordKey;
  }

  Set<String> _barcodeCandidates(
      String value,
      ) {
    final String normalizedBarcode =
    _normalizeBarcode(
      value,
    );

    if (normalizedBarcode.isEmpty) {
      return const <String>{};
    }

    final Set<String> candidates = <String>{
      normalizedBarcode,
    };

    if (normalizedBarcode.length == 13 &&
        normalizedBarcode.startsWith('0')) {
      candidates.add(
        normalizedBarcode.substring(1),
      );
    }

    if (normalizedBarcode.length == 12) {
      candidates.add(
        '0$normalizedBarcode',
      );
    }

    return candidates;
  }

  String _normalizeBarcode(
      String value,
      ) {
    return value.replaceAll(
      RegExp(r'[^0-9A-Za-z]'),
      '',
    );
  }
}

  /// Identifies the exact collectible release itself for CONFIRMED duplicates.
  ///
  /// No single external identifier is trusted by itself.

  /// reuse UPCs or third-party catalogs may group multiple physical variants

  String get collectibleIdentityKey {
    final catalogId = normalizeIdentityText(sourceCatalogId);
    final collectibleId = normalizeIdentityText(sourceCollectibleId);
    final upc = normalizeIdentityText(sourceUpc);

  /// Broad identity key used to group POSSIBLE duplicates for review.
  ///
  /// This intentionally ignores release-defining differences such as Chase,
  /// stickers, finishes, signatures, box versions, and production codes.
  ///
  /// Examples:
  ///
  /// Same UPC but Common vs Chase:
  ///   grouped for review, not confirmed as duplicates.
  ///
  /// Same number and series with slightly different names:
  ///   grouped for review.
  ///
  /// Same number in different Funko lines:
  ///   not grouped together.
  String get similarityReviewKey {
    final upc = normalizeIdentityText(sourceUpc);

    if (upc.isNotEmpty) {
      return 'upc_review|$upc';
    }

    final collectibleId = normalizeIdentityText(sourceCollectibleId);

    if (collectibleId.isNotEmpty) {
      return 'collectible_review|$collectibleId';
    }

    final catalogId = normalizeIdentityText(sourceCatalogId);

    if (catalogId.isNotEmpty) {
      return 'catalog_review|$catalogId';
    }

    final normalizedNumber = normalizeIdentityText(number);
    final normalizedCategory = normalizeIdentityText(category);
    final normalizedSeries = normalizeIdentityText(series);
    final normalizedName = normalizeIdentityText(name);

    if (normalizedNumber.isEmpty) {
      return '';
    }

    // Preferred manual review identity:
    // Number + Funko line/category + series.
    //
    // Name is intentionally omitted here so entries such as:
    // "Spider-Man" and "Spider Man"
    // can still be reviewed together.
    if (normalizedCategory.isNotEmpty && normalizedSeries.isNotEmpty) {
      return <String>[
        'line_review',
        normalizedNumber,
        normalizedCategory,
        normalizedSeries,
      ].join('|');
    }

    // When series is missing, retain the name to avoid grouping every #01
    // in a broad category together.
    if (normalizedCategory.isNotEmpty && normalizedName.isNotEmpty) {
      return <String>[
        'category_review',
        normalizedNumber,
        normalizedCategory,
        normalizedName,
      ].join('|');
    }

    // Last-resort review key. This is deliberately conservative.
    if (normalizedSeries.isNotEmpty && normalizedName.isNotEmpty) {
      return <String>[
        'series_review',
        normalizedNumber,
        normalizedSeries,
        normalizedName,
      ].join('|');
    }

    return '';

  /// Builds the strongest available core identity anchor.
  ///
  /// The anchor identifies the probable catalog object. Physical release
  /// attributes are appended separately through [_releaseIdentityParts].
  String _buildIdentityAnchor({
    required String catalogId,
    required String collectibleId,
    required String upc,
    required String normalizedName,
    required String normalizedNumber,
    required String normalizedCategory,
    required String normalizedSeries,
  }) {
    if (collectibleId.isNotEmpty) {
      return 'collectible|$collectibleId';
    }

    if (catalogId.isNotEmpty) {
      return 'catalog|$catalogId';
    }

    if (upc.isNotEmpty) {
      return 'upc|$upc';

      return '';

    if (normalizedCategory.isEmpty && normalizedSeries.isEmpty) {
      return '';
    }

    return <String>[
      'manual',
      normalizedName,
      normalizedNumber,
      normalizedCategory,
      normalizedSeries,
    ].join('|');

  static double _readDouble(dynamic rawValue) {
    if (rawValue is num) {
      return rawValue.toDouble();
    }

    return double.tryParse(
      rawValue?.toString().trim() ?? '',
    ) ??
        0;
  }

  static int _readInt(
      dynamic rawValue, {
        required int fallback,
      }) {
    if (rawValue is int) {
      return rawValue;
    }

    if (rawValue is num) {
      return rawValue.toInt();
    }

    return int.tryParse(
      rawValue?.toString().trim() ?? '',
    ) ??
        fallback;
  }

  static List<ValueHistoryEntry> _readValueHistory(
      dynamic rawHistory, {
        required double currentValue,
        required String dateAdded,
        required String lastUpdated,
      }) {
    final entries = <ValueHistoryEntry>[];

    if (rawHistory is List) {
      for (final rawEntry in rawHistory) {
        if (rawEntry is! Map) {
          continue;
        }

        final entry = ValueHistoryEntry.fromJson(
          Map<String, dynamic>.from(rawEntry),
        );

        if (entry.value <= 0) {
          continue;
        }

        entries.add(entry);
      }
    }

    if (entries.isNotEmpty) {
      entries.sort(
            (a, b) => a.recordedAt.compareTo(b.recordedAt),
      );

      return List<ValueHistoryEntry>.unmodifiable(entries);
    }

    if (currentValue <= 0) {
      return const [];
    }

    final recordedAt =
        DateTime.tryParse(lastUpdated) ??
            DateTime.tryParse(dateAdded) ??
            DateTime.fromMillisecondsSinceEpoch(0);

    return List<ValueHistoryEntry>.unmodifiable(
      <ValueHistoryEntry>[
        ValueHistoryEntry(
          value: currentValue,
          recordedAt: recordedAt,
          source: ValueHistoryEntry.legacyValueSource,
        ),
      ],
    );
  }

  static List<AtlasTimelineEvent> _readTimeline(
      dynamic rawTimeline, {
        required String popId,
        required String dateAdded,
      }) {
    final events = <AtlasTimelineEvent>[];

    if (rawTimeline is List) {
      for (final rawEvent in rawTimeline) {
        if (rawEvent is! Map) {
          continue;
        }

        final event = AtlasTimelineEvent.fromJson(
          Map<String, dynamic>.from(rawEvent),
        );

        events.add(event);
      }
    }

    if (events.isNotEmpty) {
      events.sort(
            (a, b) => a.occurredAt.compareTo(b.occurredAt),
      );

      return List<AtlasTimelineEvent>.unmodifiable(events);
    }

    final addedAt = DateTime.tryParse(dateAdded);

    if (popId.isEmpty || addedAt == null) {
      return const [];
    }

    return List<AtlasTimelineEvent>.unmodifiable(
      <AtlasTimelineEvent>[
        AtlasTimelineEvent(
          id: 'legacy-added-$popId',
          popId: popId,
          type: AtlasTimelineEventType.addedToVault,
          occurredAt: addedAt,
          title: 'Added to Vault',
          createdBy: AtlasTimelineEvent.systemCreatedBy,
          createdAt: addedAt,
        ),
      ],
    );
  }

  // Cleans strings by lowercasing and stripping non-alphanumeric characters.
  // Converts "Spider-Man #01!" -> "spiderman01"
  static String normalizeIdentityText(String value) {
    return value
        .trim()
        .toLowerCase()
        .replaceAll(RegExp(r'[^a-z0-9]'), '');
  }

  static String normalizeVariantIdentity(String value) {
    final cleaned = normalizeIdentityText(value);

    if (cleaned.isEmpty ||
        cleaned == 'none' ||
        cleaned == 'no' ||
        cleaned == 'na' ||
        cleaned == 'common' ||
        cleaned == 'standard') {
      return 'common';
    }

    return cleaned;
  }

  static String normalizeOptionalIdentity(String value) {
    final cleaned = normalizeIdentityText(value);

    if (cleaned.isEmpty ||
        cleaned == 'none' ||
        cleaned == 'no' ||
        cleaned == 'na') {
      return 'none';
    }

    return cleaned;
  }

  static String normalizeSignatureIdentity(String value) {
    final cleaned = normalizeIdentityText(value);

    if (cleaned.isEmpty ||
        cleaned == 'none' ||
        cleaned == 'no' ||
        cleaned == 'na' ||
        cleaned == 'unsigned') {
      return 'unsigned';
    }

    return cleaned;
  }

  static bool hasMeaningfulIdentityLabel(String value) {
    final cleaned = normalizeIdentityText(value);

    return cleaned.isNotEmpty &&
        cleaned != 'none' &&
        cleaned != 'no' &&
        cleaned != 'na' &&
        cleaned != 'unsigned' &&
        cleaned != 'common' &&
        cleaned != 'standard';
  }

    return List<String>.unmodifiable(
      _collections.entries
          .where(
            (entry) => entry.value.any(
              (storedPop) =>
                  storedPop.id.trim() == normalizedId &&
                  storedPop.collectionRecordKey == recordKey,
            ),
          )
          .map((entry) => entry.key),
    );
  }

  List<Pop> get selectedCollectionPops =>
      _collections[_selectedCollection] ?? [];
  List<Pop> get selectedCollectionWishlistPops {
    final List<Pop> collectionPops =
        _collections[_selectedCollection] ?? const <Pop>[];

    final Set<String> overlayIdentityKeys =
        _ownedWishlistIdentityKeysByCollection[_selectedCollection] ??
        const <String>{};

    return _buildWishlistPops(
      collectionPops: collectionPops,
      overlayIdentityKeys: overlayIdentityKeys,
    );
  }

  List<Pop> get wishlistPops {
    final List<Pop> results = <Pop>[];

    _collections.forEach((String collectionName, List<Pop> collectionPops) {
      final Set<String> overlayIdentityKeys =
          _ownedWishlistIdentityKeysByCollection[collectionName] ??
          const <String>{};

      results.addAll(
        _buildWishlistPops(
          collectionPops: collectionPops,
          overlayIdentityKeys: overlayIdentityKeys,
        ),
      );
    });

    return List<Pop>.unmodifiable(results);
  }

  List<Pop> _buildWishlistPops({
    required List<Pop> collectionPops,
    required Set<String> overlayIdentityKeys,
  }) {
    final List<Pop> results = <Pop>[];
    final Set<String> addedRecordIds = <String>{};
    final Set<String> wishlistRecordIdentityKeys = <String>{};

    for (final Pop pop in collectionPops) {
      if (!pop.isWishlist) {
        continue;
      }

      results.add(pop);

      addedRecordIds.add(pop.id);

      final String identityKey = pop.collectibleIdentityKey.trim();

      if (identityKey.isNotEmpty) {
        wishlistRecordIdentityKeys.add(identityKey);
      }
    }

    for (final Pop pop in collectionPops) {
      if (pop.isWishlist || addedRecordIds.contains(pop.id)) {
        continue;
      }

      final String identityKey = pop.collectibleIdentityKey.trim();

      if (identityKey.isEmpty || !overlayIdentityKeys.contains(identityKey)) {
        continue;
      }

      // Prevent an owned overlay and an old Wishlist-only
      // record from showing the same release twice.
      if (wishlistRecordIdentityKeys.contains(identityKey)) {
        continue;
      }

      results.add(pop);

      addedRecordIds.add(pop.id);
    }

    return List<Pop>.unmodifiable(results);
  }

  Future<void> loadData() async {
    final data = HiveService.loadData('vault_data');

    if (data != null) {
      _selectedCollection =
          data['selectedCollection'] ?? 'Entire Family Collection';

      _collections.clear();
      _activities.clear();

      final activitiesData = data['activities'];
      final profileData = data['collectorProfile'];
      final ownedWishlistData = data['ownedWishlistIdentityKeys'];

      if (profileData != null) {
        _collectorProfile = CollectorProfile.fromJson(
          Map<String, dynamic>.from(profileData),
        );
      }

      if (activitiesData != null) {
        _activities.addAll(
          (activitiesData as List)
              .map((item) => Activity.fromJson(Map<String, dynamic>.from(item)))
              .toList(),
        );
      }

      final collectionsData = data['collections'] as Map<String, dynamic>;

      collectionsData.forEach((key, value) {
        _collections[key] = (value as List)
            .map((item) => Pop.fromJson(Map<String, dynamic>.from(item)))
            .toList();
      });
      _ownedWishlistIdentityKeysByCollection.clear();

      if (ownedWishlistData is Map) {
        ownedWishlistData.forEach((dynamic collectionName, dynamic rawKeys) {
          if (collectionName is! String || rawKeys is! List) {
            return;
          }

          _ownedWishlistIdentityKeysByCollection[collectionName] = rawKeys
              .map((dynamic value) => value.toString().trim())
              .where((String value) => value.isNotEmpty)
              .toSet();
        });
      }

      for (final String collectionName in _collections.keys) {
        _ownedWishlistIdentityKeysByCollection.putIfAbsent(
          collectionName,
          () => <String>{},
        );
      }
    }

    notifyListeners();
  }

  Future<void> _saveData({
    String? selectedCollection,
    Map<String, List<Pop>>? collections,
    Map<String, Set<String>>? ownedWishlistIdentityKeysByCollection,
  }) async {
    final collectionsToSave = collections ?? _collections;
    final wishlistKeysToSave =
        ownedWishlistIdentityKeysByCollection ??
        _ownedWishlistIdentityKeysByCollection;
    final collectionsJson = <String, dynamic>{};

    collectionsToSave.forEach((key, value) {
      collectionsJson[key] = value.map((pop) => pop.toJson()).toList();
    });
    final Map<String, dynamic> ownedWishlistIdentityJson = <String, dynamic>{};

    wishlistKeysToSave.forEach((
      String collectionName,
      Set<String> identityKeys,
    ) {
      ownedWishlistIdentityJson[collectionName] = identityKeys.toList(
        growable: false,
      );
    });
    await HiveService.saveData('vault_data', {
      'selectedCollection': selectedCollection ?? _selectedCollection,
      'collections': collectionsJson,
      'activities': _activities.map((activity) => activity.toJson()).toList(),
      'collectorProfile': _collectorProfile.toJson(),
      'ownedWishlistIdentityKeys': ownedWishlistIdentityJson,
    });
  }

  Future<CollectionOperationResult> setSelectedCollection(String value) async {
    final matchingName = _matchingCollectionName(value);

    if (matchingName == null) {
      return CollectionOperationResult.notFound;
    }

    if (_selectedCollection == matchingName) {
      return CollectionOperationResult.success;
    }

    try {
      await _saveData(selectedCollection: matchingName);
    } catch (_) {
      return CollectionOperationResult.persistenceFailure;
    }

    _selectedCollection = matchingName;
    notifyListeners();

    return CollectionOperationResult.success;
  }

  Future<CollectionOperationResult> addCollection(String name) async {
    final trimmedName = name.trim();

    if (trimmedName.isEmpty) {
      return CollectionOperationResult.invalidName;
    }

    if (_matchingCollectionName(trimmedName) != null) {
      return CollectionOperationResult.duplicateName;
    }

    final updatedCollections = Map<String, List<Pop>>.from(_collections)
      ..[trimmedName] = <Pop>[];
    final updatedWishlistKeys = _copyWishlistKeys()..[trimmedName] = <String>{};

    try {
      await _saveData(
        collections: updatedCollections,
        ownedWishlistIdentityKeysByCollection: updatedWishlistKeys,
      );
    } catch (_) {
      return CollectionOperationResult.persistenceFailure;
    }

    _replaceCollections(updatedCollections);
    _replaceWishlistKeys(updatedWishlistKeys);
    notifyListeners();

    return CollectionOperationResult.success;
  }

  Future<CollectionOperationResult> renameCollection(
    String currentName,
    String newName,
  ) async {
    final existingName = _matchingCollectionName(currentName);
    final trimmedNewName = newName.trim();

    if (existingName == null) {
      return CollectionOperationResult.notFound;
    }

    if (trimmedNewName.isEmpty) {
      return CollectionOperationResult.invalidName;
    }

    final duplicateName = _matchingCollectionName(trimmedNewName);

    if (duplicateName != null && duplicateName != existingName) {
      return CollectionOperationResult.duplicateName;
    }

    if (trimmedNewName == existingName) {
      return CollectionOperationResult.success;
    }

    final updatedCollections = <String, List<Pop>>{};

    for (final entry in _collections.entries) {
      updatedCollections[entry.key == existingName
              ? trimmedNewName
              : entry.key] =
          entry.value;
    }

    final updatedWishlistKeys = <String, Set<String>>{};

    for (final entry in _ownedWishlistIdentityKeysByCollection.entries) {
      updatedWishlistKeys[entry.key == existingName
          ? trimmedNewName
          : entry.key] = Set<String>.from(
        entry.value,
      );
    }

    updatedWishlistKeys.putIfAbsent(trimmedNewName, () => <String>{});

    final updatedSelection = _selectedCollection == existingName
        ? trimmedNewName
        : _selectedCollection;

    try {
      await _saveData(
        selectedCollection: updatedSelection,
        collections: updatedCollections,
        ownedWishlistIdentityKeysByCollection: updatedWishlistKeys,
      );
    } catch (_) {
      return CollectionOperationResult.persistenceFailure;
    }

    _replaceCollections(updatedCollections);
    _replaceWishlistKeys(updatedWishlistKeys);
    _selectedCollection = updatedSelection;
    notifyListeners();

    return CollectionOperationResult.success;
  }

  Future<CollectionOperationResult> deleteCollection(String name) async {
    final matchingName = _matchingCollectionName(name);

    if (matchingName == null) {
      return CollectionOperationResult.notFound;
    }

    if (_collections.length <= 1) {
      return CollectionOperationResult.lastCollection;
    }

    final updatedCollections = Map<String, List<Pop>>.from(_collections)
      ..remove(matchingName);
    final updatedWishlistKeys = _copyWishlistKeys()..remove(matchingName);
    final updatedSelection = _selectedCollection == matchingName
        ? updatedCollections.keys.first
        : _selectedCollection;

    try {
      await _saveData(
        selectedCollection: updatedSelection,
        collections: updatedCollections,
        ownedWishlistIdentityKeysByCollection: updatedWishlistKeys,
      );
    } catch (_) {
      return CollectionOperationResult.persistenceFailure;
    }

    _replaceCollections(updatedCollections);
    _replaceWishlistKeys(updatedWishlistKeys);
    _selectedCollection = updatedSelection;
    notifyListeners();

    return CollectionOperationResult.success;
  }

  String? _matchingCollectionName(String value) {
    final normalizedName = value.trim().toLowerCase();

    if (normalizedName.isEmpty) {
      return null;
    }

    for (final collectionName in _collections.keys) {
      if (collectionName.trim().toLowerCase() == normalizedName) {
        return collectionName;
      }
    }

    return null;
  }

  Map<String, Set<String>> _copyWishlistKeys() {
    return _ownedWishlistIdentityKeysByCollection.map(
      (collectionName, identityKeys) =>
          MapEntry(collectionName, Set<String>.from(identityKeys)),
    );
  }

  void _replaceCollections(Map<String, List<Pop>> collections) {
    _collections
      ..clear()
      ..addAll(collections);
  }

  void _replaceWishlistKeys(Map<String, Set<String>> wishlistKeys) {
    _ownedWishlistIdentityKeysByCollection
      ..clear()
      ..addAll(wishlistKeys);
  }

  bool isOwnedPopWishlisted(Pop pop, {String? collectionName}) {
    final String identityKey = pop.collectibleIdentityKey.trim();

    if (identityKey.isEmpty) {
      return false;
    }

    final String? targetCollectionName = collectionName == null
        ? _selectedCollection
        : _matchingCollectionName(collectionName);

    if (targetCollectionName == null) {
      return false;
    }

    final Set<String> identityKeys =
        _ownedWishlistIdentityKeysByCollection[targetCollectionName] ??
        const <String>{};

    return identityKeys.contains(identityKey);
  }

  bool isCollectibleWishlisted(Pop pop, {String? collectionName}) {
    final String identityKey = pop.collectibleIdentityKey.trim();

    if (identityKey.isEmpty) {
      return false;
    }

    final String? targetCollectionName = collectionName == null
        ? _selectedCollection
        : _matchingCollectionName(collectionName);

    if (targetCollectionName == null) {
      return false;
    }

    if (isOwnedPopWishlisted(pop, collectionName: targetCollectionName)) {
      return true;
    }

    final List<Pop> pops = _collections[targetCollectionName] ?? const <Pop>[];

    return pops.any((Pop existingPop) {
      return existingPop.isWishlist &&
          existingPop.collectibleIdentityKey == identityKey;
    });
  }

  Future<bool> addPop(Pop pop, {String? collectionName}) async {
    final targetCollectionName = collectionName == null
        ? _selectedCollection
        : _matchingCollectionName(collectionName);
    final pops = targetCollectionName == null
        ? null
        : _collections[targetCollectionName];

    if (pops == null) {
      return false;
    }

    final now = DateTime.now().toIso8601String();

    final newRecordKey = pop.collectionRecordKey;

    final existingIndex = newRecordKey.isEmpty
        ? -1
        : pops.indexWhere(
            (existingPop) => existingPop.collectionRecordKey == newRecordKey,
          );

    if (existingIndex != -1) {
      final existingPop = pops[existingIndex];

      pops[existingIndex] = existingPop.copyWith(
        quantity: existingPop.quantity + pop.quantity,
        lastUpdated: now,
      );

      await addActivity(
        Activity(
          id: DateTime.now().millisecondsSinceEpoch.toString(),
          popId: existingPop.id,
          action: 'quantity_increased',
          popName: pop.name,
          popNumber: pop.number,
          owner: pop.owner,
          actor: pop.owner,
          details: '${pop.name} quantity increased',
          timestamp: DateTime.now(),
        ),
      );
    } else {
      final addedAt =
          DateTime.tryParse(
            pop.dateAdded.trim().isEmpty ? now : pop.dateAdded,
          ) ??
          DateTime.now();

      final existingTimeline = List<AtlasTimelineEvent>.from(pop.timeline);
      final hasAddedEvent = existingTimeline.any(
        (event) => event.type == AtlasTimelineEventType.addedToVault,
      );

      if (!hasAddedEvent) {
        existingTimeline.add(
          AtlasTimelineEvent(
            id: 'added-${pop.id}-${addedAt.microsecondsSinceEpoch}',
            popId: pop.id,
            type: AtlasTimelineEventType.addedToVault,
            occurredAt: addedAt,
            title: 'Added to Vault',
            createdBy: AtlasTimelineEvent.systemCreatedBy,
            createdAt: addedAt,
          ),
        );
      }

      existingTimeline.sort((a, b) => a.occurredAt.compareTo(b.occurredAt));

      final datedPop = pop.copyWith(
        dateAdded: addedAt.toIso8601String(),
        lastUpdated: pop.lastUpdated.trim().isEmpty ? now : pop.lastUpdated,
        timeline: List<AtlasTimelineEvent>.unmodifiable(existingTimeline),
      );

      pops.add(datedPop);

      await addActivity(
        Activity(
          id: DateTime.now().millisecondsSinceEpoch.toString(),
          popId: datedPop.id,
          action: 'pop_added',
          popName: datedPop.name,
          popNumber: datedPop.number,
          owner: datedPop.owner,
          actor: datedPop.owner,
          details: '${datedPop.name} added',
          timestamp: DateTime.now(),
        ),
      );
    }

    await _saveData();

    notifyListeners();

    return true;
  }

  Future<PopImportResult> importPopsSafely(
    List<Pop> importedPops, {
    String? collectionName,
  }) async {
    final String targetCollection = collectionName ?? _selectedCollection;

    final String? matchingCollection = _matchingCollectionName(
      targetCollection,
    );

    if (matchingCollection == null) {
      return PopImportResult(
        attempted: importedPops.length,
        added: 0,
        skipped: importedPops.length,
      );
    }

    final List<Pop> currentPops = _collections[matchingCollection]!;

    final indexesByRecordKey = <String, int>{};

    for (var index = 0; index < currentPops.length; index++) {
      final key = currentPops[index].collectionRecordKey;

      if (key.isNotEmpty) {
        indexesByRecordKey[key] = index;
      }
    }

    final now = DateTime.now().toIso8601String();

    var added = 0;
    var consolidated = 0;

    for (final importedPop in importedPops) {
      final key = importedPop.collectionRecordKey;
      final existingIndex = key.isEmpty ? null : indexesByRecordKey[key];

      if (existingIndex != null) {
        final existingPop = currentPops[existingIndex];

        currentPops[existingIndex] = existingPop.copyWith(
          quantity: existingPop.quantity + importedPop.quantity,
          value: existingPop.value > 0 ? existingPop.value : importedPop.value,
          pricePaid: existingPop.pricePaid > 0
              ? existingPop.pricePaid
              : importedPop.pricePaid,
          imageUrl: existingPop.imageUrl.trim().isNotEmpty
              ? existingPop.imageUrl
              : importedPop.imageUrl,
          lastUpdated: now,
        );

        consolidated++;
        continue;
      }

      final datedPop = importedPop.copyWith(
        dateAdded: importedPop.dateAdded.trim().isEmpty
            ? now
            : importedPop.dateAdded,
        lastUpdated: importedPop.lastUpdated.trim().isEmpty
            ? now
            : importedPop.lastUpdated,
      );

      currentPops.add(datedPop);

      if (key.isNotEmpty) {
        indexesByRecordKey[key] = currentPops.length - 1;
      }

      added++;
    }

    await addActivity(
      Activity(
        id: DateTime.now().millisecondsSinceEpoch.toString(),
        action: 'csv_import',
        popName: 'CSV Import',
        popNumber: '',
        owner: matchingCollection,
        actor: _collectorProfile.displayName,
        details:
            'CSV import complete: $added records added, '
            '$consolidated matching copies consolidated',
        timestamp: DateTime.now(),
      ),
    );

    await _saveData();

    notifyListeners();

    return PopImportResult(
      attempted: importedPops.length,
      added: added,
      skipped: consolidated,
    );
  }

  Future<void> updateOwnerForPops(
      List<String> popIds,
      String owner, {
        String? collectionName,
      }) async {
    final String targetCollection =
        collectionName ?? _selectedCollection;

    final String? matchingCollection =
    _matchingCollectionName(targetCollection);

    if (matchingCollection == null) {
      return;
    }

    final pops = _collections[matchingCollection];

    if (pops == null) {
      return;
    }

    final now = DateTime.now().toIso8601String();

    for (var i = 0; i < pops.length; i++) {
      if (popIds.contains(pops[i].id)) {
        final oldPop = pops[i];

        pops[i] = oldPop.copyWith(
          owner: owner,
          lastUpdated: now,
        );

        await addActivity(
          Activity(
            id: DateTime.now()
                .millisecondsSinceEpoch
                .toString(),
            popId: oldPop.id,
            action: 'owner_changed',
            popName: oldPop.name,
            popNumber: oldPop.number,
            owner: owner,
            actor: owner,
            details: '${oldPop.name} assigned to $owner',
            timestamp: DateTime.now(),
          ),
        );
      }
    }

    await _saveData();

    notifyListeners();
  }

  Future<Pop?> updatePop(
    Pop updatedPop, {
    String? collectionName,
    String? existingRecordKey,
  }) async {
    final location = _resolveStoredPop(
      popId: updatedPop.id,
      collectionName: collectionName,
      recordKey: existingRecordKey,
    );

    if (location == null) {
      return null;
    }

    final existingPop = location.pops[location.index];
    final now = DateTime.now();
    final nowIso = now.toIso8601String();

    const timelineEngine = AtlasTimelineEngine();

    final detectedEvents = timelineEngine.comparePopChanges(
      previousPop: existingPop,
      updatedPop: updatedPop,
      occurredAt: now,
      source: 'Manual',
    );

    final timeline = <AtlasTimelineEvent>[
      ...existingPop.timeline,
      ...detectedEvents,
    ];

    final sortedTimeline = timelineEngine.chronological(timeline);

    final valueHistory = _updatedValueHistory(
      existingPop: existingPop,
      updatedPop: updatedPop,
      occurredAt: now,
    );

    final persistedPop = updatedPop.copyWith(
      dateAdded: existingPop.dateAdded,
      lastUpdated: nowIso,
      valueHistory: valueHistory,
      timeline: sortedTimeline,
    );

    location.pops[location.index] = persistedPop;

    await _saveData();

    notifyListeners();

    return persistedPop;
  }

  _StoredPopLocation? _resolveStoredPop({
    required String popId,
    String? collectionName,
    String? recordKey,
  }) {
    final normalizedId = popId.trim();

    if (normalizedId.isEmpty) {
      return null;
    }

    final Iterable<MapEntry<String, List<Pop>>> collectionsToSearch;

    if (collectionName != null) {
      final matchingName = _matchingCollectionName(collectionName);

      if (matchingName == null) {
        return null;
      }

      collectionsToSearch = <MapEntry<String, List<Pop>>>[
        MapEntry(matchingName, _collections[matchingName]!),
      ];
    } else {
      collectionsToSearch = _collections.entries;
    }

    final matches = <_StoredPopLocation>[];

    for (final collection in collectionsToSearch) {
      for (var index = 0; index < collection.value.length; index++) {
        final storedPop = collection.value[index];

        if (storedPop.id.trim() != normalizedId) {
          continue;
        }

        if (recordKey != null && storedPop.collectionRecordKey != recordKey) {
          continue;
        }

        matches.add((
          collectionName: collection.key,
          pops: collection.value,
          index: index,
        ));

        if (matches.length > 1) {
          return null;
        }
      }
    }

    return matches.singleOrNull;
  }

  List<ValueHistoryEntry> _updatedValueHistory({
    required Pop existingPop,
    required Pop updatedPop,
    required DateTime occurredAt,
  }) {
    if (existingPop.value == updatedPop.value) {
      return existingPop.valueHistory;
    }

    final entries = <ValueHistoryEntry>[...existingPop.valueHistory];

    if (entries.isEmpty && existingPop.value > 0) {
      entries.add(
        ValueHistoryEntry(
          value: existingPop.value,
          recordedAt:
              DateTime.tryParse(existingPop.lastUpdated) ??
              DateTime.tryParse(existingPop.dateAdded) ??
              occurredAt,
          source: ValueHistoryEntry.legacyValueSource,
        ),
      );
    }

    if (updatedPop.value > 0) {
      entries.add(
        ValueHistoryEntry(
          value: updatedPop.value,
          recordedAt: occurredAt,
          source: ValueHistoryEntry.manualSource,
        ),
      );
    }

  Future<bool> consolidateDuplicateGroup(
    List<String> popIds, {
    String? collectionName,
  }) async {
    if (popIds.length < 2) {
      return false;
    }

    List<Pop>? matchingCollection;
    List<Pop>? matchingPops;

    if (collectionName != null) {
      final String? matchingName = _matchingCollectionName(collectionName);

      if (matchingName == null) {
        return false;
      }

      final List<Pop> collection = _collections[matchingName]!;

      final List<Pop> foundPops = collection
          .where((pop) => popIds.contains(pop.id))
          .toList();

      if (foundPops.length != popIds.length) {
        return false;
      }

      matchingCollection = collection;
      matchingPops = foundPops;
    } else {
      final matchingLocations = <({List<Pop> collection, List<Pop> pops})>[];

      for (final collection in _collections.values) {
        final foundPops = collection
            .where((pop) => popIds.contains(pop.id))
            .toList();

        if (foundPops.length == popIds.length) {
          matchingLocations.add((collection: collection, pops: foundPops));
        }
      }

      if (matchingLocations.length != 1) {
        return false;
      }

      matchingCollection = matchingLocations.single.collection;

      matchingPops = matchingLocations.single.pops;
    }

    if (matchingPops.length < 2) {
      return false;
    }

    final identityKeys = matchingPops
        .map((pop) => pop.collectionRecordKey)
        .where((key) => key.isNotEmpty)
        .toSet();

    if (identityKeys.length != 1) {
      return false;
    }

    final now = DateTime.now().toIso8601String();

    final primaryPop = matchingPops.first;

    final totalQuantity = matchingPops.fold<int>(
      0,
      (total, pop) => total + pop.quantity,
    );

    final bestValue = matchingPops.fold<double>(
      0,
      (highest, pop) => pop.value > highest ? pop.value : highest,
    );

    final bestPricePaid = matchingPops.fold<double>(
      0,
      (highest, pop) => pop.pricePaid > highest ? pop.pricePaid : highest,
    );

    final bestImageUrl = matchingPops
        .map((pop) => pop.imageUrl.trim())
        .firstWhere(
          (value) => value.isNotEmpty,
          orElse: () => primaryPop.imageUrl,
        );

    final earliestDateAdded = _earliestSavedDate(
      matchingPops.map((pop) => pop.dateAdded),
    );

    final consolidatedPop = primaryPop.copyWith(
      quantity: totalQuantity,
      value: bestValue,
      pricePaid: bestPricePaid,
      imageUrl: bestImageUrl,
      dateAdded: earliestDateAdded.isEmpty
          ? primaryPop.dateAdded
          : earliestDateAdded,
      lastUpdated: now,
    );

    final primaryIndex = matchingCollection.indexWhere(
      (pop) => pop.id == primaryPop.id,
    );

    if (primaryIndex == -1) {
      return false;
    }

    matchingCollection[primaryIndex] = consolidatedPop;

    final removedIds = matchingPops.skip(1).map((pop) => pop.id).toSet();

    matchingCollection.removeWhere((pop) => removedIds.contains(pop.id));

    _activities.add(
      Activity(
        id: DateTime.now().millisecondsSinceEpoch.toString(),
        popId: consolidatedPop.id,
        action: 'duplicates_consolidated',
        popName: consolidatedPop.name,
        popNumber: consolidatedPop.number,
        owner: consolidatedPop.owner,
        actor: _collectorProfile.displayName,
        details:
            '${matchingPops.length} matching records consolidated into '
            'quantity $totalQuantity',
        timestamp: DateTime.now(),
      ),
    );

    if (_activities.length > 500) {
      _activities.removeAt(0);
    }

    await _saveData();

    notifyListeners();

    return true;
  }

  String _earliestSavedDate(Iterable<String> values) {
    DateTime? earliestDate;
    var earliestValue = '';

    for (final value in values) {
      final cleaned = value.trim();
      final parsed = DateTime.tryParse(cleaned);

      if (parsed == null) {
        continue;
      }

      if (earliestDate == null || parsed.isBefore(earliestDate)) {
        earliestDate = parsed;
        earliestValue = cleaned;
      }
    }

    return earliestValue;
  }

  Future<bool> mergeDuplicates({String? collectionName}) async {
    final String targetCollection = collectionName ?? _selectedCollection;

    final String? matchingCollection = _matchingCollectionName(
      targetCollection,
    );

    if (matchingCollection == null) {
      return false;
    }

    final List<Pop> pops = _collections[matchingCollection]!;

    final now = DateTime.now().toIso8601String();
    final merged = <String, Pop>{};

    for (final pop in pops) {
      final key = pop.collectionRecordKey;

      if (key.isEmpty) {
        merged[pop.id] = pop;
        continue;
      }

      if (merged.containsKey(key)) {
        final existing = merged[key]!;

        merged[key] = existing.copyWith(
          quantity: existing.quantity + pop.quantity,
          lastUpdated: now,
        );
      } else {
        merged[key] = pop;
      }
    }

    _collections[matchingCollection] = merged.values.toList();

    await _saveData();

    notifyListeners();

    return true;
  }
