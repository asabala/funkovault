  Future<void> _captureStep(
    AtlasGuidedInspectionStep step, {
    TextEditingController? textTarget,
  }) async {
    final source = await showModalBottomSheet<AtlasInspectionImageSource>(
      context: context,
      builder: (context) => SafeArea(
        child: Wrap(
          children: [
            ListTile(
              leading: const Icon(Icons.photo_camera_outlined),
              title: const Text('Camera'),
              onTap: () =>
                  Navigator.pop(context, AtlasInspectionImageSource.camera),
            ),
            ListTile(
              leading: const Icon(Icons.photo_library_outlined),
              title: const Text('Gallery'),
              onTap: () =>
                  Navigator.pop(context, AtlasInspectionImageSource.gallery),
            ),
          ],
        ),
      ),
    );

    if (source == null) return;

    setState(() => _busy = true);
    try {
      final ScanImage? image = await _capture.capture(
        source: source,
        type: step.imageType,
      );
      if (image == null) return;

      String recognizedText = '';
      if (step.supportsTextRecognition &&
          textTarget != null &&
          image.hasLocalPath) {
        recognizedText = await _recognizer.recognize(image.localPath!);
      }

      if (!mounted) return;
      setState(() {
        _images.removeWhere((ScanImage item) => item.type == step.imageType);
        _images.add(image);
        if (recognizedText.isNotEmpty && textTarget != null) {
          textTarget.text = recognizedText;
          if (step == AtlasGuidedInspectionStep.coa) {
            final interpretation =
                AtlasAuthenticationEvidenceInterpreter.interpret(
                  provider: _provider.text,
                  certificateNumber: _number.text,
                  certificationText: recognizedText,
                  hasCoa: _coa,
                );
            if (_provider.text.trim().isEmpty &&
                interpretation.detectedProviderDisplayName.isNotEmpty) {
              _provider.text = interpretation.detectedProviderDisplayName;
            }
            if (_number.text.trim().isEmpty &&
                interpretation.detectedCertificateNumber.isNotEmpty) {
              _number.text = interpretation.detectedCertificateNumber;
            }
          }
        }
      });
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

import 'atlas_authenticity_evidence.dart';
import 'atlas_authenticity_signal.dart';
import 'atlas_production_stamp.dart';

enum AtlasAuthenticityStatus {
  strongEvidence,
  developingEvidence,
  reviewRecommended,
  insufficientEvidence,
}

enum AtlasAuthenticityConcernLevel {
  noObviousConcerns,
  reviewRecommended,
  significantInconsistencies,
  strongCounterfeitIndicators,
  insufficientData,
}

final class AtlasAuthenticityAssessment {
  AtlasAuthenticityAssessment({
    required this.status,
    required this.concernLevel,
    required double evidenceConfidence,
    required double referenceCoverage,
    required double inspectionCoverage,
    required double aggregatedConcernScore,
    required this.summary,
    required List<AtlasAuthenticitySignal> signals,
    List<AtlasEvaluatedAuthenticitySignal> evaluatedSignals =
        const <AtlasEvaluatedAuthenticitySignal>[],
    List<AtlasInspectionArea> uninspectedAreas = const <AtlasInspectionArea>[],
    List<String> auditLog = const <String>[],
    this.productionStamp,
  }) : evidenceConfidence = _unitScore(
         evidenceConfidence,
         'evidenceConfidence',
       ),
       referenceCoverage = _unitScore(referenceCoverage, 'referenceCoverage'),
       inspectionCoverage = _unitScore(
         inspectionCoverage,
         'inspectionCoverage',
       ),
       aggregatedConcernScore = _unitScore(
         aggregatedConcernScore,
         'aggregatedConcernScore',
       ),
       signals = List<AtlasAuthenticitySignal>.unmodifiable(signals),
       evaluatedSignals = List<AtlasEvaluatedAuthenticitySignal>.unmodifiable(
         evaluatedSignals,
       ),
       uninspectedAreas = List<AtlasInspectionArea>.unmodifiable(
         uninspectedAreas,
       ),
       auditLog = List<String>.unmodifiable(auditLog);

  final AtlasAuthenticityStatus status;
  final AtlasAuthenticityConcernLevel concernLevel;

  /// How clearly Atlas could observe the supplied evidence.
  final double evidenceConfidence;

  /// How complete/trustworthy the product reference criteria are.
  final double referenceCoverage;

  /// How much of the guided inspection surface was actually inspected.
  final double inspectionCoverage;

  /// Normalized concern score after category caps and correlation clamping.
  final double aggregatedConcernScore;

  final String summary;
  final List<AtlasAuthenticitySignal> signals;
  final List<AtlasEvaluatedAuthenticitySignal> evaluatedSignals;
  final List<AtlasInspectionArea> uninspectedAreas;
  final List<String> auditLog;
  final AtlasProductionStamp? productionStamp;

  // Backward-compatible accessors used by the existing engine/UI.
  int get confidenceScore =>
      (evidenceConfidence * 100).round().clamp(0, 100).toInt();
  int get evidenceCoverage =>
      (inspectionCoverage * 100).round().clamp(0, 100).toInt();

  bool get requiresReview =>
      concernLevel == AtlasAuthenticityConcernLevel.reviewRecommended ||
      concernLevel ==
          AtlasAuthenticityConcernLevel.significantInconsistencies ||
      concernLevel ==
          AtlasAuthenticityConcernLevel.strongCounterfeitIndicators ||
      concernLevel == AtlasAuthenticityConcernLevel.insufficientData;

  bool get hasWarnings => signals.any((AtlasAuthenticitySignal signal) {
    return signal.isWarning;
  });

  bool get hasEnoughReferenceData => referenceCoverage >= 0.50;

  static double _unitScore(double value, String name) {
    if (!value.isFinite || value < 0.0 || value > 1.0) {
      throw ArgumentError.value(
        value,
        name,
        '$name must be between 0.0 and 1.0.',
      );
    }
    return value;
  }
}

final class AtlasVisualEvidenceObservation {
  static const int currentSchemaVersion = 1;

  AtlasVisualEvidenceObservation({
    required String observationId,
    required String sourceEngineId,
    required this.area,
    required this.category,
    required String label,
    required String observedValue,
    required double confidence,
    String? scanImageId,
    this.cropRegion,
    Map<String, Object?> attributes = const <String, Object?>{},
    DateTime? observedAt,
  }) : observationId = _requireText(observationId, 'observationId'),
       sourceEngineId = _requireText(sourceEngineId, 'sourceEngineId'),
       label = _requireText(label, 'label'),
       observedValue = observedValue.trim(),
       confidence = _normalizeConfidence(confidence),
       scanImageId = _optionalText(scanImageId),
       attributes = _deepFreezeMap(attributes),
       observedAt = (observedAt ?? DateTime.now()).toUtc();

  final String observationId;
  final String sourceEngineId;
  final AtlasInspectionArea area;
  final AtlasEvidenceCategory category;
  final String label;
  final String observedValue;
  final double confidence;
  final String? scanImageId;
  final NormalizedImageRegion? cropRegion;
  final Map<String, Object?> attributes;
  final DateTime observedAt;

  Map<String, Object?> toJson() => <String, Object?>{
    'schemaVersion': currentSchemaVersion,
    'observationId': observationId,
    'sourceEngineId': sourceEngineId,
    'area': area.name,
    'category': category.name,
    'label': label,
    'observedValue': observedValue,
    'confidence': confidence,
    'scanImageId': scanImageId,
    'cropRegion': cropRegion?.toJson(),
    'attributes': attributes,
    'observedAt': observedAt.toIso8601String(),
  };

  static AtlasVisualEvidenceObservation? fromJson(Map<String, Object?> json) {
    try {
      if (json['schemaVersion'] != currentSchemaVersion) {
        return null;
      }

      final AtlasInspectionArea? area = _parseArea(json['area']);
      final AtlasEvidenceCategory? category = _parseCategory(json['category']);
      if (area == null || category == null) {
        return null;
      }

      NormalizedImageRegion? crop;
      final Object? rawCrop = json['cropRegion'];
      if (rawCrop != null) {
        if (rawCrop is! Map) {
          return null;
        }
        crop = NormalizedImageRegion.fromJson(
          Map<String, Object?>.from(rawCrop),
        );
        if (crop == null) {
          return null;
        }
      }

      final Object? rawAttributes = json['attributes'];
      final Map<String, Object?> attributes = rawAttributes is Map
          ? Map<String, Object?>.from(rawAttributes)

Map<String, Object?> _deepFreezeMap(Map<String, Object?> values) {
  final Map<String, Object?> copy = <String, Object?>{};
  for (final MapEntry<String, Object?> entry in values.entries) {
    copy[entry.key] = _deepFreeze(entry.value);
  }
  return Map<String, Object?>.unmodifiable(copy);
}

Object? _deepFreeze(Object? value) {
  if (value is Map) {
    final Map<String, Object?> mapped = <String, Object?>{};
    for (final MapEntry<Object?, Object?> entry in value.entries) {
      mapped[entry.key.toString()] = _deepFreeze(entry.value);
    }
    return Map<String, Object?>.unmodifiable(mapped);
  }
  if (value is List) {
    return List<Object?>.unmodifiable(value.map<Object?>(_deepFreeze));
  }
  if (value is Set) {
    return Set<Object?>.unmodifiable(value.map<Object?>(_deepFreeze));
  }
  return value;
}

import '../contributions/contribution_conflict_policy.dart';
import '../contributions/visual_inspection_contribution.dart';
import '../engine/engine_context.dart';
import '../models/scan_session.dart';
import '../variant/atlas_variant_resolution_bridge.dart';
import 'atlas_authenticity_engine.dart';
import 'engines/atlas_high_signal_inspection_pipeline.dart';
import 'models/atlas_visual_inspection.dart';

class AtlasVisualInspectionService {
  const AtlasVisualInspectionService({
    this.authenticityEngine = const AtlasAuthenticityEngine(),
    this.highSignalPipeline = const AtlasHighSignalInspectionPipeline(),
    this.variantResolutionBridge = const AtlasVariantResolutionBridge(),
  });

  final AtlasAuthenticityEngine authenticityEngine;
  final AtlasHighSignalInspectionPipeline highSignalPipeline;
  final AtlasVariantResolutionBridge variantResolutionBridge;

  Future<ScanSession> inspect({
    required ScanSession session,
    required AtlasVisualInspection inspection,
  }) async {
    final AtlasVisualInspection enrichedInspection = highSignalPipeline.enrich(
      sessionId: session.sessionId,
      inspection: inspection,
      expectedResolvedVariant: session.resolvedProduct?.variantName,
    );

    var updated = session;

    for (final image in enrichedInspection.images) {
      updated = updated.addImage(image);
    }

    updated = updated.addContribution(
      VisualInspectionContribution(
        contributionId: 'visual-inspection-${session.sessionId}',
        sourceEngineId: 'atlas.visual-inspection',
        createdAt: DateTime.now().toUtc(),
        inspection: enrichedInspection,
        confidence: enrichedInspection.resolvedEvidenceConfidence,
        attributes: <String, Object?>{
          'inspectionAreas': enrichedInspection.inspectedAreas
              .map((area) => area.name)
              .toList(growable: false),
          'structuredObservationCount': enrichedInspection.observations.length,
          'hasReferenceCriteria': enrichedInspection.referenceCriteria != null,
          'highSignalEnginesApplied': true,
        },
      ),
      conflictPolicy: ContributionConflictPolicy.replaceExisting,
    );

    // Scanner 5D interprets the structured sticker evidence for the existing
    // variant-resolution model. This intentionally happens outside the session
    // projector so visual evidence cannot replace canonical product identity.
    final variantResolution = variantResolutionBridge.resolve(updated);
    if (variantResolution != null) {
      updated = updated.copyWith(variantResolution: variantResolution);
    }

    final result = await authenticityEngine.process(
      updated,
      EngineContext(
        requestId: session.sessionId,
        createdAt: DateTime.now().toUtc(),
        deviceType: EngineDeviceType.mobile,
      ),
    );

    return result.session;
  }
}
