import 'package:image_picker/image_picker.dart';

import '../models/scan_image.dart';

enum AtlasInspectionImageSource { camera, gallery }

abstract interface class AtlasInspectionImageCapture {
  Future<ScanImage?> capture({
    required AtlasInspectionImageSource source,
    required ScanImageType type,
  });
}

final class AtlasInspectionImageCaptureService
    implements AtlasInspectionImageCapture {
  AtlasInspectionImageCaptureService({ImagePicker? picker})
    : _picker = picker ?? ImagePicker();

  final ImagePicker _picker;

  @override
  Future<ScanImage?> capture({
    required AtlasInspectionImageSource source,
    required ScanImageType type,
  }) async {
    final file = await _picker.pickImage(
      source: source == AtlasInspectionImageSource.camera
          ? ImageSource.camera
          : ImageSource.gallery,
      imageQuality: 92,
    );
    if (file == null) return null;
    final capturedAt = DateTime.now().toUtc();
    return ScanImage(
      imageId: 'inspection_${capturedAt.microsecondsSinceEpoch}_${type.name}',
      type: type,
      status: ScanImageStatus.captured,
      capturedAt: capturedAt,
      localPath: file.path,
      mimeType: file.mimeType,
      byteSize: await file.length(),
    );
  }
}

import 'package:google_mlkit_text_recognition/google_mlkit_text_recognition.dart';

abstract interface class AtlasTextRecognition {
  Future<String> recognize(String path);
  Future<void> close();
}

final class AtlasTextRecognitionService implements AtlasTextRecognition {
  AtlasTextRecognitionService({TextRecognizer? recognizer})
    : _recognizer = recognizer ?? TextRecognizer();

  final TextRecognizer _recognizer;

  @override
  Future<String> recognize(String path) async {
    final normalized = path.trim();
    if (normalized.isEmpty) return '';
    final result = await _recognizer.processImage(
      InputImage.fromFilePath(normalized),
    );
    return result.text.trim();
  }

  @override
  Future<void> close() => _recognizer.close();
}

// Immutable image reference associated with a scan session.
//
// Images may initially exist only on the local device. Remote storage

  const ScanImage({
    required this.imageId,
    required this.type,
    required this.status,
    required this.capturedAt,
    this.localPath,
    this.remoteUrl,
    this.mimeType,
    this.width,
    this.height,
    this.byteSize,
    this.imageFingerprint,
    this.processingNotes = const [],
  });

  final String imageId;

  final ScanImageType type;

  final ScanImageStatus status;

  final DateTime capturedAt;

  final String? localPath;

  final String? remoteUrl;

  final String? mimeType;

  final int? width;

  final int? height;

  final int? byteSize;

  final String? imageFingerprint;

  final List<String> processingNotes;

  bool get hasLocalPath {
    return localPath != null && localPath!.trim().isNotEmpty;
  }

  bool get hasRemoteUrl {
    return remoteUrl != null && remoteUrl!.trim().isNotEmpty;
  }

  bool get hasFingerprint {
    return imageFingerprint != null && imageFingerprint!.trim().isNotEmpty;
  }

  bool get isAvailable {
    return hasLocalPath || hasRemoteUrl;
  }

  bool get isOriginal {
    return type == ScanImageType.original;
  }

  bool get isProcessed {
    return status == ScanImageStatus.processed ||
        status == ScanImageStatus.approved;
  }

  bool get requiresSync {
    return hasLocalPath && !hasRemoteUrl;
  }

  ScanImage copyWith({
    String? imageId,
    ScanImageType? type,
    ScanImageStatus? status,
    DateTime? capturedAt,
    String? localPath,
    String? remoteUrl,

    int? byteSize,
    String? imageFingerprint,
    List<String>? processingNotes,
  }) {

import '../contributions/contribution_conflict_policy.dart';
import '../contributions/visual_inspection_contribution.dart';
import '../engine/engine_context.dart';
import '../models/scan_session.dart';
import 'atlas_authenticity_engine.dart';
import 'models/atlas_visual_inspection.dart';

class AtlasVisualInspectionService {
  const AtlasVisualInspectionService({
    this.authenticityEngine = const AtlasAuthenticityEngine(),
  });

  final AtlasAuthenticityEngine authenticityEngine;

  Future<ScanSession> inspect({
    required ScanSession session,
    required AtlasVisualInspection inspection,
  }) async {
    var updated = session;

    for (final image in inspection.images) {
      updated = updated.addImage(image);
    }

    updated = updated.addContribution(
      VisualInspectionContribution(
        contributionId: 'visual-inspection-${session.sessionId}',
        sourceEngineId: 'atlas.visual-inspection',
        createdAt: DateTime.now().toUtc(),
        inspection: inspection,
        confidence: inspection.resolvedEvidenceConfidence,
        attributes: <String, Object?>{
          'inspectionAreas': inspection.inspectedAreas
              .map((area) => area.name)
              .toList(growable: false),
          'structuredObservationCount': inspection.observations.length,
          'hasReferenceCriteria': inspection.referenceCriteria != null,
        },
      ),
      conflictPolicy: ContributionConflictPolicy.replaceExisting,
    );

    final result = await authenticityEngine.process(
      updated,
      EngineContext(
        requestId: session.sessionId,
        createdAt: DateTime.now().toUtc(),
        deviceType: EngineDeviceType.mobile,
      ),
    );

    return result.session;
  }
}

            sum + observation.confidence,
      );
      return total / observations.length;

      return 0.85;

    if (hasAnyLegacyEvidence) {
      return 0.65;

      result.add(AtlasInspectionArea.autograph);
    }
    if (hasCoa != null || hasAuthenticationEvidence) {
      result.add(AtlasInspectionArea.coa);
    }

    return Set<AtlasInspectionArea>.unmodifiable(result);
  }

Object? _deepFreeze(Object? value) {
