import 'package:flutter/foundation.dart';

import 'package:funkovault2/models/pop.dart';

@immutable
class AtlasCollectorState {
  AtlasCollectorState({
    required List<Pop> ownedMatches,
    required List<Pop> wishlistMatches,
    required List<Pop> relatedMatches,
    this.isOwnedWishlisted = false,
  })  : ownedMatches = List<Pop>.unmodifiable(
    ownedMatches,
  ),
        wishlistMatches = List<Pop>.unmodifiable(
          wishlistMatches,
        ),
        relatedMatches = List<Pop>.unmodifiable(
          relatedMatches,
        );

  factory AtlasCollectorState.empty() {
    return AtlasCollectorState(
      ownedMatches: const <Pop>[],
      wishlistMatches: const <Pop>[],
      relatedMatches: const <Pop>[],
    );
  }

  final List<Pop> ownedMatches;
  final List<Pop> wishlistMatches;
  final List<Pop> relatedMatches;

  /// True when an owned collectible is also tracked through the
  /// AppState Wishlist overlay.
  final bool isOwnedWishlisted;

  bool get isOwned {
    return ownedMatches.isNotEmpty;
  }

  bool get hasWishlistRecord {
    return wishlistMatches.isNotEmpty;
  }

  bool get isWishlisted {
    return isOwnedWishlisted || hasWishlistRecord;
  }

  bool get hasRelatedMatches {
    return relatedMatches.isNotEmpty;
  }

  int get ownedQuantity {
    return ownedMatches.fold<int>(
      0,
          (
          int total,
          Pop pop,
          ) {
        return total + pop.quantity;
      },
    );
  }

  Pop? get primaryOwnedMatch {
    if (ownedMatches.isEmpty) {
      return null;
    }

    return ownedMatches.first;
  }

  Pop? get primaryWishlistMatch {
    if (wishlistMatches.isEmpty) {
      return null;
    }

    return wishlistMatches.first;
  }

  Pop? get bestExistingMatch {
    return primaryOwnedMatch ?? primaryWishlistMatch;
  }

  List<Pop> get allExactMatches {
    return List<Pop>.unmodifiable(
      <Pop>[
        ...ownedMatches,
        ...wishlistMatches,
      ],
    );
  }

  AtlasCollectorState copyWith({
    List<Pop>? ownedMatches,
    List<Pop>? wishlistMatches,
    List<Pop>? relatedMatches,
    bool? isOwnedWishlisted,
  }) {
    return AtlasCollectorState(
      ownedMatches: ownedMatches ?? this.ownedMatches,
      wishlistMatches:
      wishlistMatches ?? this.wishlistMatches,
      relatedMatches:
      relatedMatches ?? this.relatedMatches,
      isOwnedWishlisted:
      isOwnedWishlisted ?? this.isOwnedWishlisted,
    );
  }

  @override
  String toString() {
    return 'AtlasCollectorState('
        'owned: ${ownedMatches.length}, '
        'ownedQuantity: $ownedQuantity, '
        'wishlistRecords: ${wishlistMatches.length}, '
        'ownedWishlisted: $isOwnedWishlisted, '
        'related: ${relatedMatches.length}'
        ')';
  }
}

final class AtlasCollectorResolver {
  AtlasCollectorResolver({
    AtlasCollectorStateEngine? collectorStateEngine,
    AtlasCollectorInsightEngine? collectorInsightEngine,
    AtlasTaxonomyService? taxonomyService,
  })  : _collectorStateEngine =
      collectorStateEngine ??
          const AtlasCollectorStateEngine(),
        _collectorInsightEngine =
            collectorInsightEngine ??
                const AtlasCollectorInsightEngine(),
        _taxonomyService =
            taxonomyService ??
                AtlasTaxonomyService.instance;

  static final AtlasCollectorResolver instance =
  AtlasCollectorResolver();

  final AtlasCollectorStateEngine
  _collectorStateEngine;

  final AtlasCollectorInsightEngine
  _collectorInsightEngine;

  final AtlasTaxonomyService
  _taxonomyService;

  AtlasResolvedProduct resolve({
    required AtlasCatalogProduct catalogProduct,
    required List<Pop> collectionPops,
    Set<String> ownedWishlistIdentityKeys =
    const <String>{},
    String collectionName =
    'All Atlas Collections',
    String source =
    'Atlas Collector Resolver',
    String scannedBarcode = '',
    DateTime? conversionTimestamp,
  }) {
    final Pop catalogPop =
    AtlasCatalogConverter.toPop(
      catalogProduct,
      owner: 'Atlas Catalog',
      quantity: 1,
      timestamp: conversionTimestamp,
    );

    final AtlasCollectorState collectorState =
    _collectorStateEngine.evaluate(
      product: catalogPop,
      collectionPops: collectionPops,
      scannedBarcode: scannedBarcode,
      ownedWishlistIdentityKeys:
      ownedWishlistIdentityKeys,
    );

    final AtlasTaxonomy taxonomy =
    _taxonomyService.resolve(
      catalogProduct,
    );

    final AtlasCollectorInsight insight =
    _collectorInsightEngine.evaluateProduct(
      product: catalogPop,
      collectionPops: collectionPops,
      collectionName: collectionName,
      ownedWishlistIdentityKeys:
      ownedWishlistIdentityKeys,
      source: source,
    );

    return AtlasResolvedProduct(
      catalogProduct: catalogProduct,
      catalogPop: catalogPop,
      collectorState: collectorState,
      taxonomy: taxonomy,
      insight: insight,
    );
  }

  List<AtlasResolvedProduct> resolveAll({
    required Iterable<AtlasCatalogProduct>
    catalogProducts,
    required List<Pop> collectionPops,
    Set<String> ownedWishlistIdentityKeys =
    const <String>{},
    String collectionName =
    'All Atlas Collections',
    String source =
    'Atlas Collector Resolver',
  }) {
    final DateTime batchTimestamp =
    DateTime.now();

    final List<AtlasResolvedProduct>
    resolvedProducts =
    <AtlasResolvedProduct>[];

    int productIndex = 0;

    for (final AtlasCatalogProduct catalogProduct
    in catalogProducts) {
      resolvedProducts.add(
        resolve(
          catalogProduct: catalogProduct,
          collectionPops: collectionPops,
          ownedWishlistIdentityKeys:
          ownedWishlistIdentityKeys,
          collectionName: collectionName,
          source: source,
          conversionTimestamp:
          batchTimestamp.add(
            Duration(
              microseconds: productIndex,
            ),
          ),
        ),
      );

      productIndex++;
    }

    return List<AtlasResolvedProduct>.unmodifiable(
      resolvedProducts,
    );
  }

  AtlasResolvedProduct resolveForBarcode({
    required AtlasCatalogProduct catalogProduct,
    required String scannedBarcode,
    required List<Pop> collectionPops,
    Set<String> ownedWishlistIdentityKeys =
    const <String>{},
    String collectionName =
    'All Atlas Collections',

    return items.where(

  int get missingReleaseCount {
    return items.where(
          (AtlasCompletionItem item) {
        return item.isMissing;
      },
    ).length;
  }

    return totalReleaseCount - ownedReleaseCount;
  }

          AtlasCompletionItem item,

        return total + item.ownedQuantity;

    if (totalReleaseCount == 0) {
      return 0;
    }

    return (
        ownedReleaseCount /
            totalReleaseCount
    ) *
        100;
  }

  double get completionRatio {
    if (totalReleaseCount == 0) {
      return 0;
    }

    return (
        ownedReleaseCount /
            totalReleaseCount
    ).clamp(
      0,
      1,
    );
  }

  double get wishlistProgressPercent {
    if (totalReleaseCount == 0) {
      return 0;
    }

    return (
        (
            ownedReleaseCount +
                wishlistReleaseCount
        ) /
            totalReleaseCount
    ) *
        100;
  }

  bool get isComplete {
    return totalReleaseCount > 0 &&
        ownedReleaseCount ==
            totalReleaseCount;
  }

  bool get hasWishlistItems {
    return wishlistReleaseCount > 0;
  }

  bool get hasMissingItems {
    return missingReleaseCount > 0;
  }

  bool get hasStarted {

  }

import 'package:funkovault2/atlas/catalog/models/atlas_catalog_product.dart';
import 'package:funkovault2/atlas/collector/models/atlas_resolved_product.dart';
import 'package:funkovault2/atlas/collector/services/atlas_collector_resolver.dart';
import 'package:funkovault2/atlas/completion/models/atlas_completion_item.dart';
import 'package:funkovault2/atlas/completion/models/atlas_completion_query.dart';
import 'package:funkovault2/atlas/completion/models/atlas_completion_result.dart';
import 'package:funkovault2/atlas/taxonomy/models/atlas_taxonomy.dart';
import 'package:funkovault2/models/pop.dart';

final class AtlasCompletionEngine {
  AtlasCompletionEngine({
    AtlasCollectorResolver? collectorResolver,
  }) : _collectorResolver =
      collectorResolver ??
          AtlasCollectorResolver.instance;

  final AtlasCollectorResolver _collectorResolver;

  AtlasCompletionResult evaluate({
    required AtlasCompletionQuery query,
    required Iterable<AtlasCatalogProduct> catalogProducts,
    required List<Pop> collectionPops,
    Set<String> ownedWishlistIdentityKeys =
    const <String>{},
    String collectionName = 'All Atlas Collections',
  }) {
    final List<AtlasCatalogProduct> normalizedProducts =
    _normalizeCatalogProducts(
      catalogProducts,
    );

    if (normalizedProducts.isEmpty) {
      return AtlasCompletionResult(
        query: query,
        items: const <AtlasCompletionItem>[],
      );
    }

    final List<AtlasResolvedProduct> resolvedProducts =
    _collectorResolver.resolveAll(
      catalogProducts: normalizedProducts,
      collectionPops: collectionPops,
      ownedWishlistIdentityKeys:
      ownedWishlistIdentityKeys,
      collectionName: collectionName,
      source: 'Atlas Completion Engine',
    );

    final List<AtlasCompletionItem> completionItems =
    <AtlasCompletionItem>[];

    for (final AtlasResolvedProduct resolvedProduct
    in resolvedProducts) {
      if (!_matchesQuery(
        taxonomy: resolvedProduct.taxonomy,
        query: query,
      )) {
        continue;
      }

      if (!query.includeInactiveCatalogItems &&
          _isInactiveCatalogProduct(
            resolvedProduct.catalogPop,
          )) {
        continue;
      }

      final bool includeWishlistState =
          query.includeWishlist &&
              resolvedProduct.collectorState.isWishlisted;

      completionItems.add(
        AtlasCompletionItem.build(
          catalogProduct:
          resolvedProduct.catalogProduct,
          taxonomy:
          resolvedProduct.taxonomy,
          ownedMatches:
          resolvedProduct.collectorState.ownedMatches,
          wishlistMatches:
          query.includeWishlist
              ? resolvedProduct
              .collectorState
              .wishlistMatches
              : const <Pop>[],
          isWishlistOverlayActive:
          includeWishlistState &&
              resolvedProduct.collectorState
                  .isOwnedWishlisted,
          releaseWeight: _releaseWeightFor(
            resolvedProduct,
          ),
        ),
      );
    }

    completionItems.sort(
          (
          AtlasCompletionItem first,
          AtlasCompletionItem second,
          ) {
        final int stateComparison =
        _stateOrder(
          first.state,
        ).compareTo(
          _stateOrder(
            second.state,
          ),
        );

        if (stateComparison != 0) {
          return stateComparison;
        }

        final int nameComparison =
        first.displayName
            .toLowerCase()
            .compareTo(
          second.displayName.toLowerCase(),
        );

        if (nameComparison != 0) {
          return nameComparison;
        }

        final int numberComparison =
        (first.productNumber ?? '')
            .compareTo(
          second.productNumber ?? '',
        );

        if (numberComparison != 0) {
          return numberComparison;
        }

        return first.productId.compareTo(
          second.productId,
        );
      },
    );

    return AtlasCompletionResult(
      query: query,
      items: completionItems,
    );
  }

  bool _matchesQuery({
    required AtlasTaxonomy taxonomy,
    required AtlasCompletionQuery query,
  }) {
    if (query.isEmpty) {
      return true;
    }

    for (final AtlasCompletionFilter filter
    in query.filters) {
      if (!taxonomy.containsNode(
        scope: filter.scope,
        id: filter.nodeId,
      )) {
        return false;
      }
    }

    return true;
  }

  List<AtlasCatalogProduct> _normalizeCatalogProducts(
      Iterable<AtlasCatalogProduct> products,
      ) {
    final Map<String, AtlasCatalogProduct> productsById =
    <String, AtlasCatalogProduct>{};

    int fallbackIndex = 0;

    for (final AtlasCatalogProduct product in products) {
      final String productId =
      product.productId.value.trim();

      final String key;

      if (productId.isNotEmpty) {
        key = productId.toLowerCase();
      } else {
        key = _fallbackProductKey(
          product: product,
          index: fallbackIndex,
        );
      }

      productsById.putIfAbsent(
        key,
            () => product,
      );

      fallbackIndex++;
    }

    return List<AtlasCatalogProduct>.unmodifiable(
      productsById.values,
    );
  }

  String _fallbackProductKey({
    required AtlasCatalogProduct product,
    required int index,
  }) {
    final String normalizedName =
    _normalizeText(
      product.displayName,
    );

    final String normalizedNumber =
    _normalizeText(
      product.productNumber ?? '',
    );

    final String normalizedVariant =
    _normalizeText(
      product.variantName ?? '',
    );

    final String barcodeKey =
    product.barcodes
        .map(_normalizeBarcode)
        .where(
          (String value) => value.isNotEmpty,
    )
        .join('|');

    return <String>[
      'fallback',
      normalizedName,
      normalizedNumber,
      normalizedVariant,
      barcodeKey,
      '$index',
    ].join(':');
  }

  bool _isInactiveCatalogProduct(
      Pop product,
      ) {
    final String normalizedStatus =
    product.productionStatus
        .trim()
        .toLowerCase();

    if (normalizedStatus.isEmpty) {
      return false;
    }

    const Set<String> inactiveStatuses =
    <String>{
      'cancelled',
      'canceled',
      'prototype',
      'concept',
      'unreleased',
      'inactive',
    };

    return inactiveStatuses.contains(
      normalizedStatus,
    );
  }

  double _releaseWeightFor(
      AtlasResolvedProduct resolvedProduct,
      ) {
    final Object? rarity =
    resolvedProduct
        .catalogProduct
        .attributes['rarity'];

    final String normalizedRarity =
        rarity
            ?.toString()
            .trim()
            .toLowerCase() ??
            '';

    if (normalizedRarity.contains(
      'chase',
    )) {
      return 2;
    }

    if (normalizedRarity.contains(
      'limited',
    ) ||
        normalizedRarity.contains(
          'numbered',
        )) {
      return 1.5;
    }

    return 1;
  }

  int _stateOrder(
      AtlasCompletionItemState state,
      ) {
    return switch (state) {
      AtlasCompletionItemState.owned => 0,
      AtlasCompletionItemState.wishlisted => 1,
      AtlasCompletionItemState.missing => 2,
    };
  }

  String _normalizeText(
      String value,
      ) {
    return value
        .trim()
        .toLowerCase()
        .replaceAll(
      RegExp(r'[^a-z0-9]+'),
      ' ',
    )
        .replaceAll(
      RegExp(r'\s+'),
      ' ',
    )
        .trim();
  }

  String _normalizeBarcode(
      String value,
      ) {
    return value.replaceAll(
      RegExp(r'[^0-9A-Za-z]'),
      '',
    );
  }
}

/// Compatibility service for the existing Vault duplicate-review workflow.
///
/// Atlas Identity Intelligence now performs the actual analysis. This service
/// converts the richer intelligence results into the older report structure
/// currently expected by Vault screens.
class DuplicateReviewService {
  const DuplicateReviewService({
    this.identityReviewService = const IdentityReviewService(),
  });

  final IdentityReviewService identityReviewService;

  DuplicateReviewReport buildReport(List<Pop> pops) {
    final identityResult = identityReviewService.analyze(pops);

    final sameIdentityGroups = identityResult.exactDuplicates
        .map(_convertGroup)
        .toList(growable: false);

    final possibleVariantGroups = identityResult.allGroups
        .where(
          (group) =>
      group.relationship != IdentityRelationship.exactDuplicate,
    )
        .map(_convertGroup)
        .toList(growable: false);

    return DuplicateReviewReport(
      sameIdentityGroups: sameIdentityGroups,
      possibleVariantGroups: possibleVariantGroups,
      identityResult: identityResult,
    );
  }

  DuplicateReviewGroup _convertGroup(
      IdentityReviewGroup group,
      ) {
    return DuplicateReviewGroup(
      key: group.key,
      pops: group.pops,
      relationship: group.relationship,
      title: group.title,
      explanation: group.explanation,
      confidence: group.confidence,
      evidence: group.evidence,
    );
  }
}

class DuplicateReviewReport {
  const DuplicateReviewReport({
    required this.sameIdentityGroups,
    required this.possibleVariantGroups,
    this.identityResult = const IdentityReviewResult(),
  });

  /// Records Atlas believes represent the same exact collectible release.
  final List<DuplicateReviewGroup> sameIdentityGroups;

  /// Related records with meaningful release differences or incomplete data.
  ///
  /// This keeps the existing property name so current Vault screens continue
  /// working while the UI is gradually upgraded to Identity Intelligence.
  final List<DuplicateReviewGroup> possibleVariantGroups;

  /// Complete Identity Intelligence result for newer Atlas features.
  final IdentityReviewResult identityResult;

  int get totalGroups {
    return sameIdentityGroups.length + possibleVariantGroups.length;
  }

  int get exactDuplicateGroupCount => sameIdentityGroups.length;

  int get reviewGroupCount => possibleVariantGroups.length;

  bool get hasGroups => totalGroups > 0;

  bool get hasExactDuplicates => sameIdentityGroups.isNotEmpty;

  bool get hasReviewGroups => possibleVariantGroups.isNotEmpty;
}

class DuplicateReviewGroup {
  const DuplicateReviewGroup({
    required this.key,
    required this.pops,
    this.relationship = IdentityRelationship.needsReview,
    this.title = 'Identity Review Needed',
    this.explanation = '',
    this.confidence = 0,
    this.evidence = const <String>[],
  });

  final String key;
  final List<Pop> pops;

  /// Richer classification supplied by Atlas Identity Intelligence.
  final IdentityRelationship relationship;

  final String title;
  final String explanation;

  /// Classification confidence from 0 to 100.
  final int confidence;

  /// Human-readable reasons supporting the classification.
  final List<String> evidence;

  Pop get firstPop => pops.first;

  int get itemCount => pops.length;

  int get totalQuantity {
    return pops.fold<int>(
      0,
          (total, pop) => total + pop.quantity,
    );
  }

  String get name {
    final value = firstPop.name.trim();
    return value.isEmpty ? 'Unnamed Pop' : value;
  }

  String get number {
    final value = firstPop.number.trim();
    return value.isEmpty ? 'No #' : '#$value';
  }

  bool get isExactDuplicate {
    return relationship == IdentityRelationship.exactDuplicate;
  }

  bool get requiresReview => !isExactDuplicate;
}

  List<Pop> pops(AppState appState, {String? collectionName}) {
    if (collectionName == null) {
      return allPops(appState);

  }

      pops(appState, collectionName: collectionName).length;

  double totalValue(AppState appState, {String? collectionName}) {
    if (collectionName == null) {
      return appState.totalValue;
    }

    return pops(
      appState,
      collectionName: collectionName,
    ).fold<double>(0, (total, pop) => total + pop.value);
  }

  List<Pop> allPops(AppState appState) {
    return appState.collections.values
        .expand((collection) => collection)
        .toList();
  }

  Map<String, int> ownerCounts(AppState appState, {String? collectionName}) {
    final counts = <String, int>{};

    for (final pop in pops(appState, collectionName: collectionName)) {
      counts[pop.owner] = (counts[pop.owner] ?? 0) + 1;
    }

    return counts;
  }

  Map<String, double> ownerValues(AppState appState, {String? collectionName}) {
    final values = <String, double>{};

    for (final pop in pops(appState, collectionName: collectionName)) {
      values[pop.owner] = (values[pop.owner] ?? 0) + pop.value;
    }

    return values;
  }

  List<MapEntry<String, double>> ownerLeaderboard(
    AppState appState, {
    String? collectionName,
  }) {
    final leaderboard = ownerValues(
      appState,
      collectionName: collectionName,
    ).entries.toList();

    leaderboard.sort((a, b) => b.value.compareTo(a.value));

    return leaderboard;
  }

  Map<String, int> categoryCounts(AppState appState, {String? collectionName}) {
    final categories = <String, int>{};

    for (final pop in pops(appState, collectionName: collectionName)) {
      final category = pop.category.isEmpty ? 'Uncategorized' : pop.category;

      categories[category] = (categories[category] ?? 0) + 1;
    }

    return categories;
  }

  Map<String, double> categoryValues(
    AppState appState, {
    String? collectionName,
  }) {
    final values = <String, double>{};

    for (final pop in pops(appState, collectionName: collectionName)) {
      final category = pop.category.isEmpty ? 'Uncategorized' : pop.category;

      values[category] = (values[category] ?? 0) + pop.value;
    }

    return values;
  }

  List<Pop> mostValuablePops(AppState appState, {String? collectionName}) {
    final scopedPops = pops(appState, collectionName: collectionName);

    );

    AppState appState, {
    String? collectionName,
  }) {
    return duplicateReviewGroups(
      appState,
      collectionName: collectionName,
    ).map((group) => group.pops).toList();
  }

    required Set<String> overlayIdentityKeys,
  }) {
    final List<Pop> results = <Pop>[];
    final Set<String> addedRecordIds = <String>{};
    final Set<String> wishlistRecordIdentityKeys = <String>{};

    for (final Pop pop in collectionPops) {
      if (!pop.isWishlist) {
        continue;
      }

      results.add(pop);

      addedRecordIds.add(pop.id);

      final String identityKey = pop.collectibleIdentityKey.trim();

      if (identityKey.isNotEmpty) {
        wishlistRecordIdentityKeys.add(identityKey);
      }
    }

    for (final Pop pop in collectionPops) {
      if (pop.isWishlist || addedRecordIds.contains(pop.id)) {
        continue;
      }

      final String identityKey = pop.collectibleIdentityKey.trim();

      if (identityKey.isEmpty || !overlayIdentityKeys.contains(identityKey)) {
        continue;
      }

      // Prevent an owned overlay and an old Wishlist-only
      // record from showing the same release twice.
      if (wishlistRecordIdentityKeys.contains(identityKey)) {
        continue;
      }

      results.add(pop);

      addedRecordIds.add(pop.id);
    }

    return List<Pop>.unmodifiable(results);

  Future<CollectionOperationResult> renameCollection(
    String currentName,
    String newName,
  ) async {
    final existingName = _matchingCollectionName(currentName);
    final trimmedNewName = newName.trim();

    if (existingName == null) {
      return CollectionOperationResult.notFound;
    }

    if (trimmedNewName.isEmpty) {
      return CollectionOperationResult.invalidName;
    }

    final duplicateName = _matchingCollectionName(trimmedNewName);

    if (duplicateName != null && duplicateName != existingName) {
      return CollectionOperationResult.duplicateName;
    }

    if (trimmedNewName == existingName) {
      return CollectionOperationResult.success;
    }

    final updatedCollections = <String, List<Pop>>{};

    for (final entry in _collections.entries) {
      updatedCollections[entry.key == existingName
              ? trimmedNewName
              : entry.key] =
          entry.value;
    }

    final updatedWishlistKeys = <String, Set<String>>{};

    for (final entry in _ownedWishlistIdentityKeysByCollection.entries) {
      updatedWishlistKeys[entry.key == existingName
          ? trimmedNewName
          : entry.key] = Set<String>.from(
        entry.value,
      );
    }

    updatedWishlistKeys.putIfAbsent(trimmedNewName, () => <String>{});

    final updatedSelection = _selectedCollection == existingName
        ? trimmedNewName
        : _selectedCollection;

    try {
      await _saveData(
        selectedCollection: updatedSelection,
        collections: updatedCollections,
        ownedWishlistIdentityKeysByCollection: updatedWishlistKeys,
      );
    } catch (_) {
      return CollectionOperationResult.persistenceFailure;
    }

    _replaceCollections(updatedCollections);
    _replaceWishlistKeys(updatedWishlistKeys);
    _selectedCollection = updatedSelection;
    notifyListeners();

    return CollectionOperationResult.success;
  }

    _replaceWishlistKeys(updatedWishlistKeys);
    _selectedCollection = updatedSelection;
    notifyListeners();

    return CollectionOperationResult.success;
  }

  Future<bool> addPop(Pop pop, {String? collectionName}) async {
    final targetCollectionName = collectionName == null
        ? _selectedCollection
        : _matchingCollectionName(collectionName);
    final pops = targetCollectionName == null
        ? null
        : _collections[targetCollectionName];

    if (pops == null) {
      return false;
    }

    final now = DateTime.now().toIso8601String();

    final newRecordKey = pop.collectionRecordKey;

    final existingIndex = newRecordKey.isEmpty
        ? -1
        : pops.indexWhere(
            (existingPop) => existingPop.collectionRecordKey == newRecordKey,
          );

    if (existingIndex != -1) {
      final existingPop = pops[existingIndex];

      pops[existingIndex] = existingPop.copyWith(
        quantity: existingPop.quantity + pop.quantity,
        lastUpdated: now,
      );

      await addActivity(
        Activity(
          id: DateTime.now().millisecondsSinceEpoch.toString(),
          popId: existingPop.id,
          action: 'quantity_increased',
          popName: pop.name,
          popNumber: pop.number,
          owner: pop.owner,
          actor: pop.owner,
          details: '${pop.name} quantity increased',
          timestamp: DateTime.now(),
        ),
      );
    } else {
      final addedAt =
          DateTime.tryParse(
            pop.dateAdded.trim().isEmpty ? now : pop.dateAdded,
          ) ??
          DateTime.now();

      final existingTimeline = List<AtlasTimelineEvent>.from(pop.timeline);
      final hasAddedEvent = existingTimeline.any(
        (event) => event.type == AtlasTimelineEventType.addedToVault,
      );

      if (!hasAddedEvent) {
        existingTimeline.add(
          AtlasTimelineEvent(
            id: 'added-${pop.id}-${addedAt.microsecondsSinceEpoch}',
            popId: pop.id,
            type: AtlasTimelineEventType.addedToVault,
            occurredAt: addedAt,
            title: 'Added to Vault',
            createdBy: AtlasTimelineEvent.systemCreatedBy,
            createdAt: addedAt,
          ),
        );
      }

      existingTimeline.sort((a, b) => a.occurredAt.compareTo(b.occurredAt));

      final datedPop = pop.copyWith(
        dateAdded: addedAt.toIso8601String(),
        lastUpdated: pop.lastUpdated.trim().isEmpty ? now : pop.lastUpdated,
        timeline: List<AtlasTimelineEvent>.unmodifiable(existingTimeline),
      );

      pops.add(datedPop);

      await addActivity(
        Activity(
          id: DateTime.now().millisecondsSinceEpoch.toString(),
          popId: datedPop.id,
          action: 'pop_added',
          popName: datedPop.name,
          popNumber: datedPop.number,
          owner: datedPop.owner,
          actor: datedPop.owner,
          details: '${datedPop.name} added',
          timestamp: DateTime.now(),
        ),
      );
    }

    await _saveData();

    notifyListeners();

    return true;
  }

  Future<PopImportResult> importPopsSafely(
    List<Pop> importedPops, {
    String? collectionName,
  }) async {
    final String targetCollection = collectionName ?? _selectedCollection;

    final String? matchingCollection = _matchingCollectionName(
      targetCollection,
    );

    if (matchingCollection == null) {
      return PopImportResult(
        attempted: importedPops.length,
        added: 0,
        skipped: importedPops.length,
      );
    }

    final List<Pop> currentPops = _collections[matchingCollection]!;

    final indexesByRecordKey = <String, int>{};

    for (var index = 0; index < currentPops.length; index++) {
      final key = currentPops[index].collectionRecordKey;

      if (key.isNotEmpty) {
        indexesByRecordKey[key] = index;
      }
    }

    final now = DateTime.now().toIso8601String();

    var added = 0;
    var consolidated = 0;

    for (final importedPop in importedPops) {
      final key = importedPop.collectionRecordKey;
      final existingIndex = key.isEmpty ? null : indexesByRecordKey[key];

      if (existingIndex != null) {
        final existingPop = currentPops[existingIndex];

        currentPops[existingIndex] = existingPop.copyWith(
          quantity: existingPop.quantity + importedPop.quantity,
          value: existingPop.value > 0 ? existingPop.value : importedPop.value,
          pricePaid: existingPop.pricePaid > 0
              ? existingPop.pricePaid
              : importedPop.pricePaid,
          imageUrl: existingPop.imageUrl.trim().isNotEmpty
              ? existingPop.imageUrl
              : importedPop.imageUrl,
          lastUpdated: now,
        );

        consolidated++;
        continue;
      }

      final datedPop = importedPop.copyWith(
        dateAdded: importedPop.dateAdded.trim().isEmpty
            ? now
            : importedPop.dateAdded,
        lastUpdated: importedPop.lastUpdated.trim().isEmpty
            ? now
            : importedPop.lastUpdated,
      );

      currentPops.add(datedPop);

      if (key.isNotEmpty) {
        indexesByRecordKey[key] = currentPops.length - 1;
      }

      added++;
    }

    await addActivity(
      Activity(
        id: DateTime.now().millisecondsSinceEpoch.toString(),
        action: 'csv_import',
        popName: 'CSV Import',
        popNumber: '',
        owner: matchingCollection,
        actor: _collectorProfile.displayName,
        details:
            'CSV import complete: $added records added, '
            '$consolidated matching copies consolidated',
        timestamp: DateTime.now(),
      ),
    );

    await _saveData();

    notifyListeners();

    return PopImportResult(
      attempted: importedPops.length,
      added: added,
      skipped: consolidated,
    );
  }

  Future<Pop?> updatePop(
    Pop updatedPop, {
    String? collectionName,
    String? existingRecordKey,
  }) async {
    final location = _resolveStoredPop(
      popId: updatedPop.id,
      collectionName: collectionName,
      recordKey: existingRecordKey,
    );

    if (location == null) {
      return null;
    }

    final existingPop = location.pops[location.index];
    final now = DateTime.now();
    final nowIso = now.toIso8601String();

    const timelineEngine = AtlasTimelineEngine();

    final detectedEvents = timelineEngine.comparePopChanges(
      previousPop: existingPop,
      updatedPop: updatedPop,
      occurredAt: now,
      source: 'Manual',
    );

    final timeline = <AtlasTimelineEvent>[
      ...existingPop.timeline,
      ...detectedEvents,
    ];

    final sortedTimeline = timelineEngine.chronological(timeline);

    final valueHistory = _updatedValueHistory(
      existingPop: existingPop,
      updatedPop: updatedPop,
      occurredAt: now,
    );

    final persistedPop = updatedPop.copyWith(
      dateAdded: existingPop.dateAdded,
      lastUpdated: nowIso,
      valueHistory: valueHistory,
      timeline: sortedTimeline,
    );

    location.pops[location.index] = persistedPop;

    await _saveData();

    notifyListeners();

    return persistedPop;
  }

  _StoredPopLocation? _resolveStoredPop({
    required String popId,
    String? collectionName,
    String? recordKey,
  }) {
    final normalizedId = popId.trim();

    if (normalizedId.isEmpty) {
      return null;
    }

    final Iterable<MapEntry<String, List<Pop>>> collectionsToSearch;

    if (collectionName != null) {
      final matchingName = _matchingCollectionName(collectionName);

      if (matchingName == null) {
        return null;
      }

      collectionsToSearch = <MapEntry<String, List<Pop>>>[
        MapEntry(matchingName, _collections[matchingName]!),
      ];
    } else {
      collectionsToSearch = _collections.entries;
    }

    final matches = <_StoredPopLocation>[];

    for (final collection in collectionsToSearch) {
      for (var index = 0; index < collection.value.length; index++) {
        final storedPop = collection.value[index];

        if (storedPop.id.trim() != normalizedId) {
          continue;
        }

        if (recordKey != null && storedPop.collectionRecordKey != recordKey) {
          continue;
        }

        matches.add((
          collectionName: collection.key,
          pops: collection.value,
          index: index,
        ));

        if (matches.length > 1) {
          return null;
        }
      }
    }

    return matches.singleOrNull;
  }

  List<ValueHistoryEntry> _updatedValueHistory({
    required Pop existingPop,
    required Pop updatedPop,
    required DateTime occurredAt,
  }) {
    if (existingPop.value == updatedPop.value) {
      return existingPop.valueHistory;
    }

    final entries = <ValueHistoryEntry>[...existingPop.valueHistory];

    if (entries.isEmpty && existingPop.value > 0) {
      entries.add(
        ValueHistoryEntry(
          value: existingPop.value,
          recordedAt:
              DateTime.tryParse(existingPop.lastUpdated) ??
              DateTime.tryParse(existingPop.dateAdded) ??
              occurredAt,
          source: ValueHistoryEntry.legacyValueSource,
        ),
      );
    }

    if (updatedPop.value > 0) {
      entries.add(
        ValueHistoryEntry(
          value: updatedPop.value,
          recordedAt: occurredAt,
          source: ValueHistoryEntry.manualSource,
        ),
      );
    }

    entries.sort((a, b) => a.recordedAt.compareTo(b.recordedAt));

    return List<ValueHistoryEntry>.unmodifiable(entries);
  }

  Future<bool> consolidateDuplicateGroup(
    List<String> popIds, {
    String? collectionName,
  }) async {
    if (popIds.length < 2) {
      return false;
    }

    List<Pop>? matchingCollection;
    List<Pop>? matchingPops;

    if (collectionName != null) {
      final String? matchingName = _matchingCollectionName(collectionName);

      if (matchingName == null) {
        return false;
      }

      final List<Pop> collection = _collections[matchingName]!;

      final List<Pop> foundPops = collection
          .where((pop) => popIds.contains(pop.id))
          .toList();

      if (foundPops.length != popIds.length) {
        return false;
      }

      matchingCollection = collection;
      matchingPops = foundPops;
    } else {
      final matchingLocations = <({List<Pop> collection, List<Pop> pops})>[];

      for (final collection in _collections.values) {
        final foundPops = collection
            .where((pop) => popIds.contains(pop.id))
            .toList();

        if (foundPops.length == popIds.length) {
          matchingLocations.add((collection: collection, pops: foundPops));
        }
      }

      if (matchingLocations.length != 1) {
        return false;
      }

      matchingCollection = matchingLocations.single.collection;

      matchingPops = matchingLocations.single.pops;
    }

    if (matchingPops.length < 2) {
      return false;
    }

    final identityKeys = matchingPops
        .map((pop) => pop.collectionRecordKey)
        .where((key) => key.isNotEmpty)
        .toSet();

    if (identityKeys.length != 1) {
      return false;
    }

    final now = DateTime.now().toIso8601String();

    final primaryPop = matchingPops.first;

    final totalQuantity = matchingPops.fold<int>(
      0,
      (total, pop) => total + pop.quantity,
    );

    final bestValue = matchingPops.fold<double>(
      0,
      (highest, pop) => pop.value > highest ? pop.value : highest,
    );

    final bestPricePaid = matchingPops.fold<double>(
      0,
      (highest, pop) => pop.pricePaid > highest ? pop.pricePaid : highest,
    );

    final bestImageUrl = matchingPops
        .map((pop) => pop.imageUrl.trim())
        .firstWhere(
          (value) => value.isNotEmpty,
          orElse: () => primaryPop.imageUrl,
        );

    final earliestDateAdded = _earliestSavedDate(
      matchingPops.map((pop) => pop.dateAdded),
    );

    final consolidatedPop = primaryPop.copyWith(
      quantity: totalQuantity,
      value: bestValue,
      pricePaid: bestPricePaid,
      imageUrl: bestImageUrl,
      dateAdded: earliestDateAdded.isEmpty
          ? primaryPop.dateAdded
          : earliestDateAdded,
      lastUpdated: now,
    );

    final primaryIndex = matchingCollection.indexWhere(
      (pop) => pop.id == primaryPop.id,
    );

    if (primaryIndex == -1) {
      return false;
    }

    matchingCollection[primaryIndex] = consolidatedPop;

    final removedIds = matchingPops.skip(1).map((pop) => pop.id).toSet();

    matchingCollection.removeWhere((pop) => removedIds.contains(pop.id));

    _activities.add(
      Activity(
        id: DateTime.now().millisecondsSinceEpoch.toString(),
        popId: consolidatedPop.id,
        action: 'duplicates_consolidated',
        popName: consolidatedPop.name,
        popNumber: consolidatedPop.number,
        owner: consolidatedPop.owner,
        actor: _collectorProfile.displayName,
        details:
            '${matchingPops.length} matching records consolidated into '
            'quantity $totalQuantity',
        timestamp: DateTime.now(),
      ),
    );

    if (_activities.length > 500) {
      _activities.removeAt(0);
    }

    await _saveData();

    notifyListeners();

    return true;
  }

  String _earliestSavedDate(Iterable<String> values) {
    DateTime? earliestDate;
    var earliestValue = '';

    for (final value in values) {
      final cleaned = value.trim();
      final parsed = DateTime.tryParse(cleaned);

      if (parsed == null) {
        continue;
      }

      if (earliestDate == null || parsed.isBefore(earliestDate)) {
        earliestDate = parsed;
        earliestValue = cleaned;
      }
    }

    return earliestValue;
  }

  Future<bool> mergeDuplicates({String? collectionName}) async {
    final String targetCollection = collectionName ?? _selectedCollection;

    final String? matchingCollection = _matchingCollectionName(
      targetCollection,
    );

    if (matchingCollection == null) {
      return false;
    }

    final List<Pop> pops = _collections[matchingCollection]!;

    final now = DateTime.now().toIso8601String();
    final merged = <String, Pop>{};

    for (final pop in pops) {
      final key = pop.collectionRecordKey;

      if (key.isEmpty) {
        merged[pop.id] = pop;
        continue;
      }

      if (merged.containsKey(key)) {
        final existing = merged[key]!;

        merged[key] = existing.copyWith(
          quantity: existing.quantity + pop.quantity,
          lastUpdated: now,
        );
      } else {
        merged[key] = pop;
      }
    }

    _collections[matchingCollection] = merged.values.toList();

    await _saveData();

    notifyListeners();

    return true;
  }

           resolutionQueue ?? AtlasResolutionQueue(maxConcurrentJobs: 4),
       ledger = ledger ?? AtlasScanLedger(),
       _catalogResolver = catalogResolver ?? AtlasBarcodeCatalogResolver();

  final AtlasResolutionQueue resolutionQueue;
  final AtlasScanLedger ledger;
  final AtlasBarcodeCatalogResolver _catalogResolver;
  final AtlasContinuousScanChanged? onChanged;

  final Map<String, AtlasContinuousScanResult> _resultsByTrack =
      <String, AtlasContinuousScanResult>{};

  bool _isDisposed = false;

  List<AtlasContinuousScanResult> get results {
    return List<AtlasContinuousScanResult>.unmodifiable(_resultsByTrack.values);
  }

  AtlasContinuousScanResult? resultForTrack(String trackId) {
    return _resultsByTrack[trackId.trim()];
  }

  AtlasContinuousScanSnapshot get snapshot {
    final AtlasResolutionQueueSnapshot queueSnapshot = resolutionQueue.snapshot;

    int observed = 0;
    int queued = 0;
    int resolving = 0;
    int resolved = 0;
    int review = 0;
    int failed = 0;
    int committed = 0;

    for (final AtlasScanLedgerEntry entry in ledger.entries) {
      switch (entry.status) {
        case AtlasScanLedgerStatus.observed:
          observed++;
        case AtlasScanLedgerStatus.queued:
          queued++;
        case AtlasScanLedgerStatus.resolving:
          resolving++;
        case AtlasScanLedgerStatus.resolved:
          resolved++;
        case AtlasScanLedgerStatus.needsReview:
          review++;
        case AtlasScanLedgerStatus.failed:
          failed++;
        case AtlasScanLedgerStatus.committed:
          committed++;
      }
    }

    return AtlasContinuousScanSnapshot(
      activeResolutionCount: queueSnapshot.activeCount,
      pendingResolutionCount: queueSnapshot.pendingCount,
      observedCount: observed,
      queuedCount: queued,
      resolvingCount: resolving,
      resolvedCount: resolved,
      reviewCount: review,
      failedCount: failed,
      committedCount: committed,
    );
  }

  /// Submits stable evidence from one physical track.
  ///
  /// Returns true only when a new background resolution job was accepted.
  bool submit({required String trackId, required ScanCode stableCode}) {
    if (_isDisposed) {
      return false;
    }

    final String normalizedTrackId = trackId.trim();

    if (normalizedTrackId.isEmpty) {
      throw ArgumentError.value(trackId, 'trackId', 'trackId cannot be empty.');
    }

    final DateTime now = DateTime.now().toUtc();
    final AtlasScanLedgerEntry? existing = ledger.entryForTrack(
      normalizedTrackId,
    );

    if (existing != null) {
      return false;
    }

    ledger.observe(trackId: normalizedTrackId, stableCode: stableCode, at: now);

    ledger.markQueued(trackId: normalizedTrackId, at: now);

    _notifyChanged();

    _enqueueResolution(
      trackId: normalizedTrackId,
      stableCode: stableCode,
      createdAt: now,
    );

    return true;
  }

  /// Explicitly retries a failed physical track.
  ///
  /// Retry never creates a second ledger entry and never auto-commits. The
  /// original stable barcode evidence is reused for a new resolution attempt.
  bool retryFailedTrack({required String trackId}) {
    if (_isDisposed) {
      return false;
    }

    final String normalizedTrackId = trackId.trim();

    if (normalizedTrackId.isEmpty ||
        resolutionQueue.containsTrack(normalizedTrackId)) {
      return false;
    }

    final AtlasScanLedgerEntry? entry = ledger.entryForTrack(normalizedTrackId);

    if (entry == null || !entry.hasFailed) {
      return false;
    }

    final DateTime now = DateTime.now().toUtc();

    ledger.markQueued(trackId: normalizedTrackId, at: now, allowRetry: true);

    _resultsByTrack.remove(normalizedTrackId);

    _notifyChanged();

    _enqueueResolution(
      trackId: normalizedTrackId,
      stableCode: entry.stableCode,
      createdAt: now,
    );

    return true;
  }

  bool markCommitted({required String trackId, DateTime? at}) {
    if (_isDisposed) {
      return false;
    }

    final AtlasScanLedgerEntry? entry = ledger.entryForTrack(trackId);

    if (entry == null) {
      return false;
    }

    if (entry.isCommitted) {
      return true;
    }

    if (!entry.isResolved && !entry.needsReview) {
      return false;
    }

    ledger.markCommitted(trackId: trackId, at: at ?? DateTime.now().toUtc());

    _notifyChanged();
    return true;
  }

  bool applyInspection({
    required String trackId,
    required ScanSession session,
  }) {
    if (_isDisposed) return false;
    final normalizedTrackId = trackId.trim();
    final existing = _resultsByTrack[normalizedTrackId];
    if (existing == null || existing.session.sessionId != session.sessionId) {
      return false;
    }
    final assessment = session.authenticityContribution?.assessment;
    if (assessment == null) return false;
    _resultsByTrack[normalizedTrackId] = existing.copyWith(session: session);
    ledger.applyInspectionReview(
      trackId: normalizedTrackId,
      requiresReview:
          assessment.requiresReview ||
          session.requiresUserAction ||
          !existing.catalogResolution.hasSingleMatch,
      at: DateTime.now().toUtc(),
    );
    _notifyChanged();
    return true;
  }

  bool get canResetCompletedSession {
    final List<AtlasScanLedgerEntry> entries = ledger.entries;

    return !_isDisposed &&
        resolutionQueue.snapshot.isIdle &&
        entries.isNotEmpty &&
        entries.every((AtlasScanLedgerEntry entry) => entry.isCommitted);
  }

  bool resetCompletedSession() {
    if (!canResetCompletedSession) {
      return false;
    }

    ledger.clear();
    _resultsByTrack.clear();
    _notifyChanged();
    return true;
  }

  void dispose() {
    if (_isDisposed) {
      return;
    }

    _isDisposed = true;
    resolutionQueue.dispose();
  }

  void _enqueueResolution({
    required String trackId,
    required ScanCode stableCode,
    required DateTime createdAt,
  }) {
    final String jobId = _createJobId(trackId: trackId, createdAt: createdAt);

    unawaited(
      resolutionQueue
          .enqueue(
            AtlasResolutionJob(
              jobId: jobId,
              trackId: trackId,
              createdAt: createdAt,
              execute: () =>
                  _resolveTrack(trackId: trackId, stableCode: stableCode),
            ),
          )
          .catchError((Object error, StackTrace stackTrace) {
            _handleQueueFailure(trackId: trackId, error: error);
          }),
    );
  }

  Future<void> _resolveTrack({
    required String trackId,
    required ScanCode stableCode,
  }) async {
    if (_isDisposed) {
      return;
    }

    final DateTime startedAt = DateTime.now().toUtc();

    ledger.markResolving(trackId: trackId, at: startedAt);
    _notifyChanged();

    try {
      final String executionId = _createExecutionId(
        startedAt,
        stableCode.normalizedValue,
      );

      final EngineContext engineContext = EngineContext(
        requestId: executionId,
        createdAt: startedAt,
        connectivity: EngineConnectivity.unknown,
        deviceType: EngineDeviceType.mobile,
        deviceCapabilities: const <String>{
          BarcodeIntelligenceEngine.capabilityId,
        },
        attributes: <String, Object?>{
          'scanSource': ScanCodeSource.camera.name,
          'scannerType': ScannerType.mobileScanner.name,
          'barcodeFormat': stableCode.format.name,
          'trackId': trackId,
          'continuousScan': true,
        },
      );

      final ScanSession initialSession = ScanSession(
        sessionId: executionId,
        startedAt: startedAt,
        status: ScanSessionStatus.analyzing,
        intent: ScanIntent.identifyProduct,
        performanceMode: ScanPerformanceMode.fastBarcodeOnly,
        scannerType: ScannerType.mobileScanner,
        detectedCodes: <ScanCode>[stableCode],
        bestCandidate: stableCode,
      );

      final pipeline = StandardScanPipeline.create(engineContext);
      final result = await pipeline.process(initialSession, engineContext);

      final ScanSession finalSession = result.finalSession;

      final ScanCode? resolvedCode =
          finalSession.bestCandidate ?? _firstDetectedCode(finalSession);

      final AtlasBarcodeResolution catalogResolution;

      if (resolvedCode == null ||
          !resolvedCode.isValid ||
          !resolvedCode.isProductBarcode) {
        catalogResolution = AtlasBarcodeResolution.invalid(
          barcode: resolvedCode?.normalizedValue ?? '',
        );
      } else if (finalSession.catalogProduct != null) {
        catalogResolution = AtlasBarcodeResolution.matches(
          barcode: resolvedCode.normalizedValue,
          matches: <AtlasCatalogProduct>[finalSession.catalogProduct!],
        );
      } else {
        catalogResolution = await _catalogResolver.resolve(
          resolvedCode.normalizedValue,
        );
      }

      final DateTime completedAt = DateTime.now().toUtc();

      final AtlasContinuousScanResult continuousResult =
          AtlasContinuousScanResult(
            trackId: trackId,
            session: finalSession,
            catalogResolution: catalogResolution,
            completedAt: completedAt,
          );

      _resultsByTrack[trackId] = continuousResult;

      if (continuousResult.requiresUserReview) {
        ledger.markNeedsReview(
          trackId: trackId,
          resolutionSessionId: finalSession.sessionId,
          at: completedAt,
        );
      } else {
        ledger.markResolved(
          trackId: trackId,
          resolutionSessionId: finalSession.sessionId,
          at: completedAt,
        );
      }
    } catch (error) {
      ledger.markFailed(
        trackId: trackId,
        error: error,
        at: DateTime.now().toUtc(),
      );
      rethrow;
    } finally {
      _notifyChanged();
    }
  }

  void _handleQueueFailure({required String trackId, required Object error}) {
    final AtlasScanLedgerEntry? entry = ledger.entryForTrack(trackId);

    if (entry == null || entry.hasFailed) {
      _notifyChanged();
      return;
    }

    if (entry.isQueued) {
      ledger.markResolving(trackId: trackId, at: DateTime.now().toUtc());
    }

    final AtlasScanLedgerEntry? current = ledger.entryForTrack(trackId);

    if (current != null && current.isResolving) {
      ledger.markFailed(
        trackId: trackId,
        error: error,
        at: DateTime.now().toUtc(),
      );
    }

    _notifyChanged();
  }

  ScanCode? _firstDetectedCode(ScanSession session) {
    if (session.detectedCodes.isEmpty) {
      return null;
    }

    return session.detectedCodes.first;
  }

  String _createExecutionId(DateTime createdAt, String code) {
    final int codeHash = code.hashCode.abs();

    return 'scan_'
        '${createdAt.microsecondsSinceEpoch}_'
        '$codeHash';
  }

  String _createJobId({required String trackId, required DateTime createdAt}) {
    return 'resolution_'
        '${createdAt.microsecondsSinceEpoch}_'
        '${trackId.hashCode.abs()}';
  }

  void _notifyChanged() {
    if (_isDisposed) {
      return;
    }

    onChanged?.call();
  }
}
